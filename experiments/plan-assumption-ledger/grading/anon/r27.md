**Goal:** Ship two feature packages that close an existing gap between three already-deployed systems (`session-reflect`, `context-baggage.py`, `compact-reinject.sh`, the RSI daily-review pipeline): (A) **RSI Reflect** — wire session-reflect's per-session `final.json` artifacts into the RSI daily-review prompt as the primary, pre-digested input instead of re-parsing raw transcripts; (B) **Self-Compact** — at ~30% of a 1M context window remaining, in opt-in autonomous contexts, Claude self-authors `tasks/mission.md` (with full live context, no detached subprocess) and ends its turn instead of asking the user what to do, and the existing post-compact hook asserts autonomous pickup instead of a passive "go re-read this" nudge.

**Tech Stack:** Bash (GNU coreutils, Linux-only — matches `session-reflect`'s existing constraint), Python 3 stdlib (`context-baggage.py` has no third-party deps today, keep it that way), `jq`, `python3 -m unittest` (matches `permission-ledger/tests/` convention).

```bash
#!/usr/bin/env bash
# list-todays-reflections.sh — deterministic session-signal enumerator for
# the RSI daily-review pipeline. Prints one line per session touched in the
# lookback window: `REFLECTED <final.json>` when session-reflect already
# produced a digest, `RAW_FALLBACK <jsonl>` when it didn't (pre-rollout
# session, or reflection errored/timed out). Keeps file-discovery logic out
# of the LLM's hands — glob/date judgment calls belong in code, not prompt.
set -euo pipefail
umask 077

if [[ "$(uname -s)" != "Linux" ]]; then
  echo "list-todays-reflections.sh: Linux only (uses GNU date/find)" >&2
  exit 2
fi

: "${REFLECT_ROOT:=$HOME/.claude/reflections}"
: "${PROJECTS_DIR:=$HOME/.claude/projects}"
: "${LOOKBACK_HOURS:=24}"

# --- REFLECTED: every final.json (parent + subagent) touched within lookback ---
reflected_sids=()
if [[ -d "$REFLECT_ROOT/sessions" ]]; then
  while IFS= read -r f; do
    [[ -n "$f" ]] || continue
    echo "REFLECTED $f"
    # sid is the directory name immediately containing final.json
    reflected_sids+=( "$(basename -- "$(dirname -- "$f")")" )
  done < <(find "$REFLECT_ROOT/sessions" -type f -name 'final.json' \
             -mtime "-$(( LOOKBACK_HOURS / 24 + 1 ))" 2>/dev/null \
           | while IFS= read -r cand; do
               # -mtime is day-granularity; re-check with a precise hour cutoff.
               mtime=$(stat -c '%Y' "$cand" 2>/dev/null || echo 0)
               cutoff=$(( $(date +%s) - LOOKBACK_HOURS * 3600 ))
               (( mtime >= cutoff )) && printf '%s\n' "$cand"
             done)
fi

# --- RAW_FALLBACK: transcripts touched within lookback with no REFLECTED sid ---
if [[ -d "$PROJECTS_DIR" ]]; then
  while IFS= read -r tp; do
    [[ -n "$tp" ]] || continue
    case "$tp" in
      */subagents/*) continue ;;  # excluded, same as existing prompt.md convention
    esac
    sid="$(basename -- "$tp" .jsonl)"
    skip=0
    for r in "${reflected_sids[@]:-}"; do
      [[ "$r" == "$sid" ]] && { skip=1; break; }
    done
    (( skip == 1 )) && continue
    echo "RAW_FALLBACK $tp"
  done < <(find "$PROJECTS_DIR" -type f -name '*.jsonl' -mmin "-$(( LOOKBACK_HOURS * 60 ))" 2>/dev/null)
fi
```

#!/usr/bin/env python3
"""
UserPromptSubmit hook: at high context fill, either nudges the user to
choose a handoff strategy (default) or, when self-compact is opted in for
this project, directs Claude to self-author tasks/mission.md and end its
turn without asking.

Context window detection (in priority order):
    1. CLAUDE_CONTEXT_MAX env var (explicit override, in tokens)
    2. Model ID prefix — Opus/Sonnet 4.x run with 1M in this setup
    3. Observed usage — if any assistant turn has exceeded 200k tokens,
       the session is 1M, regardless of model name
    4. Default 200k
"""
import json
import os
import sys
import time

BASE_MAX = 200_000
ONE_M = 1_000_000
# Raised from 0.25 -> 0.70 on 2026-08-08 after measuring 95.5% cache-hit
# ratio across 7 days / $3,073 spend (probes/analyze_sessions.py). At that
# hit ratio, holding baggage is cheap and resetting is expensive
# (cache_write ~$5-10 per cold event). Nudge should fire when handoff to
# a fresh session starts to pay off, not on any drift. 0.70 fill == 30%
# of the window remaining — the self-compact trigger point.
THRESHOLD = 0.70
MESSAGE_INTERVAL = 5

# Self-compact freshness window: if tasks/mission.md was written by an
# earlier self-compact firing in this same session, don't re-fire the
# directive (and force another commit/write cycle) until it goes stale.
# Mirrors scripts/idle-handoff.sh's IDLE_HANDOFF_MISSION_MAX_AGE_SECONDS.
SELF_COMPACT_MISSION_FRESH_SECONDS = 600

# Claude Code model IDs that run at 1M context by default. Prefixes, so
# dated variants (claude-opus-4-1-20250805) match too.
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)

# Explicit 1M marker in the model ID, e.g. `claude-opus-5[1m]`. Checked
# before the prefix list so it keeps working across model generations
# without another edit here — the prefix list going stale is exactly how
# this hook came to report 89% full during a session that was at 18% of a
# 1M window (every reading was fullness-of-200k).
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model: str | None, observed_max: int) -> int:
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def self_compact_enabled(cwd: str) -> bool:
    """Opt-in check: an armed ralph loop or an explicit marker file.
    Absence of both means self-compact never fires for this project —
    the ask-the-user nudge is the only behavior a normal interactive
    session ever sees."""
    if not cwd or not os.path.isdir(cwd):
        return False
    ralph_state = os.path.join(cwd, ".claude", "ralph-loop.local.md")
    marker = os.path.join(cwd, "tasks", ".self-compact-enabled")
    return os.path.isfile(ralph_state) or os.path.isfile(marker)


def mission_is_fresh(cwd: str, max_age_seconds: int) -> bool:
    mission_path = os.path.join(cwd, "tasks", "mission.md")
    if not os.path.isfile(mission_path):
        return False
    age = time.time() - os.path.getmtime(mission_path)
    return age < max_age_seconds


def build_message(fill_pct: float, user_msg_count: int, max_tokens: int, self_compact_active: bool) -> str:
    pct_display = int(fill_pct * 100)
    window_display = f"{max_tokens // 1000}k"
    if self_compact_active:
        return (
            f"Context is {pct_display}% full ({user_msg_count} messages in this session; "
            f"window: {window_display} tokens). Self-compact is ACTIVE for this project "
            "(an armed ralph loop or tasks/.self-compact-enabled is present). "
            "Do NOT ask the user what to do. This turn: "
            "(1) finish or checkpoint the current atomic unit of work; "
            "(2) write tasks/mission.md YOURSELF now, following the structure in "
            "~/.claude/skills/session-reflect/prompts/mission.md (Done / Next / Blocked / "
            "Open questions / State on disk) — you hold full live context right now, which "
            "produces a better mission.md than the detached reflection worker session-reflect "
            "uses elsewhere; "
            "(3) commit any pending changes; "
            "(4) end this turn cleanly. Context reset and autonomous pickup from "
            "tasks/mission.md are handled by the session-lifecycle hooks "
            "(skills/orchestrate/hooks/compact-reinject.sh) — do not wait for permission "
            "to continue after reset; resume the top item in mission.md's Next section "
            "immediately."
        )
    return (
        f"Context is {pct_display}% full ({user_msg_count} messages in this session; "
        f"window: {window_display} tokens). "
        "At this fill, a fresh-session handoff via tasks/mission.md is usually cheaper "
        "than an in-place /compact: --fork-session on a warm cache measured ~77% hit "
        "(session-reflect empirical), while /compact invalidates the prefix and forces "
        "cache_write on subsequent turns. "
        "Suggest to the user: (a) write tasks/mission.md (done / next / blocked / open "
        "questions), (b) start a fresh session in the same cwd and load it, OR (c) "
        "continue here and accept the drift. Ask which."
    )


def main():
    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    transcript_path = data.get("transcript_path", "")
    if not transcript_path or not os.path.exists(transcript_path):
        sys.exit(0)

    cwd = data.get("cwd", "")
    file_size = os.path.getsize(transcript_path)

    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue

            obj_type = obj.get("type")

            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = (
                        usage.get("input_tokens", 0)
                        + usage.get("cache_creation_input_tokens", 0)
                        + usage.get("cache_read_input_tokens", 0)
                    )
                    if total > observed_max:
                        observed_max = total

    if user_msg_count == 0 or user_msg_count % MESSAGE_INTERVAL != 0:
        sys.exit(0)

    max_tokens = detect_max_tokens(last_model, observed_max)

    if last_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )
        fill_pct = total_tokens / max_tokens
    else:
        fill_pct = file_size / (max_tokens * 4)

    if fill_pct < THRESHOLD:
        sys.exit(0)

    self_compact_active = self_compact_enabled(cwd)
    if self_compact_active and mission_is_fresh(cwd, SELF_COMPACT_MISSION_FRESH_SECONDS):
        # Already handled this cycle — don't force a redundant mission.md
        # rewrite + commit every 5 messages while nothing new has happened.
        sys.exit(0)

    message = build_message(fill_pct, user_msg_count, max_tokens, self_compact_active)
    print(json.dumps({"hookSpecificOutput": {"hookEventName": "UserPromptSubmit", "additionalContext": message}}))


if __name__ == "__main__":
    main()
