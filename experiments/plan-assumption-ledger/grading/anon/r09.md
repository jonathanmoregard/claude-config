```python
#!/usr/bin/env python3
"""
UserPromptSubmit hook: warns when context is over THRESHOLD with potentially
stale/off-topic content. Fires every MESSAGE_INTERVAL user messages once
threshold is met.

Context window detection (in priority order):
    1. CLAUDE_CONTEXT_MAX env var (explicit override, in tokens)
    2. Model ID prefix — Opus/Sonnet 4.x run with 1M in this setup
    3. Observed usage — if any assistant turn has exceeded 200k tokens,
       the session is 1M, regardless of model name
    4. Default 200k

compute_fill() is also imported by
skills/self-compact/hooks/context-threshold-hook.py so both the
interactive nudge (here) and the autonomous reset trigger share one tuned
THRESHOLD and one token-accounting implementation.
"""
import json
import os
import sys

BASE_MAX = 200_000
ONE_M = 1_000_000
# Raised from 0.25 -> 0.70 on 2026-08-08 after measuring 95.5% cache-hit
# ratio across 7 days / $3,073 spend (probes/analyze_sessions.py). At that
# hit ratio, holding baggage is cheap and resetting is expensive
# (cache_write ~$5-10 per cold event). Nudge should fire when handoff to
# a fresh session starts to pay off, not on any drift.
THRESHOLD = 0.70
MESSAGE_INTERVAL = 5

# Claude Code model IDs that run at 1M context by default. Prefixes, so
# dated variants (claude-opus-4-1-20250805) match too.
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)

# Explicit 1M marker in the model ID, e.g. `claude-opus-5[1m]`. Checked
# before the prefix list so it keeps working across model generations
# without another edit here — the prefix list going stale is exactly how
# this hook came to report 89% full during a session that was at 18% of a
# 1M window (every reading was fullness-of-200k).
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model: str | None, observed_max: int) -> int:
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def compute_fill(transcript_path: str):
    """Read transcript_path and return (fill_pct, total_tokens, max_tokens,
    user_msg_count) from the last assistant usage entry, or None if the
    transcript is missing or carries no usable data.

    Pure function, no side effects — safe to import from another hook.
    """
    if not transcript_path or not os.path.exists(transcript_path):
        return None

    file_size = os.path.getsize(transcript_path)
    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue

            obj_type = obj.get("type")

            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = (
                        usage.get("input_tokens", 0)
                        + usage.get("cache_creation_input_tokens", 0)
                        + usage.get("cache_read_input_tokens", 0)
                    )
                    if total > observed_max:
                        observed_max = total

    max_tokens = detect_max_tokens(last_model, observed_max)

    if last_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )
        fill_pct = total_tokens / max_tokens
    else:
        total_tokens = 0
        fill_pct = file_size / (max_tokens * 4)

    return fill_pct, total_tokens, max_tokens, user_msg_count


def main():
    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    transcript_path = data.get("transcript_path", "")
    result = compute_fill(transcript_path)
    if result is None:
        sys.exit(0)
    fill_pct, total_tokens, max_tokens, user_msg_count = result

    if user_msg_count == 0 or user_msg_count % MESSAGE_INTERVAL != 0:
        sys.exit(0)

    if fill_pct < THRESHOLD:
        sys.exit(0)

    pct_display = int(fill_pct * 100)
    message = (
        f"Context is {pct_display}% full ({user_msg_count} messages in this session; "
        f"window: {max_tokens // 1000}k tokens). "
        "At this fill, a fresh-session handoff via tasks/mission.md is usually cheaper "
        "than an in-place /compact: --fork-session on a warm cache measured ~77% hit "
        "(session-reflect empirical), while /compact invalidates the prefix and forces "
        "cache_write on subsequent turns. "
        "Suggest to the user: (a) write tasks/mission.md (done / next / blocked / open "
        "questions), (b) start a fresh session in the same cwd and load it, OR (c) "
        "continue here and accept the drift. Ask which."
    )
    print(json.dumps({"hookSpecificOutput": {"hookEventName": "UserPromptSubmit", "additionalContext": message}}))


if __name__ == "__main__":
    main()
```

#!/usr/bin/env python3
"""
UserPromptSubmit hook: the autonomous half of context-fill handling.

context-baggage.py (also on UserPromptSubmit) asks a human which of
(a) write mission.md, (b) start fresh, (c) continue — every MESSAGE_INTERVAL
turns once fill crosses THRESHOLD. That's right for an attended session, but
an autonomous run (ralph-loop armed via .claude/ralph-loop.local.md, see
skills/orchestrate/SKILL.md Phase 0) has nobody to answer. This hook fires
on EVERY turn (no MESSAGE_INTERVAL throttle — ralph-loop's Stop hook
re-injects a new user turn per iteration, so every iteration gets a fresh
chance to see the directive) and, once an autonomous run crosses the same
THRESHOLD, injects an explicit hand-off directive instead of a question.

The directive tells the model to: write tasks/mission.md + tasks/progress.md,
then run `rm .claude/ralph-loop.local.md` (the ralph-loop plugin's own
documented kill-switch — see docs/superpowers/plans/2026-08-02-sigil-gated-
autonomous-orchestration.md Task 4 Step 5) to end this leg without
re-arming. A fresh `claude -p` process picks the mission back up — see
scripts/self-compact-supervisor.sh and skills/self-compact/SKILL.md.

This hook never blocks, never edits anything itself — it only prints
additionalContext. All actual state changes are the model's own tool calls,
so the normal permission/audit trail applies.
"""
import importlib.util
import json
import os
import sys

RALPH_STATE_REL = os.path.join(".claude", "ralph-loop.local.md")
COMPLETE_MARKER_REL = os.path.join("tasks", ".mission-complete")


def _load_context_baggage():
    path = os.environ.get(
        "CONTEXT_BAGGAGE_PATH", os.path.expanduser("~/.claude/context-baggage.py")
    )
    spec = importlib.util.spec_from_file_location("context_baggage", path)
    if spec is None or spec.loader is None:
        return None
    module = importlib.util.module_from_spec(spec)
    try:
        spec.loader.exec_module(module)
    except Exception:
        return None
    return module


def main():
    # Recursion break, mirrors skills/session-reflect/reflect.sh's
    # REFLECT_SUBPROCESS sentinel: reflect.sh's `mission` mode invokes
    # `claude -p --resume ... --tools ""` from the SAME cwd this hook would
    # otherwise inspect, and that subprocess fires its own UserPromptSubmit.
    if os.environ.get("REFLECT_SUBPROCESS", "0") == "1":
        sys.exit(0)

    try:
        payload = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    cwd = payload.get("cwd") or ""
    transcript_path = payload.get("transcript_path") or ""
    if not cwd or not os.path.isdir(cwd):
        sys.exit(0)

    ralph_state = os.path.join(cwd, RALPH_STATE_REL)
    if not os.path.isfile(ralph_state):
        sys.exit(0)  # not an autonomous run — context-baggage.py's human nudge already covers this

    complete_marker = os.path.join(cwd, COMPLETE_MARKER_REL)
    if os.path.isfile(complete_marker):
        sys.exit(0)  # mission already marked done; let the sigil/Stop path finish normally

    cb = _load_context_baggage()
    if cb is None:
        sys.exit(0)  # fail silent — never block an autonomous run on our own import error

    result = cb.compute_fill(transcript_path)
    if result is None:
        sys.exit(0)
    fill_pct, _total_tokens, max_tokens, _user_msg_count = result

    if fill_pct < cb.THRESHOLD:
        sys.exit(0)

    pct_display = int(fill_pct * 100)
    message = (
        f"[self-compact] Context is {pct_display}% full ({max_tokens // 1000}k window) "
        "during an ARMED autonomous run (.claude/ralph-loop.local.md present). "
        "Hand off to a fresh leg now, in this turn:\n"
        "1. Update tasks/mission.md and tasks/progress.md with current done / next / "
        "blocked / open-questions state — concise enough for a fresh process with no "
        "memory of this conversation to resume from cold.\n"
        "2. Run: rm .claude/ralph-loop.local.md "
        "(the ralph-loop plugin's own documented kill-switch — disarms this leg, does "
        "NOT affect the mission itself).\n"
        "3. End your turn. Do not emit the completion sigil unless the mission is "
        "genuinely, verifiably done. If it IS done, also run: "
        "touch tasks/.mission-complete before stopping, so a supervising process knows "
        "not to start another leg.\n"
        "A fresh process will read tasks/mission.md and continue automatically — this "
        "is expected handoff behavior, not an error condition."
    )
    print(json.dumps({"hookSpecificOutput": {"hookEventName": "UserPromptSubmit", "additionalContext": message}}))


if __name__ == "__main__":
    main()

#!/usr/bin/env bash
# self-compact-supervisor.sh — outer process-restart loop for autonomous
# orchestrate/ralph-loop runs.
#
# Why this exists: Claude Code hooks (Stop, PreCompact, UserPromptSubmit)
# can change LOOP CONTROL FLOW within a running `claude` process, but they
# cannot shrink that process's context — only a brand-new process gets a
# genuinely empty context window. The ralph-loop plugin's in-session Stop
# hook re-feeds the SAME prompt into the SAME process, so context only
# grows across iterations. This script is the missing outer loop: each
# `claude -p` invocation below is a genuinely fresh process. tasks/mission.md
# (see skills/session-reflect/ and
# skills/self-compact/hooks/context-threshold-hook.py) is the sole handoff
# channel between legs — nothing else survives a restart.
#
# Usage: self-compact-supervisor.sh <worktree-root> [max-legs]
#
# Each leg:
#   1. Runs `claude -p` with a bootstrap prompt that reads tasks/mission.md,
#      tasks/session-constraints.md, tasks/progress.md, and re-arms the
#      ralph-loop (orchestrate SKILL.md Phase 0 step 4) if not already armed.
#   2. Waits for that process to exit — via the ralph-loop sigil (real
#      completion), the self-compact context-threshold hand-off (leg
#      boundary, NOT completion), or a crash.
#   3. Checks tasks/.mission-complete. Present -> done, exit 0. Absent ->
#      spin up another leg.
#
# Permission mode: acceptEdits, not bypassPermissions. bypassPermissions
# propagates to subagents and cannot be cancelled mid-run (see
# skills/locked-down-headless-agent/SKILL.md anti-patterns) — acceptEdits
# auto-accepts file edits while the existing settings.json deny list
# (force-push, rm -rf, sudo, etc.) still gates genuinely dangerous ops.
set -euo pipefail

WORKTREE="${1:?usage: self-compact-supervisor.sh <worktree-root> [max-legs]}"
MAX_LEGS="${2:-20}"
CLAUDE_BIN="${CLAUDE_BIN:-claude}"
MODEL="${SELF_COMPACT_MODEL:-claude-opus-5}"

cd "$WORKTREE"
LOG_DIR="$WORKTREE/logs/self-compact"
mkdir -p "$LOG_DIR"

if [[ ! -f tasks/mission.md ]]; then
  echo "self-compact-supervisor: tasks/mission.md missing — run orchestrate Phase 0 first" >&2
  exit 1
fi

BOOTSTRAP='Read tasks/mission.md, tasks/session-constraints.md, and tasks/progress.md, then continue from where progress.md left off (empty progress.md = start at the first unchecked acceptance criterion). If .claude/ralph-loop.local.md is absent, re-arm it per orchestrate SKILL.md Phase 0 step 4 (bash the setup script, never exec it directly) before proceeding. Orchestrate via subagents; do not implement inline; paste tasks/session-constraints.md verbatim into every subagent briefing.'

for (( leg=1; leg<=MAX_LEGS; leg++ )); do
  LOG="$LOG_DIR/leg-$(date -u +%Y%m%dT%H%M%SZ)-$leg.log"
  echo "self-compact-supervisor: starting leg $leg/$MAX_LEGS, log=$LOG"
  "$CLAUDE_BIN" -p "$BOOTSTRAP" \
    --model "$MODEL" \
    --permission-mode acceptEdits \
    >"$LOG" 2>&1 || true

  if [[ -f tasks/.mission-complete ]]; then
    echo "self-compact-supervisor: mission complete after leg $leg"
    exit 0
  fi
  echo "self-compact-supervisor: leg $leg ended without completion marker — restarting"
done

echo "self-compact-supervisor: hit MAX_LEGS=$MAX_LEGS without completion — see pending_for_human.md" >&2
{
  echo ""
  echo "## self-compact-supervisor exhausted legs — $(date -u +%Y-%m-%dT%H:%M:%SZ)"
  echo "Worktree: $WORKTREE"
  echo "Legs run: $MAX_LEGS"
  echo "Action needed: inspect tasks/mission.md and tasks/progress.md, decide whether to raise max-legs or intervene manually."
} >> pending_for_human.md
exit 1
```

name: self-compact
description: Use when an autonomous orchestrate/ralph-loop run needs to survive past ~70% context fill without a human present to decide whether to compact, clear, or continue. # Self-Compact

Ralph-loop's in-session Stop hook re-feeds the same prompt into the SAME
process — context only grows across iterations, it never resets. Claude
Code hooks can change loop control flow but cannot shrink a running
process's context; only a brand-new process gets a genuinely empty window.
Self-compact is the missing outer loop: when an armed autonomous run
crosses ~70% context fill (~30% of a 1M window remaining), the current leg
writes a fresh `tasks/mission.md`, disarms itself, and a supervisor spins
up a brand-new `claude -p` process that picks the mission back up. No
human needed at the boundary — that's the whole point.

## Composition — don't reimplement these

| Piece | Owner |
|---|---|
| Mission-file writer | The LIVE model updates `tasks/mission.md`/`tasks/progress.md` itself in response to the hand-off directive — cheaper and more accurate than shelling out to `reflect.sh mission`'s cold Haiku resume, since the live model already has full up-to-date state |
| In-session iteration within one leg | ralph-loop plugin (official) — unmodified |
| Leg-boundary disarm | ralph-loop's own documented `rm .claude/ralph-loop.local.md` kill-switch (verified: `docs/superpowers/plans/2026-08-02-sigil-gated-autonomous-orchestration.md` Task 4 Step 5) |
| Cross-leg persistence | `tasks/mission.md`, `tasks/progress.md`, `tasks/session-constraints.md` — the same files orchestrate already maintains |
| Real process reset | `scripts/self-compact-supervisor.sh` — outer loop, one fresh `claude -p` per leg |
| Fill computation | `context-baggage.py:compute_fill()` — single source of truth shared with the interactive nudge, same tuned `THRESHOLD = 0.70` |

## How a leg ends

1. `skills/self-compact/hooks/context-threshold-hook.py` fires on every
   `UserPromptSubmit`. No-ops unless `.claude/ralph-loop.local.md` is
   present (autonomous run armed) AND fill ≥ `THRESHOLD` AND
   `tasks/.mission-complete` is absent.

## Running under the supervisor

```bash
bash scripts/self-compact-supervisor.sh /path/to/worktree 20
```

Runs up to 20 legs. Each leg is a genuinely fresh `claude -p --model
claude-opus-5 --permission-mode acceptEdits` process — never

`skills/locked-down-headless-agent/SKILL.md`). Existing `settings.json`
deny rules (force-push, rm -rf, sudo, etc.) remain the safety rail
regardless of permission mode. Logs land in `logs/self-compact/`. On
`--max-legs` exhaustion without a completion marker, appends to
`pending_for_human.md` and exits 1 — a stop condition to investigate, not
a silent infinite retry.
