- Self-compact fires ONLY when `.claude/ralph-loop.local.md` has `active: true` AND `.claude/self-compact.enabled` exists in the same directory — no exceptions, no default-on.

```bash
#!/usr/bin/env bash
# session-index.sh — deterministic session listing for RSI daily-review.
#
# Prints one JSON object per line (JSONL) for every Claude Code session
# touched in the last WINDOW_HOURS hours, across ~/.claude/projects/*/*.jsonl.
# Each line reports whether a session-reflect artifact (final.json,
# modes.sessionend.status == "complete") exists for that session, so the RSI
# daily-review prompt can read the compact reflection instead of re-parsing
# the raw transcript — this closes the loop that
# proposals/manual/from 4th of august - rsi token efficiency.md built
# session-reflect for.
#
# Usage: session-index.sh [--window-hours N] [--projects-dir DIR] [--reflect-root DIR]
#
# Output (one JSON object per line):
#   {"session_id","transcript_path","project_dir",
#    "has_reflection":true|false, "reflection_path":"..."|null,
#    "reflection_status":"complete"|"error"|"timeout"|"missing"}
set -euo pipefail

WINDOW_HOURS=24
PROJECTS_DIR="$HOME/.claude/projects"
REFLECT_ROOT="${REFLECT_ROOT:-$HOME/.claude/reflections}"

while [[ $# -gt 0 ]]; do
  case "$1" in
    --window-hours) WINDOW_HOURS="$2"; shift 2 ;;
    --projects-dir) PROJECTS_DIR="$2"; shift 2 ;;
    --reflect-root) REFLECT_ROOT="$2"; shift 2 ;;
    *) echo "session-index.sh: unknown arg: $1" >&2; exit 2 ;;
  esac
done

[[ "$WINDOW_HOURS" =~ ^[0-9]+$ ]] || { echo "session-index.sh: --window-hours must be an integer" >&2; exit 2; }
[[ -d "$PROJECTS_DIR" ]] || exit 0

find "$PROJECTS_DIR" -maxdepth 1 -mindepth 1 -type d | while IFS= read -r project_dir; do
  find "$project_dir" -maxdepth 1 -type f -name '*.jsonl' -mmin "-$(( WINDOW_HOURS * 60 ))" \
    | while IFS= read -r transcript; do
    sid=$(basename "$transcript" .jsonl)

    manifest="$REFLECT_ROOT/sessions/$sid/manifest.json"
    reflection_status="missing"
    has_reflection=false
    reflection_path=""
    if [[ -f "$manifest" ]]; then
      status=$(jq -r '.modes.sessionend.status // "missing"' "$manifest" 2>/dev/null || echo "missing")
      reflection_status="$status"
      if [[ "$status" == "complete" ]]; then
        fp="$REFLECT_ROOT/sessions/$sid/final.json"
        if [[ -f "$fp" ]]; then
          has_reflection=true
          reflection_path="$fp"
        fi
      fi
    fi

    jq -cn \
      --arg sid "$sid" \
      --arg tp "$transcript" \
      --arg pd "$project_dir" \
      --argjson has "$has_reflection" \
      --arg rp "$reflection_path" \
      --arg rs "$reflection_status" \
      '{session_id:$sid, transcript_path:$tp, project_dir:$pd,
        has_reflection:$has,
        reflection_path:(if $has then $rp else null end),
        reflection_status:$rs}'
  done
done
```

#!/usr/bin/env python3
"""
Shared context-fill computation for Claude Code transcripts. Used by
context-baggage.py (interactive nudge) and self-compact.sh (autonomous
reset trigger) so both agree on exactly one definition of "how full is
context" — a threshold drift between the two would mean self-compact
fires at a different point than what the interactive nudge already told
the user to expect.
"""
from __future__ import annotations

import argparse
import json
import os
import sys

BASE_MAX = 200_000
ONE_M = 1_000_000
DEFAULT_THRESHOLD = 0.70  # matches context-baggage.py's empirically-set nudge point
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model: str | None, observed_max: int) -> int:
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def compute_fill(transcript_path: str) -> dict:
    """Scan a transcript JSONL and return fill stats.

    Raises FileNotFoundError if the transcript does not exist — callers
    decide how to handle that (context-baggage.py exits 0 silently per
    hook convention; the CLI wrapper below exits 1).
    """
    if not transcript_path or not os.path.exists(transcript_path):
        raise FileNotFoundError(transcript_path)

    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue

            obj_type = obj.get("type")
            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = (
                        usage.get("input_tokens", 0)
                        + usage.get("cache_creation_input_tokens", 0)
                        + usage.get("cache_read_input_tokens", 0)
                    )
                    if total > observed_max:
                        observed_max = total

    max_tokens = detect_max_tokens(last_model, observed_max)

    if last_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )
        fill_pct = total_tokens / max_tokens
    else:
        total_tokens = 0
        file_size = os.path.getsize(transcript_path)
        fill_pct = file_size / (max_tokens * 4)

    return {
        "fill_pct": fill_pct,
        "max_tokens": max_tokens,
        "total_tokens": total_tokens,
        "user_msg_count": user_msg_count,
        "last_model": last_model,
    }


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--transcript-path", required=True)
    parser.add_argument("--threshold", type=float, default=DEFAULT_THRESHOLD)
    args = parser.parse_args()

    try:
        stats = compute_fill(args.transcript_path)
    except FileNotFoundError:
        print(json.dumps({"error": "transcript_not_found", "transcript_path": args.transcript_path}))
        return 1

    stats["crossed_threshold"] = stats["fill_pct"] >= args.threshold
    stats["threshold"] = args.threshold
    print(json.dumps(stats))
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env bash
# self-compact.sh — proactive context reset for ralph-loop-armed autonomous
# sessions. Intended to run on a systemd user timer (systemd/self-compact.timer).
#
# For each LIVE session whose cwd has BOTH an active ralph-loop state file
# (.claude/ralph-loop.local.md, active: true) AND an explicit opt-in marker
# (.claude/self-compact.enabled), computes context fill via context-fill.py.
# At >= SELF_COMPACT_THRESHOLD (default 0.70, i.e. ~30% of a 1M window
# remaining — matches context-baggage.py's empirically-set nudge threshold),
# fires a reset: write fresh tasks/mission.md, gracefully kill the process,
# relaunch headless in the same cwd. ralph-loop.local.md is untouched on
# disk, so the Stop hook keeps gating the SAME loop in the fresh process;
# compact-reinject.sh (SessionStart startup|compact|clear) already
# re-injects doctrine + points at the fresh mission.md — no new resume-side
# code needed.
#
# Safety boundaries (do not relax without updating the opt-in contract):
#   - NEVER touches a directory without BOTH files present. Arming a ralph
#     loop alone does NOT opt in.
#   - Global kill switch: touch ~/.claude/self-compact.disable
#   - Debounced per session_id (SELF_COMPACT_RESET_DIR/<sid>.json) — fires
#     at most once per session.
#   - Graceful SIGTERM first; SIGKILL only after SELF_COMPACT_KILL_GRACE_SECONDS.
#
# Linux-only: uses /proc/<pid>/cwd, pgrep, GNU find -mmin.
set -uo pipefail
umask 077

if [[ "$(uname -s)" != "Linux" ]]; then
  echo "self-compact: Linux only (uses /proc, pgrep, GNU find)" >&2
  exit 2
fi

: "${REFLECT_ROOT:=$HOME/.claude/reflections}"
: "${SELF_COMPACT_ROOT:=$REFLECT_ROOT/self-compact}"
: "${SELF_COMPACT_THRESHOLD:=0.70}"
: "${SELF_COMPACT_KILL_GRACE_SECONDS:=15}"
: "${SELF_COMPACT_DISABLE_FLAG:=$HOME/.claude/self-compact.disable}"
: "${SELF_COMPACT_LOG_DIR:=$SELF_COMPACT_ROOT/logs}"
: "${SELF_COMPACT_RESET_DIR:=$SELF_COMPACT_ROOT/resets}"
: "${PROJECTS_DIR:=$HOME/.claude/projects}"
: "${REFLECT_SH:=$HOME/.claude/skills/session-reflect/reflect.sh}"
: "${CONTEXT_FILL_PY:=$HOME/.claude/context-fill.py}"

DRY_RUN=0
[[ "${1:-}" == "--dry-run" ]] && DRY_RUN=1

mkdir -p "$SELF_COMPACT_LOG_DIR" "$SELF_COMPACT_RESET_DIR" 2>/dev/null || exit 0

log() {
  printf '%s %s\n' "$(date -u +%Y-%m-%dT%H:%M:%SZ)" "$*" >> "$SELF_COMPACT_LOG_DIR/self-compact.log"
}

if [[ -f "$SELF_COMPACT_DISABLE_FLAG" ]]; then
  log "disabled (flag file exists)"
  exit 0
fi
[[ -d "$PROJECTS_DIR" ]] || exit 0

# find_live_pid <cwd> — PID of a running `claude` process whose /proc cwd
# resolves exactly to <cwd>. Deliberately stricter than idle-handoff's
# `--resume` command-line grep: this PID may get killed, so a false
# positive is far worse than a false negative.
find_live_pid() {
  local target_cwd="$1" pid link
  target_cwd="$(cd -- "$target_cwd" 2>/dev/null && pwd -P)" || { printf ''; return 0; }
  for pid in $(pgrep -x claude 2>/dev/null || true); do
    link="$(readlink -f "/proc/$pid/cwd" 2>/dev/null || true)"
    if [[ -n "$link" && "$link" == "$target_cwd" ]]; then
      printf '%s' "$pid"
      return 0
    fi
  done
  printf ''
}

for project_dir in "$PROJECTS_DIR"/*/; do
  [[ -d "$project_dir" ]] || continue
  transcript=$(find "$project_dir" -maxdepth 1 -type f -name '*.jsonl' \
    -printf '%T@ %p\n' 2>/dev/null | sort -rn | head -1 | awk '{print $2}')
  [[ -n "$transcript" && -f "$transcript" ]] || continue
  sid=$(basename "$transcript" .jsonl)

  [[ -f "$SELF_COMPACT_RESET_DIR/$sid.json" ]] && continue

  candidate_cwd=$(tac "$transcript" 2>/dev/null | \
    jq -r 'select(type=="object") | .cwd // empty' 2>/dev/null | head -1)
  [[ -n "$candidate_cwd" && -d "$candidate_cwd" ]] || continue

  ralph_state="$candidate_cwd/.claude/ralph-loop.local.md"
  optin_marker="$candidate_cwd/.claude/self-compact.enabled"
  [[ -f "$ralph_state" && -f "$optin_marker" ]] || continue
  grep -q '^active: true' "$ralph_state" 2>/dev/null || continue

  pid=$(find_live_pid "$candidate_cwd")
  [[ -n "$pid" ]] || continue

  fill_json=$(python3 "$CONTEXT_FILL_PY" --transcript-path "$transcript" \
    --threshold "$SELF_COMPACT_THRESHOLD" 2>/dev/null) || { log "context-fill failed for sid=$sid"; continue; }
  crossed=$(jq -r '.crossed_threshold // false' <<<"$fill_json")
  fill_pct=$(jq -r '.fill_pct // 0' <<<"$fill_json")
  [[ "$crossed" == "true" ]] || continue

  if (( DRY_RUN )); then
    log "WOULD RESET sid=$sid cwd=$candidate_cwd pid=$pid fill_pct=$fill_pct"
    continue
  fi

  log "FIRE-NOT-YET-IMPLEMENTED sid=$sid — dry-run-only build (Task 4); see Task 5"
done
