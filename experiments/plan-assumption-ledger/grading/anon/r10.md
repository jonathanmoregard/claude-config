```python
#!/usr/bin/env python3
"""Shared context-window / fill-percentage detection.

Extracted from context-baggage.py so the same detection logic (context
window sizing, fill %, cwd derivation) is used by both the interactive
70%-threshold nudge (context-baggage.py, UserPromptSubmit) and the
proactive 30%-threshold autonomous mission-write (self-compact-hook.py,
Stop). Not executed standalone.
"""
from __future__ import annotations

import json
import os

BASE_MAX = 200_000
ONE_M = 1_000_000

# Claude Code model IDs that run at 1M context by default. Prefixes, so
# dated variants (claude-opus-4-1-20250805) match too.
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)

# Explicit 1M marker in the model ID, e.g. `claude-opus-5[1m]`. Checked
# before the prefix list so it keeps working across model generations
# without another edit here.
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model: str | None, observed_max: int) -> int:
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def scan_transcript(transcript_path: str):
    """Returns (user_msg_count, last_usage, last_model, observed_max, file_size)."""
    file_size = os.path.getsize(transcript_path)
    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue

            obj_type = obj.get("type")

            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = (
                        usage.get("input_tokens", 0)
                        + usage.get("cache_creation_input_tokens", 0)
                        + usage.get("cache_read_input_tokens", 0)
                    )
                    if total > observed_max:
                        observed_max = total

    return user_msg_count, last_usage, last_model, observed_max, file_size


def compute_fill_pct(transcript_path: str):
    """Returns (fill_pct, max_tokens, last_model, last_usage), or None if the
    transcript is missing/unreadable."""
    if not transcript_path or not os.path.exists(transcript_path):
        return None

    user_msg_count, last_usage, last_model, observed_max, file_size = scan_transcript(transcript_path)
    max_tokens = detect_max_tokens(last_model, observed_max)

    if last_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )
        fill_pct = total_tokens / max_tokens
    else:
        fill_pct = file_size / (max_tokens * 4)

    return fill_pct, max_tokens, last_model, last_usage


def extract_cwd_from_transcript(transcript_path: str) -> str:
    """Most-recent `cwd` field found scanning the transcript backward.
    Mirrors scripts/idle-handoff.sh's `tac transcript | jq .cwd | head -1`
    derivation — the Stop hook payload's top level does not reliably carry
    cwd (verified against step-back-detector.py, which derives it the same
    way from individual transcript records rather than trusting a top-level
    payload field)."""
    try:
        with open(transcript_path, "r", errors="replace") as f:
            lines = f.readlines()
    except OSError:
        return ""
    for line in reversed(lines):
        line = line.strip()
        if not line:
            continue
        try:
            obj = json.loads(line)
        except json.JSONDecodeError:
            continue
        cwd = obj.get("cwd")
        if cwd:
            return cwd
    return ""
```

```python
#!/usr/bin/env python3
"""Stop hook: proactively write tasks/mission.md when context fill crosses
a low, autonomous-safe threshold (default 30%), so mission.md is always
fresh whenever Claude Code's own auto-compact, a later manual /compact, or
/clear resets the session.

Complements context-baggage.py's 70%-threshold interactive nudge, which
asks the user which of three options to take. This hook does not ask —
autonomous / unattended sessions (ralph-loop runs, long orchestrated
sessions) have no user present to answer, so mission.md must already be
current by the time any reset happens. It reuses reflect.sh's existing
`mission` mode unmodified (the same mode idle-handoff.sh and reflect.sh's
own "2nd-compact defense" already call) — this script is purely a new,
earlier, non-idle-dependent TRIGGER, not a new mission-writing engine.

Fires on every Stop event (every assistant turn, interactive or
autonomous-loop). Cheap to skip: the freshness check below means the
actual reflect.sh subprocess only runs once per SELF_COMPACT_MISSION_MAX_AGE_SECONDS
window, not on every turn once above threshold.
"""
import json
import os
import subprocess
import sys
import time

import context_fill

THRESHOLD = float(os.environ.get("SELF_COMPACT_THRESHOLD", "0.30"))
MISSION_MAX_AGE_SECONDS = int(os.environ.get("SELF_COMPACT_MISSION_MAX_AGE_SECONDS", str(15 * 60)))
DISABLE_FLAG = os.path.expanduser(
    os.environ.get("SELF_COMPACT_DISABLE_FLAG", "~/.claude/self-compact.disable")
)
REFLECT_ROOT = os.path.expanduser(os.environ.get("REFLECT_ROOT", "~/.claude/reflections"))
REFLECT_SH = os.path.expanduser("~/.claude/skills/session-reflect/reflect.sh")
LOG_PATH = os.path.join(REFLECT_ROOT, "self-compact-logs", "self-compact.log")


def log(msg: str) -> None:
    try:
        os.makedirs(os.path.dirname(LOG_PATH), exist_ok=True)
        with open(LOG_PATH, "a") as f:
            f.write(f"{time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())} {msg}\n")
    except OSError:
        pass


def main():
    if os.path.exists(DISABLE_FLAG):
        sys.exit(0)

    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    session_id = data.get("session_id", "")
    transcript_path = data.get("transcript_path", "")
    if not session_id or not transcript_path or not os.path.exists(transcript_path):
        sys.exit(0)

    result = context_fill.compute_fill_pct(transcript_path)
    if result is None:
        sys.exit(0)
    fill_pct, _max_tokens, _last_model, _last_usage = result
    if fill_pct < THRESHOLD:
        sys.exit(0)

    cwd = context_fill.extract_cwd_from_transcript(transcript_path)
    if not cwd or not os.path.isdir(cwd):
        sys.exit(0)

    mission_path = os.path.join(cwd, "tasks", "mission.md")
    if os.path.exists(mission_path):
        age = time.time() - os.path.getmtime(mission_path)
        if age < MISSION_MAX_AGE_SECONDS:
            sys.exit(0)

    payload = json.dumps({
        "session_id": session_id,
        "transcript_path": transcript_path,
        "hook_event_name": "Stop",
        "cwd": cwd,
    })

    log(f"writing mission.md sid={session_id} fill={fill_pct:.2f} cwd={cwd}")
    try:
        proc = subprocess.run(
            ["bash", REFLECT_SH, "mission"],
            input=payload,
            capture_output=True,
            text=True,
            timeout=90,
        )
        if proc.returncode != 0:
            log(f"mission write failed sid={session_id} rc={proc.returncode} stderr={proc.stderr[:500]!r}")
    except subprocess.TimeoutExpired:
        log(f"mission write timed out sid={session_id}")
    except OSError as e:
        log(f"mission write error sid={session_id} err={e}")

    sys.exit(0)


if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""Aggregates session-reflect JSON artifacts (final.json + PreCompact
checkpoints) from the last N hours into a de-duplicated, frequency-ranked
digest for the recursive-self-improvement daily_review prompt.

Cheap triage layer over raw transcript reads: reflect artifacts are tiny
(1-5 KB) pre-synthesized JSON, versus multi-MB raw .jsonl transcripts that
daily_review's own Step 2 otherwise reads in full for every session
touched in the last 24h. This is the "separate later system that reads
these artifacts" that skills/session-reflect/README.md explicitly
deferred as a non-goal of session-reflect itself.

Usage: aggregate-reflections.py [--hours N] [--root PATH]
Output: rendered digest on stdout. No side effects, no writes.
"""
from __future__ import annotations

import argparse
import json
import os
import re
import time
from collections import defaultdict

REFLECT_ROOT_DEFAULT = os.path.expanduser(os.environ.get("REFLECT_ROOT", "~/.claude/reflections"))
KEYS = ["obstacles_to_autonomy", "patterns", "discoveries", "turning_points", "digressions"]


def normalize(text: str) -> str:
    return re.sub(r"\s+", " ", text.strip().lower())


def find_artifacts(root: str, since_epoch: float) -> list[str]:
    paths = []
    sessions_dir = os.path.join(root, "sessions")
    if not os.path.isdir(sessions_dir):
        return paths
    for dirpath, _dirnames, filenames in os.walk(sessions_dir):
        for fn in filenames:
            if fn == "final.json" or fn.endswith("-precompact.json"):
                full = os.path.join(dirpath, fn)
                try:
                    if os.path.getmtime(full) >= since_epoch:
                        paths.append(full)
                except OSError:
                    continue
    return paths


def load_artifact(path: str) -> dict | None:
    try:
        with open(path) as f:
            return json.load(f)
    except (OSError, json.JSONDecodeError):
        return None


def aggregate(root: str, hours: float):
    since = time.time() - hours * 3600
    digest = {k: defaultdict(lambda: {"count": 0, "sessions": set(), "sources": set(), "example": ""}) for k in KEYS}
    scanned = 0
    for path in find_artifacts(root, since):
        art = load_artifact(path)
        if not art:
            continue
        scanned += 1
        sid = art.get("session_id", "unknown")
        src = art.get("source_transcript", "")
        for k in KEYS:
            for item in art.get(k, []) or []:
                if not isinstance(item, str) or not item.strip():
                    continue
                norm = normalize(item)
                entry = digest[k][norm]
                entry["count"] += 1
                entry["sessions"].add(sid)
                if src:
                    entry["sources"].add(src)
                if not entry["example"]:
                    entry["example"] = item.strip()
    return digest, scanned


def render(digest, scanned: int, hours: float) -> str:
    lines = [f"# Reflection digest — last {hours}h ({scanned} artifacts scanned)", ""]
    any_items = False
    for k in KEYS:
        entries = sorted(digest[k].items(), key=lambda kv: -kv[1]["count"])
        if not entries:
            continue
        any_items = True
        lines.append(f"## {k} ({len(entries)} distinct)")
        for _norm, meta in entries:
            sessions = ", ".join(sorted(meta["sessions"]))
            lines.append(f"- ({meta['count']}x, sessions: {sessions}) {meta['example']}")
        lines.append("")
    if not any_items:
        lines.append("(no reflection artifacts in this window)")
    return "\n".join(lines)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--hours", type=float, default=24)
    ap.add_argument("--root", default=REFLECT_ROOT_DEFAULT)
    args = ap.parse_args()
    digest, scanned = aggregate(args.root, args.hours)
    print(render(digest, scanned, args.hours))


if __name__ == "__main__":
    main()
```
