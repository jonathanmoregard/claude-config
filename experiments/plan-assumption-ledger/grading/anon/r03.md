```python
Invokes the hook script as a subprocess (never imports it — its filename

Run: python3 tests/test_self_compact.py
Exits 0 on success; non-zero + stderr detail on first failure.

import json
import os
import subprocess
import sys
import tempfile

HERE = os.path.dirname(os.path.abspath(__file__))
REPO_ROOT = os.path.dirname(HERE)
SCRIPT = os.path.join(REPO_ROOT, "self-compact.py")

FAILURES: list[str] = []


def check(label: str, cond: bool, detail: str = "") -> None:
    mark = "OK " if cond else "FAIL"
    print(f"  [{mark}] {label}" + (f" :: {detail}" if detail else ""))
    if not cond:
        FAILURES.append(label)


def write_transcript(lines):
    fd, path = tempfile.mkstemp(suffix=".txt")  # not .jsonl — see .gitignore note
    with os.fdopen(fd, "w") as f:
        for obj in lines:
            f.write(json.dumps(obj) + "\n")
    return path


def user_turn(i):
    return {"type": "user", "message": {"role": "user", "content": f"m{i}"}}


def assistant_turn(model, total_tokens):
    return {
        "type": "assistant",
        "message": {
            "model": model,
            "usage": {
                "input_tokens": total_tokens,
                "cache_creation_input_tokens": 0,
                "cache_read_input_tokens": 0,
            },
        },
    }


def run_hook(payload, env_extra=None):
    env = dict(os.environ)
    if env_extra:
        env.update(env_extra)
    proc = subprocess.run(
        [sys.executable, SCRIPT],
        input=json.dumps(payload),
        capture_output=True,
        text=True,
        env=env,
    )
    return proc


def fires(proc) -> bool:
    return proc.returncode == 0 and proc.stdout.strip() != ""


# --- 1. Below threshold on a recognized 1M model: stays quiet ---------------
```

```python
#!/usr/bin/env python3
"""
UserPromptSubmit hook: at high context fill, instructs Claude to
autonomously write/refresh tasks/mission.md (CLAUDE.md's Context checkpoint
schema: done / next / blocked / open questions) via its OWN Write tool call,
and continue working — no user question, no A/B/C menu.

Supersedes context-baggage.py's ask-the-user nudge (2026-08-08..2026-08-17):
self-compact acts instead of asking, because the action itself (one Write
call) is cheap and CLAUDE.md's global default is autonomous execution —
asking wastes a turn re-litigating something already delegated.

The write happens in the LIVE session via its own Write tool, not via a
`reflect.sh mission` re-invocation: that path shells out to
`claude --resume <sid> --fork-session`, which scripts/idle-handoff.sh
documents fails against an actively-loaded session
("No conversation found in fork context") — exactly the situation here.

Context window detection (in priority order), unchanged from
context-baggage.py:
    1. CLAUDE_CONTEXT_MAX env var (explicit override, in tokens)
    2. Model ID prefix — Opus/Sonnet 4.x/5.x run with 1M in this setup
    3. Observed usage — if any assistant turn has exceeded 200k tokens,
       the session is 1M, regardless of model name
    4. Default 200k
"""
import json
import os
import sys

BASE_MAX = 200_000
ONE_M = 1_000_000
# 30% headroom threshold. Empirically tuned 2026-08-08 (context-baggage.py):
# at 95.5% cache-hit ratio across 7 days, holding baggage is cheap and
# resetting is expensive (cache_write ~$5-10 per cold event); this is the
# fill level where a fresh-session handoff starts to pay off.
THRESHOLD = 0.70
MESSAGE_INTERVAL = 5

ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model: str | None, observed_max: int) -> int:
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def main():
    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    transcript_path = data.get("transcript_path", "")
    if not transcript_path or not os.path.exists(transcript_path):
        sys.exit(0)

    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue

            obj_type = obj.get("type")
            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = (
                        usage.get("input_tokens", 0)
                        + usage.get("cache_creation_input_tokens", 0)
                        + usage.get("cache_read_input_tokens", 0)
                    )
                    if total > observed_max:
                        observed_max = total

    if user_msg_count == 0 or user_msg_count % MESSAGE_INTERVAL != 0:
        sys.exit(0)

    max_tokens = detect_max_tokens(last_model, observed_max)

    if last_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )
        fill_pct = total_tokens / max_tokens
    else:
        fill_pct = os.path.getsize(transcript_path) / (max_tokens * 4)

    if fill_pct < THRESHOLD:
        sys.exit(0)

    pct_display = int(fill_pct * 100)
    message = (
        f"[self-compact] Context is {pct_display}% full ({user_msg_count} messages this "
        f"session; window: {max_tokens // 1000}k tokens) — 30% or less headroom remains. "
        "Write tasks/mission.md now via the Write tool (schema: Done / Next / Blocked / Open "
        "questions, per CLAUDE.md's Context checkpoint convention) — resumable without this "
        "conversation's history. This is expected background hygiene: do not ask the user for "
        "permission first, do not present an A/B/C menu. Skip the rewrite only if "
        "tasks/mission.md was already refreshed in the last few turns. Then continue the "
        "user's current request as normal."
    )
    print(json.dumps({"hookSpecificOutput": {"hookEventName": "UserPromptSubmit", "additionalContext": message}}))


if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""
Digest builder for the RSI daily-review pipeline: aggregates session-reflect
artifacts (~/.claude/reflections/sessions/**) from the last N hours into one
Markdown digest, so the review agent reads cheap Haiku-generated summaries
instead of raw multi-MB transcript JSONL by default. See
skills/session-reflect/README.md for the on-disk layout this reads.

Usage:
  python3 reflection_digest.py [--since 24h] [--reflect-root PATH]

Output: Markdown to stdout. Read-only against the reflections tree.
"""
from __future__ import annotations

import argparse
import json
import os
import re
import sys
from datetime import datetime, timedelta, timezone

# Mirrors skills/session-reflect/lib.sh:is_valid_sid — same charset, same
# leading-dot rejection, so a directory name that would have been refused
# there is refused here too (defense in depth against a seeded reflections
# tree with a traversal-shaped session id).
SID_RE = re.compile(r"^[A-Za-z0-9._-]{1,128}$")


def is_valid_sid(sid: str) -> bool:
    return bool(sid) and bool(SID_RE.match(sid)) and not sid.startswith(".")


def parse_since(value: str) -> timedelta:
    m = re.fullmatch(r"(\d+)([hd])", value.strip())
    if not m:
        raise ValueError(f"--since must look like '24h' or '3d', got {value!r}")
    n, unit = int(m.group(1)), m.group(2)
    return timedelta(hours=n) if unit == "h" else timedelta(days=n)


def parse_iso(ts):
    if not ts:
        return None
    try:
        return datetime.strptime(ts, "%Y-%m-%dT%H:%M:%SZ").replace(tzinfo=timezone.utc)
    except ValueError:
        return None


def load_json(path):
    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    except (OSError, json.JSONDecodeError):
        return None


def newest_checkpoint(checkpoint_dir):
    if not os.path.isdir(checkpoint_dir):
        return None
    files = sorted(f for f in os.listdir(checkpoint_dir) if f.endswith(".json"))
    if not files:
        return None
    return load_json(os.path.join(checkpoint_dir, files[-1]))


def session_activity_time(manifest):
    times = []
    for mode in ("sessionend", "precompact", "subagentstop"):
        m = manifest.get("modes", {}).get(mode, {})
        for key in ("completed_at", "started_at"):
            t = parse_iso(m.get(key))
            if t:
                times.append(t)
    return max(times) if times else None


def collect_sessions(reflect_root, cutoff, base=None, parent_sid=None):
    """Yields (sid, parent_sid, session_dir) for every session under base (or
    reflect_root/sessions on the top call) whose manifest.json exists and
    whose most recent mode activity is >= cutoff. Recurses one level into
    subagents/ for parent sessions. A session with NO parseable activity
    timestamp is skipped (conservative: don't surface undated artifacts as
    "recent")."""
    sessions_root = base or os.path.join(reflect_root, "sessions")
    if not os.path.isdir(sessions_root):
        return
    for name in sorted(os.listdir(sessions_root)):
        if name in ("_malformed",) or not is_valid_sid(name):
            continue
        session_dir = os.path.join(sessions_root, name)
        manifest_path = os.path.join(session_dir, "manifest.json")
        manifest = load_json(manifest_path)
        if manifest is not None:
            activity = session_activity_time(manifest)
            if activity is not None and activity >= cutoff:
                yield name, parent_sid, session_dir
        subagents_dir = os.path.join(session_dir, "subagents")
        if os.path.isdir(subagents_dir):
            yield from collect_sessions(reflect_root, cutoff, subagents_dir, name)


def digest_entry(sid, parent_sid, session_dir):
    manifest = load_json(os.path.join(session_dir, "manifest.json")) or {}
    modes = manifest.get("modes", {})
    source_transcript = manifest.get("source_transcript", "unknown")

    final = None
    status = "missing"
    if modes.get("sessionend", {}).get("final_complete"):
        final = load_json(os.path.join(session_dir, "final.json"))
        status = "final" if final else "error"
    elif modes.get("precompact", {}).get("status") == "complete":
        final = newest_checkpoint(os.path.join(session_dir, "checkpoints"))
        status = "checkpoint_only" if final else "error"
    elif modes.get("sessionend", {}).get("status") in ("error", "timeout"):
        status = modes["sessionend"]["status"]
    elif modes.get("precompact", {}).get("status") in ("error", "timeout"):
        status = modes["precompact"]["status"]

    header = f"## Session {sid}"
    if parent_sid:
        header += f" (subagent of {parent_sid})"
    lines = [header, f"- status: {status}", f"- source_transcript: {source_transcript}"]

    if final:
        for key in ("patterns", "discoveries", "turning_points", "digressions", "obstacles_to_autonomy"):
            items = final.get(key) or []
            if items:
                lines.append(f"- {key}:")
                for item in items:
                    lines.append(f"  - {item}")
        usage = final.get("usage") or {}
        if usage:
            lines.append(f"- usage: {json.dumps(usage, sort_keys=True)}")
    else:
        lines.append("- reflection unavailable — read source_transcript directly for this session")

    return "\n".join(lines)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--since", default="24h")
    ap.add_argument("--reflect-root", default=os.path.expanduser("~/.claude/reflections"))
    args = ap.parse_args()

    try:
        window = parse_since(args.since)
    except ValueError as e:
        print(str(e), file=sys.stderr)
        return 2

    cutoff = datetime.now(timezone.utc) - window
    sessions = list(collect_sessions(args.reflect_root, cutoff))

    if not sessions:
        print(f"No reflection activity in the last {args.since}.")
        return 0

    print(f"# Reflection digest — last {args.since} ({len(sessions)} session(s))\n")
    for sid, parent_sid, session_dir in sessions:
        print(digest_entry(sid, parent_sid, session_dir))
        print()
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
