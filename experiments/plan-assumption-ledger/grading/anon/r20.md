**Goal:** Lower the context threshold at which long-running autonomous sessions hand off to a fresh session (writing `tasks/mission.md` automatically, well before the ~70%/native-compact point), and wire the now-more-frequent `session-reflect` artifacts this produces into the `recursive-self-improvement` daily-review pipeline so nightly proposal generation becomes cheap enough to re-enable.

**Architecture:** Two packages sharing one foundation fix.

1. **Foundation fix** — `session-reflect`'s `mission` mode currently can only successfully write `tasks/mission.md` **once per session** (a latch bug in `lib.sh`'s idempotency tracking). 2. **self-compact** — a new **soft** threshold (default 30% of context, `SELF_COMPACT_THRESHOLD`) that, for sessions running an armed `ralph-loop` (i.e. `autonomous-orchestration`/`orchestrate` sessions — never plain interactive human sessions), silently keeps `tasks/mission.md` fresh via a new `Stop` hook (`hooks/self-compact-check.py`), reusing `reflect.sh mission` rather than reimplementing reflection. This piece is safety-gated and ships **disabled by default** (per-worktree opt-in file, global kill switch) because killing/relaunching a live process is a hard-to-reverse, blast-radius action.
3. **rsi-reflect** — the `recursive-self-improvement` daily reviewer currently re-reads full raw `.jsonl` transcripts with Opus (expensive, currently disabled, and documented as producing noisy "duplicate-of-shipped-work" proposals). A new `collect-reflections.sh` script feeds it the already-distilled `session-reflect` `final.json` artifacts instead (cheap, Haiku-generated, structurally aligned with the reviewer's own categories), falling back to raw-transcript reading only for sessions with no reflection artifact. #!/usr/bin/env python3
```python
"""Shared context-window fill-percentage detection.

Used by context-baggage.py (human-facing 70% nudge) and
self-compact-check.py (autonomous 30% mission.md refresh trigger) so the
two hooks can never silently disagree on what "100% full" means.
"""
from __future__ import annotations

import json
import os
import sys
from dataclasses import dataclass

BASE_MAX = 200_000
ONE_M = 1_000_000

# Claude Code model IDs that run at 1M context by default. Prefixes, so
# dated variants (claude-opus-4-1-20250805) match too.
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)

# Explicit 1M marker in the model ID, e.g. `claude-opus-5[1m]`. Checked
# before the prefix list so it keeps working across model generations
# without another edit here.
ONE_M_MARKER = "[1m]"


@dataclass
class TranscriptSignals:
    user_msg_count: int
    last_usage: dict | None
    last_model: str | None
    observed_max: int
    file_size: int


def detect_max_tokens(model: str | None, observed_max: int, override: str | None = None) -> int:
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def read_transcript_signals(transcript_path: str) -> TranscriptSignals | None:
    if not transcript_path or not os.path.exists(transcript_path):
        return None

    file_size = os.path.getsize(transcript_path)
    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue

            obj_type = obj.get("type")
            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = (
                        usage.get("input_tokens", 0)
                        + usage.get("cache_creation_input_tokens", 0)
                        + usage.get("cache_read_input_tokens", 0)
                    )
                    if total > observed_max:
                        observed_max = total

    return TranscriptSignals(user_msg_count, last_usage, last_model, observed_max, file_size)


def compute_fill_pct(signals: TranscriptSignals, max_tokens: int) -> float:
    if signals.last_usage:
        total_tokens = (
            signals.last_usage.get("input_tokens", 0)
            + signals.last_usage.get("cache_creation_input_tokens", 0)
            + signals.last_usage.get("cache_read_input_tokens", 0)
        )
        return total_tokens / max_tokens
    return signals.file_size / (max_tokens * 4)


def ralph_loop_active(cwd: str) -> bool:
    """True if an armed ralph-loop state file with active: true sits in cwd."""
    if not cwd:
        return False
    state_path = os.path.join(cwd, ".claude", "ralph-loop.local.md")
    if not os.path.exists(state_path):
        return False
    try:
        with open(state_path, "r", errors="replace") as f:
            for line in f:
                if line.strip() == "active: true":
                    return True
    except OSError:
        return False
    return False


def _cli() -> int:
    if len(sys.argv) < 2:
        print("usage: context_fill.py <transcript_path> [model_override]", file=sys.stderr)
        return 2
    transcript_path = sys.argv[1]
    model_override = sys.argv[2] if len(sys.argv) > 2 else None
    signals = read_transcript_signals(transcript_path)
    if signals is None:
        print("0.0 " + str(BASE_MAX))
        return 0
    model = model_override or signals.last_model
    max_tokens = detect_max_tokens(model, signals.observed_max, override=os.environ.get("CLAUDE_CONTEXT_MAX"))
    fill_pct = compute_fill_pct(signals, max_tokens)
    print(f"{fill_pct} {max_tokens}")
    return 0


if __name__ == "__main__":
    sys.exit(_cli())
```

```python
import json
import os
import subprocess
import sys
import tempfile
import time
import unittest

def make_ralph_state(cwd, active=True):
    d = os.path.join(cwd, ".claude")
    os.makedirs(d, exist_ok=True)
    with open(os.path.join(d, "ralph-loop.local.md"), "w") as f:

def make_transcript(fill_tokens, max_tokens_hint="claude-sonnet-5"):
    f = tempfile.NamedTemporaryFile(mode="w", suffix=".jsonl", delete=False)
    f.write(json.dumps({
        "type": "assistant",
        "message": {
            "model": max_tokens_hint,
            "usage": {"input_tokens": fill_tokens, "cache_creation_input_tokens": 0, "cache_read_input_tokens": 0},
        },
    }) + "\n")
    f.close()
    return f.name


def make_stub_reflect_sh(marker_path):
    f = tempfile.NamedTemporaryFile(mode="w", suffix=".sh", delete=False)
    f.write(f"""#!/usr/bin/env bash
cat >/dev/null || true
printf '%s %s\\n' "$1" "${{REFLECT_FORCE:-}}" >> "{marker_path}"
""")
    f.close()
    os.chmod(f.name, 0o755)
    return f.name

    env = dict(os.environ)
    env.update(env_extra)
    p = subprocess.run(

        input=json.dumps(payload),
        text=True,
        env=env,
        capture_output=True,
        timeout=15,

    return p
```

```python
#!/usr/bin/env python3
"""
Stop hook: for sessions running an armed ralph-loop (autonomous-orchestration),
keep tasks/mission.md fresh once context crosses SELF_COMPACT_THRESHOLD (soft,
default 30% — well below context-baggage.py's 70% human-facing nudge and below
the point native auto-compact or self-compact-supervisor.sh would act).

Never blocks the Stop event — pure side effect. Runs alongside ralph-loop's
own Stop hook (which does the actual block/continue decision); this script
must never emit a "decision" field.

Recursion guard: reflect.sh's mission worker spawns its own `claude -p
--resume --fork-session` subprocess under REFLECT_SUBPROCESS=1 (see
skills/session-reflect/reflect.sh). That subprocess's own Stop event would
otherwise re-fire this hook. Short-circuit identically to reflect.sh's own
guard.
"""
import json
import os
import subprocess
import sys
import time

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from context_fill import detect_max_tokens, read_transcript_signals, compute_fill_pct, ralph_loop_active  # noqa: E402

DEFAULT_THRESHOLD = 0.30
DEFAULT_MIN_INTERVAL_SECONDS = 300
DEFAULT_REFLECT_SH = os.path.expanduser("~/.claude/skills/session-reflect/reflect.sh")
DEFAULT_LOG_DIR = os.path.expanduser("~/.claude/reflections/self-compact")


def log_event(log_dir: str, record: dict) -> None:
    try:
        os.makedirs(log_dir, exist_ok=True)
        with open(os.path.join(log_dir, "log.jsonl"), "a") as f:
            f.write(json.dumps(record) + "\n")
    except OSError:
        pass


def main() -> int:
    if os.environ.get("REFLECT_SUBPROCESS") == "1":
        return 0

    try:
        raw = sys.stdin.read()
        payload = json.loads(raw)
    except Exception:
        return 0

    cwd = payload.get("cwd", "")
    transcript_path = payload.get("transcript_path", "")
    session_id = payload.get("session_id", "")

    if not ralph_loop_active(cwd):
        return 0

    signals = read_transcript_signals(transcript_path)
    if signals is None:
        return 0

    max_tokens = detect_max_tokens(signals.last_model, signals.observed_max, override=os.environ.get("CLAUDE_CONTEXT_MAX"))
    fill_pct = compute_fill_pct(signals, max_tokens)

    threshold = float(os.environ.get("SELF_COMPACT_THRESHOLD", DEFAULT_THRESHOLD))
    if fill_pct < threshold:
        return 0

    min_interval = int(os.environ.get("SELF_COMPACT_MIN_INTERVAL_SECONDS", DEFAULT_MIN_INTERVAL_SECONDS))
    mission_path = os.path.join(cwd, "tasks", "mission.md")
    if os.path.exists(mission_path):
        age = time.time() - os.path.getmtime(mission_path)
        if age < min_interval:
            return 0

    reflect_sh = os.environ.get("SELF_COMPACT_REFLECT_SH", DEFAULT_REFLECT_SH)
    log_dir = os.environ.get("SELF_COMPACT_LOG_DIR", DEFAULT_LOG_DIR)

    env = dict(os.environ)
    env["REFLECT_FORCE"] = "1"
    try:
        subprocess.run(
            ["bash", reflect_sh, "mission"],
            input=raw,
            text=True,
            cwd=cwd or None,
            env=env,
            timeout=70,
        )
    except (OSError, subprocess.TimeoutExpired) as e:
        log_event(log_dir, {
            "ts": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
            "session_id": session_id,
            "fill_pct": fill_pct,
            "status": "error",
            "error": str(e),
        })
        return 0

    log_event(log_dir, {
        "ts": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "session_id": session_id,
        "fill_pct": fill_pct,
        "max_tokens": max_tokens,
        "status": "triggered",
    })
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```bash
#!/usr/bin/env bash
# self-compact-supervisor.sh — hard-threshold reset + relaunch for armed
# ralph-loop sessions.
#
# SAFETY MODEL (read before enabling):
#   - Per-worktree opt-in ONLY: acts on a session only if
#     <cwd>/tasks/.self-compact-supervisor-enabled exists. Create it once
#     during Phase 0 arming (see skills/orchestrate/SKILL.md) — never
#     created automatically by this script.
#   - Global kill switch: touch SELF_COMPACT_SUPERVISOR_DISABLE_FLAG
#     (default ~/.claude/self-compact-supervisor.disable) to pause
#     everything without touching per-worktree opt-in files.
#   - Refuses to act without a tasks/mission.md fresher than
#     SELF_COMPACT_SUPERVISOR_MISSION_MAX_AGE_SECONDS (default 600s) — a
#     stale or missing mission.md means self-compact-check.py (the soft
#     30% hook) isn't running or isn't keeping up, and acting without a
#     good handoff artifact would strand the mission.
#   - Refuses to act on a loop armed with max_iterations: 0 (unbounded —
#     nothing numeric to trip).
#   - Acts at most once per session_id (idempotency marker under
#     SELF_COMPACT_SUPERVISOR_MARKER_DIR).
#
# MECHANISM: rather than fighting the ralph-loop plugin's Stop hook (which
# unconditionally blocks every stop while active:true), this script trips
# the ALREADY-DOCUMENTED max-iterations release path (see
# docs/superpowers/plans/2026-08-02-sigil-gated-autonomous-orchestration.md
# Task 4 Step 4): set iteration == max_iterations in the state file, then
# the live session's own NEXT Stop event releases cleanly (plugin removes
# the state file, session ends). This script then confirms the process
# actually exited, re-arms a fresh loop with the SAME sigil, and spawns a
# detached headless claude process to pick up autonomously from the
# now-fresh mission.md.
#
# Intended to run on a periodic timer (systemd user timer / cron), same
# pattern as scripts/idle-handoff.sh. Scheduling is NOT bundled here — see
# that script for the precedent (installed externally, e.g. nixos-config).
set -uo pipefail
umask 077

: "${SELF_COMPACT_SUPERVISOR_PROJECTS_DIR:=$HOME/.claude/projects}"
: "${SELF_COMPACT_SUPERVISOR_THRESHOLD:=0.60}"
: "${SELF_COMPACT_SUPERVISOR_MISSION_MAX_AGE_SECONDS:=600}"
: "${SELF_COMPACT_SUPERVISOR_RELEASE_TIMEOUT_SECONDS:=600}"
: "${SELF_COMPACT_SUPERVISOR_POLL_INTERVAL_SECONDS:=5}"
: "${SELF_COMPACT_SUPERVISOR_LOG_DIR:=$HOME/.claude/reflections/self-compact/supervisor-logs}"
: "${SELF_COMPACT_SUPERVISOR_MARKER_DIR:=$HOME/.claude/reflections/self-compact/supervisor}"
: "${SELF_COMPACT_SUPERVISOR_LOCK:=$HOME/.claude/reflections/self-compact/supervisor.lock}"
: "${SELF_COMPACT_SUPERVISOR_DISABLE_FLAG:=$HOME/.claude/self-compact-supervisor.disable}"
: "${SELF_COMPACT_SUPERVISOR_RALPH_SETUP:=}"
CONTEXT_FILL_PY="$HOME/.claude/hooks/context_fill.py"

mkdir -p "$SELF_COMPACT_SUPERVISOR_LOG_DIR" "$SELF_COMPACT_SUPERVISOR_MARKER_DIR" 2>/dev/null || exit 0

log() {
  printf '%s %s\n' "$(date -u +%Y-%m-%dT%H:%M:%SZ)" "$*" >> "$SELF_COMPACT_SUPERVISOR_LOG_DIR/supervisor.log"
}

if [[ -f "$SELF_COMPACT_SUPERVISOR_DISABLE_FLAG" ]]; then
  log "disabled (global flag exists)"
  exit 0
fi

exec 8>>"$SELF_COMPACT_SUPERVISOR_LOCK" 2>/dev/null || { log "cannot open lock"; exit 0; }
if ! flock -n 8; then
  log "another supervisor run in progress, skipping"
  exit 0
fi

is_live_pid_for_sid() {
  local sid="$1"
  pgrep -af 'claude --resume' 2>/dev/null | grep -qF -- "$sid"
}

resolve_ralph_setup() {
  if [[ -n "$SELF_COMPACT_SUPERVISOR_RALPH_SETUP" ]]; then
    printf '%s' "$SELF_COMPACT_SUPERVISOR_RALPH_SETUP"
    return 0
  fi
  ls -d "$HOME"/.claude/plugins/cache/claude-plugins-official/ralph-loop/*/scripts/setup-ralph-loop.sh 2>/dev/null | sort -V | tail -1
}

for project_dir in "$SELF_COMPACT_SUPERVISOR_PROJECTS_DIR"/*/; do
  [[ -d "$project_dir" ]] || continue

  transcript=$(find "$project_dir" -maxdepth 1 -type f -name '*.jsonl' \
    -printf '%T@ %p\n' 2>/dev/null | sort -rn | head -1 | awk '{print $2}')
  [[ -n "$transcript" && -f "$transcript" ]] || continue

  sid=$(basename "$transcript" .jsonl)

  cwd=$(tac "$transcript" 2>/dev/null | jq -r 'select(type=="object") | .cwd // empty' 2>/dev/null | head -1)
  [[ -n "$cwd" && -d "$cwd" ]] || continue

  # --- Gate 1: opt-in marker -------------------------------------------------
  [[ -f "$cwd/tasks/.self-compact-supervisor-enabled" ]] || continue

  # --- Gate 2: armed ralph loop ----------------------------------------------
  state="$cwd/.claude/ralph-loop.local.md"
  [[ -f "$state" ]] || continue
  grep -q '^active: true$' "$state" || continue

  # --- Gate 3: idempotency (act at most once per session) -------------------
  marker="$SELF_COMPACT_SUPERVISOR_MARKER_DIR/${sid}.acted"
  [[ -f "$marker" ]] && continue

  # --- Gate 4: fill % above hard threshold -----------------------------------
  read -r fill_pct max_tokens < <(python3 "$CONTEXT_FILL_PY" "$transcript" 2>/dev/null || echo "0 0")
  awk -v f="$fill_pct" -v t="$SELF_COMPACT_SUPERVISOR_THRESHOLD" 'BEGIN{exit !(f>=t)}' || continue

  # --- Gate 5: fresh mission.md (safety invariant) ---------------------------
  mission="$cwd/tasks/mission.md"
  if [[ ! -f "$mission" ]]; then
    log "sid=$sid REFUSE: no tasks/mission.md — self-compact-check.py hasn't run yet"
    continue
  fi
  mission_age=$(( $(date +%s) - $(stat -c '%Y' "$mission" 2>/dev/null || echo 0) ))
  if (( mission_age > SELF_COMPACT_SUPERVISOR_MISSION_MAX_AGE_SECONDS )); then
    log "sid=$sid REFUSE: mission.md is ${mission_age}s old (max ${SELF_COMPACT_SUPERVISOR_MISSION_MAX_AGE_SECONDS}s)"
    continue
  fi

  # --- Extract state-file fields (data, not code — no eval/interpolation) ---
  max_iter=$(awk '/^max_iterations:/{sub(/^max_iterations: */,""); print; exit}' "$state")
  [[ "$max_iter" =~ ^[0-9]+$ ]] || { log "sid=$sid REFUSE: unparseable max_iterations"; continue; }
  if (( max_iter == 0 )); then
    log "sid=$sid REFUSE: max_iterations=0 (unbounded) — nothing to trip"
    continue
  fi
  sigil=$(awk -F': ' '/^completion_promise:/{ $1=""; sub(/^ /,""); gsub(/^"|"$/,""); print; exit }' "$state")
  # Body is everything after the SECOND '---' line (frontmatter close).
  body=$(awk 'BEGIN{c=0} /^---$/{c++; next} c>=2{print}' "$state")
  [[ -n "$sigil" && -n "$body" ]] || { log "sid=$sid REFUSE: could not extract sigil/body from state file"; continue; }

  log "sid=$sid ACT: fill=${fill_pct} threshold=${SELF_COMPACT_SUPERVISOR_THRESHOLD} cwd=$cwd — tripping max-iterations release"

  # Mark BEFORE waiting so a re-run mid-wait can't double-act.
  touch "$marker"

  # --- Trip the release: set iteration == max_iterations ---------------------
  tmp_state="$(mktemp -p "$(dirname -- "$state")" ".ralph-loop-XXXXXX")"
  sed "s/^iteration: .*/iteration: ${max_iter}/" "$state" > "$tmp_state"
  mv "$tmp_state" "$state"

  # --- Poll for the live session to actually exit -----------------------------
  waited=0
  released=0
  while (( waited < SELF_COMPACT_SUPERVISOR_RELEASE_TIMEOUT_SECONDS )); do
    if [[ ! -f "$state" ]] && ! is_live_pid_for_sid "$sid"; then
      released=1
      break
    fi
    sleep "$SELF_COMPACT_SUPERVISOR_POLL_INTERVAL_SECONDS"
    waited=$(( waited + ${SELF_COMPACT_SUPERVISOR_POLL_INTERVAL_SECONDS%.*} ))
    waited=$(( waited > 0 ? waited : waited + 1 ))  # guard against sub-1s intervals stalling the loop
  done

  if (( released == 0 )); then
    log "sid=$sid STUCK: session did not release within ${SELF_COMPACT_SUPERVISOR_RELEASE_TIMEOUT_SECONDS}s — leaving alone, rm $marker to retry"
    continue
  fi

  # --- Re-arm + relaunch --------------------------------------------------------
  setup=$(resolve_ralph_setup)
  if [[ -z "$setup" || ! -f "$setup" ]]; then
    log "sid=$sid ERROR: ralph-loop setup script not found — cannot re-arm"
    continue
  fi

  (
    cd "$cwd" || exit 1
    bash "$setup" "$body" --completion-promise "$sigil" --max-iterations "$max_iter" >>"$SELF_COMPACT_SUPERVISOR_LOG_DIR/${sid}.rearm.log" 2>&1
  )
  if [[ ! -f "$state" ]]; then
    log "sid=$sid ERROR: re-arm failed, state file missing after setup script"
    continue
  fi

  (
    cd "$cwd" || exit 1
    setsid env _SELF_COMPACT_RELAUNCH=1 nohup claude -p "$body" \
      </dev/null >>"$SELF_COMPACT_SUPERVISOR_LOG_DIR/${sid}.relaunch.log" 2>&1 &
    disown
  )

  log "sid=$sid DONE: re-armed and relaunched a fresh headless session in $cwd"
done

log "run complete"
```

```bash
#!/usr/bin/env bash
# collect-reflections.sh — gathers session-reflect final.json artifacts from
# the last 24h into a compact JSON digest for the RSI daily reviewer.
#
# Stdout: JSON array of {session_id, parent_session_id, source_transcript,
#         patterns, discoveries, turning_points, digressions,
#         obstacles_to_autonomy, generated_at}.
# Stderr: one "MISSING_REFLECTION:<transcript_path>" line per session in the
#         window that has no usable final.json (no reflection attempted, or
#         it errored/timed out) — recursive-self-improvement/config/prompt.md
#         falls back to reading that transcript raw.
#
# Cheap by design: reads already-generated Haiku digests instead of
# re-reading full raw transcripts with Opus (see
# skills/session-reflect/README.md for the artifact schema this consumes).
set -uo pipefail

: "${REFLECT_ROOT:=$HOME/.claude/reflections}"
: "${RSI_REFLECT_WINDOW_HOURS:=24}"
: "${RSI_REFLECT_BUDGET_BYTES:=200000}"

SESSIONS_DIR="$REFLECT_ROOT/sessions"
[[ -d "$SESSIONS_DIR" ]] || { echo "[]"; exit 0; }

now_epoch=$(date -u +%s)
window_seconds=$(( RSI_REFLECT_WINDOW_HOURS * 3600 ))

# One JSON object per line for easy age-sort; missing transcripts noted separately.
tmp_entries="$(mktemp)"
trap 'rm -f "$tmp_entries"' EXIT

for session_dir in "$SESSIONS_DIR"/*/; do
  [[ -d "$session_dir" ]] || continue
  sid=$(basename "$session_dir")
  [[ "$sid" == "_malformed" ]] && continue

  final="$session_dir/final.json"
  manifest="$session_dir/manifest.json"

  if [[ -f "$final" ]]; then
    mtime=$(stat -c '%Y' "$final" 2>/dev/null || echo 0)
    age=$(( now_epoch - mtime ))
    (( age <= window_seconds )) || continue
    jq -c '.' "$final" 2>/dev/null >> "$tmp_entries" || true
  elif [[ -f "$manifest" ]]; then
    mtime=$(stat -c '%Y' "$manifest" 2>/dev/null || echo 0)
    age=$(( now_epoch - mtime ))
    (( age <= window_seconds )) || continue
    tp=$(jq -r '.source_transcript // empty' "$manifest" 2>/dev/null || true)
    [[ -n "$tp" ]] && echo "MISSING_REFLECTION:$tp" >&2
  fi
done

if [[ ! -s "$tmp_entries" ]]; then
  echo "[]"
  exit 0
fi

# Sort newest-first, then greedily keep within the byte budget (mirrors
# reflect.sh's cp_budget_bytes newest-first-keep pattern) — drop oldest
# entries first when the digest would otherwise blow the reviewer's budget.
sort -t$'\t' -k1,1rn <(while IFS= read -r line; do
    gen=$(jq -r '.generated_at // "1970-01-01T00:00:00Z"' <<<"$line")
    epoch=$(date -u -d "$gen" +%s 2>/dev/null || echo 0)
    printf '%s\t%s\n' "$epoch" "$line"
  done < "$tmp_entries") \
| awk -F'\t' '
  { total += length($2) + 1
    if (total <= budget || kept == 0) { print $2; kept++ }
  }
' budget="$RSI_REFLECT_BUDGET_BYTES" \
| jq -c '{
    session_id: (.session_id // null),
    parent_session_id: (.parent_session_id // null),
    source_transcript: (.source_transcript // null),
    generated_at: (.generated_at // null),
    patterns: (.patterns // []),
    discoveries: (.discoveries // []),
    turning_points: (.turning_points // []),
    digressions: (.digressions // []),
    obstacles_to_autonomy: (.obstacles_to_autonomy // [])
  }' \
| jq -s '.'
```
