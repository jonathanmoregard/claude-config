```python
#!/usr/bin/env python3
"""Shared context-fill computation for context-baggage.py and self-compact.py.

Both hooks need identical model-detection and fill-percentage logic — drift
between them would mean one fires at a different threshold than the other
believes is active. This module is the single source of truth.
"""
import json
import os

BASE_MAX = 200_000
ONE_M = 1_000_000

# Fraction of the detected context window considered "full enough to act".
# Raised from 0.25 -> 0.70 on 2026-08-08 after measuring 95.5% cache-hit
# ratio across 7 days / $3,073 spend (probes/analyze_sessions.py). At that
# hit ratio, holding baggage is cheap and resetting is expensive, so the
# threshold fires only once handoff starts to pay off. self-compact.py
# reuses this AS-IS: "≈30% of window remaining" == fill_pct >= FILL_THRESHOLD.
FILL_THRESHOLD = 0.70

# Claude Code model IDs that run at 1M context by default. Prefixes, so
# dated variants (claude-opus-4-1-20250805) match too.
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)

# Explicit 1M marker in the model ID, e.g. `claude-opus-5[1m]`. Checked
# before the prefix list so it keeps working across model generations
# without another edit here — the prefix list going stale is exactly how
# this hook once reported 89% full during a session that was at 18% of a
# 1M window (every reading was fullness-of-200k).
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model, observed_max):
    """Context window detection, in priority order:
    1. CLAUDE_CONTEXT_MAX env var (explicit override, in tokens)
    2. Model ID prefix / [1m] marker — Opus/Sonnet 4.x+ run at 1M here
    3. Observed usage — if any assistant turn exceeded 200k tokens, the
       session is 1M regardless of model name
    4. Default 200k
    """
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def compute_fill_pct(transcript_path):
    """Scan a transcript JSONL and return fill stats, or None if unreadable.

    Returns a dict with fill_pct, total_tokens, max_tokens, model,
    user_msg_count — or None if transcript_path is empty/missing.
    """
    if not transcript_path or not os.path.exists(transcript_path):
        return None

    file_size = os.path.getsize(transcript_path)
    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue

            obj_type = obj.get("type")
            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = (
                        usage.get("input_tokens", 0)
                        + usage.get("cache_creation_input_tokens", 0)
                        + usage.get("cache_read_input_tokens", 0)
                    )
                    if total > observed_max:
                        observed_max = total

    max_tokens = detect_max_tokens(last_model, observed_max)

    if last_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )
        fill_pct = total_tokens / max_tokens
    else:
        total_tokens = 0
        fill_pct = file_size / (max_tokens * 4)

    return {
        "fill_pct": fill_pct,
        "total_tokens": total_tokens,
        "max_tokens": max_tokens,
        "model": last_model,
        "user_msg_count": user_msg_count,
    }
```

```python
#!/usr/bin/env python3
"""
UserPromptSubmit hook: warns when context is over FILL_THRESHOLD with
potentially stale/off-topic content. Fires every MESSAGE_INTERVAL user
messages once threshold is met.

Model detection + fill-percentage math live in context_fill_lib.py, shared
with self-compact.py so the two hooks can never disagree about "how full
is the context".
"""
import json
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from context_fill_lib import compute_fill_pct, FILL_THRESHOLD

MESSAGE_INTERVAL = 5


def main():
    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    result = compute_fill_pct(data.get("transcript_path", ""))
    if result is None:
        sys.exit(0)

    if result["user_msg_count"] == 0 or result["user_msg_count"] % MESSAGE_INTERVAL != 0:
        sys.exit(0)

    if result["fill_pct"] < FILL_THRESHOLD:
        sys.exit(0)

    pct_display = int(result["fill_pct"] * 100)
    message = (
        f"Context is {pct_display}% full ({result['user_msg_count']} messages in this session; "
        f"window: {result['max_tokens'] // 1000}k tokens). "
        "At this fill, a fresh-session handoff via tasks/mission.md is usually cheaper "
        "than an in-place /compact: --fork-session on a warm cache measured ~77% hit "
        "(session-reflect empirical), while /compact invalidates the prefix and forces "
        "cache_write on subsequent turns. "
        "Suggest to the user: (a) write tasks/mission.md (done / next / blocked / open "
        "questions), (b) start a fresh session in the same cwd and load it, OR (c) "
        "continue here and accept the drift. Ask which."
    )
    print(json.dumps({"hookSpecificOutput": {"hookEventName": "UserPromptSubmit", "additionalContext": message}}))


if __name__ == "__main__":
    main()
```
