```python
#!/usr/bin/env python3
"""
UserPromptSubmit hook: at high context fill, auto-refreshes tasks/mission.md
via session-reflect's `mission` mode and instructs the model to self-compact.

Context window detection (in priority order):
    1. CLAUDE_CONTEXT_MAX env var (explicit override, in tokens)
    2. Model ID prefix — Opus/Sonnet 4.x run with 1M in this setup
    3. Observed usage — if any assistant turn has exceeded 200k tokens,
       the session is 1M, regardless of model name
    4. Default 200k

Self-compact (2026-08-17): once fill_pct crosses THRESHOLD, this hook
synchronously calls `skills/session-reflect/reflect.sh mission` to write a
FRESH tasks/mission.md (not just suggest one), then tells the model to
either (SELF_COMPACT_MODE=autonomous) invoke the SlashCommand tool to run
/compact and continue from mission.md without asking, or
(SELF_COMPACT_MODE=advisory_only) recommend /compact or a fresh session to
the user — mission.md is guaranteed current either way. Kill switch: touch
~/.claude/self-compact.disable to fall back to the pre-2026-08-17
advisory-only text with no auto-write.
"""
import json
import os
import subprocess
import sys
import time

BASE_MAX = 200_000
ONE_M = 1_000_000
THRESHOLD = 0.70
MESSAGE_INTERVAL = 5

ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)
ONE_M_MARKER = "[1m]"

REFLECT_ROOT = os.environ.get("REFLECT_ROOT", os.path.expanduser("~/.claude/reflections"))
REFLECT_SH = os.path.expanduser("~/.claude/skills/session-reflect/reflect.sh")
SELF_COMPACT_DISABLE_FLAG = os.path.expanduser("~/.claude/self-compact.disable")
SELF_COMPACT_LOG_DIR = os.path.join(REFLECT_ROOT, "self-compact-logs")
MISSION_WRITE_TIMEOUT_SECONDS = 55
# "autonomous" | "advisory_only" — set by Task 1's empirical finding.
SELF_COMPACT_MODE = os.environ.get("SELF_COMPACT_MODE", "autonomous")


def detect_max_tokens(model: str | None, observed_max: int) -> int:
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def _log_self_compact(line: str) -> None:
    try:
        os.makedirs(SELF_COMPACT_LOG_DIR, exist_ok=True)
        stamp = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
        path = os.path.join(SELF_COMPACT_LOG_DIR, time.strftime("%Y-%m-%d", time.gmtime()) + ".log")
        with open(path, "a") as f:
            f.write(f"{stamp} {line}\n")
    except OSError:
        pass


def write_mission_now(session_id: str, transcript_path: str, cwd: str) -> tuple[bool, str]:
    """Synchronously invoke reflect.sh mission mode. Returns (success, mission_path_or_reason)."""
    if not (session_id and transcript_path and cwd and os.path.isdir(cwd)):
        return False, "missing_session_fields"
    if not os.access(REFLECT_SH, os.X_OK):
        return False, "reflect_sh_not_executable"
    payload = json.dumps(
        {
            "session_id": session_id,
            "transcript_path": transcript_path,
            "hook_event_name": "SelfCompact",
            "cwd": cwd,
        }
    )
    env = dict(os.environ)
    env["REFLECT_FORCE"] = "1"
    try:
        proc = subprocess.run(
            ["bash", REFLECT_SH, "mission"],
            input=payload,
            capture_output=True,
            text=True,
            timeout=MISSION_WRITE_TIMEOUT_SECONDS,
            env=env,
        )
    except subprocess.TimeoutExpired:
        return False, "mission_write_timeout"
    if proc.returncode != 0:
        return False, f"mission_write_exit_{proc.returncode}"
    mission_path = os.path.join(cwd, "tasks", "mission.md")
    if not os.path.isfile(mission_path):
        return False, "mission_file_not_found_after_write"
    return True, mission_path


def build_self_compact_message(pct_display: int, user_msg_count: int, max_tokens: int, mission_path: str) -> str:
    base = (
        f"Context is {pct_display}% full ({user_msg_count} messages in this session; "
        f"window: {max_tokens // 1000}k tokens). tasks/mission.md was just refreshed "
        f"({mission_path}) — this happened automatically just now, it is current."
    )
    if SELF_COMPACT_MODE == "autonomous":
        return base + (
            " Before doing anything else this turn: invoke the SlashCommand tool to run "
            "/compact, then immediately continue the work from tasks/mission.md's Next "
            "section without waiting for further user input. Do not ask the user whether "
            "to compact — mission.md is fresh, nothing will be lost."
        )
    return base + (
        " Recommend to the user now: run /compact (safe — mission.md already captures "
        "done/next/blocked/open questions), or start a fresh session in this cwd and load "
        "mission.md. Ask which they'd prefer."
    )


def build_advisory_message(pct_display: int, user_msg_count: int, max_tokens: int) -> str:
    return (
        f"Context is {pct_display}% full ({user_msg_count} messages in this session; "
        f"window: {max_tokens // 1000}k tokens). "
        "At this fill, a fresh-session handoff via tasks/mission.md is usually cheaper "
        "than an in-place /compact: --fork-session on a warm cache measured ~77% hit "
        "(session-reflect empirical), while /compact invalidates the prefix and forces "
        "cache_write on subsequent turns. "
        "Suggest to the user: (a) write tasks/mission.md (done / next / blocked / open "
        "questions), (b) start a fresh session in the same cwd and load it, OR (c) "
        "continue here and accept the drift. Ask which."
    )


def build_hook_response(data: dict) -> dict | None:
    """Pure-ish core: parsed hook payload in, hookSpecificOutput dict (or None)
    out. Isolated from stdin/stdout for testability."""
    transcript_path = data.get("transcript_path", "")
    if not transcript_path or not os.path.exists(transcript_path):
        return None

    file_size = os.path.getsize(transcript_path)

    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue

            obj_type = obj.get("type")

            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = (
                        usage.get("input_tokens", 0)
                        + usage.get("cache_creation_input_tokens", 0)
                        + usage.get("cache_read_input_tokens", 0)
                    )
                    if total > observed_max:
                        observed_max = total

    if user_msg_count == 0 or user_msg_count % MESSAGE_INTERVAL != 0:
        return None

    max_tokens = detect_max_tokens(last_model, observed_max)

    if last_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )
        fill_pct = total_tokens / max_tokens
    else:
        fill_pct = file_size / (max_tokens * 4)

    if fill_pct < THRESHOLD:
        return None

    pct_display = int(fill_pct * 100)

    if os.path.exists(SELF_COMPACT_DISABLE_FLAG):
        message = build_advisory_message(pct_display, user_msg_count, max_tokens)
        return {"hookSpecificOutput": {"hookEventName": "UserPromptSubmit", "additionalContext": message}}

    session_id = data.get("session_id", "")
    cwd = data.get("cwd", "")
    ok, result = write_mission_now(session_id, transcript_path, cwd)
    if ok:
        _log_self_compact(f"session={session_id} mode={SELF_COMPACT_MODE} mission={result}")
        message = build_self_compact_message(pct_display, user_msg_count, max_tokens, result)
    else:
        _log_self_compact(f"session={session_id} mission_write_failed reason={result}")
        message = build_advisory_message(pct_display, user_msg_count, max_tokens)

    return {"hookSpecificOutput": {"hookEventName": "UserPromptSubmit", "additionalContext": message}}


def main():
    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    response = build_hook_response(data)
    if response is not None:
        print(json.dumps(response))


if __name__ == "__main__":
    main()
```

```bash
#!/usr/bin/env bash
# reflection-digest.sh — compact digest of recent session-reflect artifacts,
# for the recursive-self-improvement review agent to read INSTEAD of raw
# transcripts wherever a reflection artifact exists.
#
# Usage: reflection-digest.sh [LOOKBACK_HOURS]   (default 24)
#
# Mirrors the session-selection rule documented in config/prompt.md Step 2:
# .jsonl files under ~/.claude/projects/ modified within LOOKBACK_HOURS,
# excluding subagents/ subdirectories.
#
# Output (stdout, single JSON object):
#   {
#     "reflected":   [ {session_id, project, source_log, patterns,
#                        discoveries, turning_points, digressions,
#                        obstacles_to_autonomy}, ... ],
#     "unreflected": [ {session_id, project, source_log, reason}, ... ]
#   }
# "unreflected" sessions have no usable sessionend reflection (missing,
# still running, timed out, or errored) — the review agent should fall
# back to reading THOSE raw transcripts only, not all of them.
set -euo pipefail

if [[ "$(uname -s)" != "Linux" ]]; then
  echo "reflection-digest: Linux only (uses GNU date/find)" >&2
  exit 2
fi

: "${REFLECT_ROOT:=$HOME/.claude/reflections}"
: "${CLAUDE_PROJECTS_DIR:=$HOME/.claude/projects}"

LOOKBACK_HOURS="${1:-24}"
[[ "$LOOKBACK_HOURS" =~ ^[0-9]+$ ]] \
  || { echo "reflection-digest: LOOKBACK_HOURS must be a positive integer (got: $LOOKBACK_HOURS)" >&2; exit 2; }

if [[ ! -d "$CLAUDE_PROJECTS_DIR" ]]; then
  jq -n '{reflected: [], unreflected: []}'
  exit 0
fi

cutoff_epoch=$(date -d "-${LOOKBACK_HOURS} hours" +%s)

reflected_entries="[]"
unreflected_entries="[]"

while IFS= read -r transcript; do
  [[ -f "$transcript" ]] || continue
  sid="$(basename -- "$transcript" .jsonl)"
  # Session id must be a safe token before it touches a reflections path.
  [[ "$sid" =~ ^[A-Za-z0-9._-]{1,128}$ ]] || continue
  project_dir="$(dirname -- "$transcript")"
  project="$(basename -- "$project_dir")"

  final_path="$REFLECT_ROOT/sessions/$sid/final.json"
  manifest_path="$REFLECT_ROOT/sessions/$sid/manifest.json"

  if [[ -f "$final_path" ]] \
     && jq -e '.modes.sessionend.final_complete == true' "$manifest_path" >/dev/null 2>&1; then
    entry="$(jq -c \
      --arg sid "$sid" --arg project "$project" --arg log "$transcript" \
      '{session_id:$sid, project:$project, source_log:$log,
        patterns:(.patterns // []), discoveries:(.discoveries // []),
        turning_points:(.turning_points // []), digressions:(.digressions // []),
        obstacles_to_autonomy:(.obstacles_to_autonomy // [])}' \
      "$final_path" 2>/dev/null || true)"
    if [[ -n "$entry" ]]; then
      reflected_entries="$(jq -c --argjson e "$entry" '. + [$e]' <<<"$reflected_entries")"
      continue
    fi
  fi

  reason="no_sessionend_recorded"
  if [[ -f "$manifest_path" ]]; then
    status="$(jq -r '.modes.sessionend.status // ""' "$manifest_path" 2>/dev/null || echo '')"
    case "$status" in
      timeout) reason="sessionend_timeout" ;;
      error) reason="sessionend_error" ;;
      started) reason="sessionend_in_progress" ;;
      complete) reason="sessionend_complete_but_unreadable" ;;
      *) reason="no_sessionend_recorded" ;;
    esac
  fi
  unreflected_entries="$(jq -c \
    --arg sid "$sid" --arg project "$project" --arg log "$transcript" --arg reason "$reason" \
    '. + [{session_id:$sid, project:$project, source_log:$log, reason:$reason}]' \
    <<<"$unreflected_entries")"
done < <(
  find "$CLAUDE_PROJECTS_DIR" -mindepth 2 -maxdepth 2 -type f -name '*.jsonl' \
    -not -path '*/subagents/*' -printf '%T@ %p\n' 2>/dev/null \
  | awk -v cutoff="$cutoff_epoch" '$1+0 >= cutoff' \
  | cut -d' ' -f2-
)

jq -cn --argjson r "$reflected_entries" --argjson u "$unreflected_entries" '{reflected: $r, unreflected: $u}'
```
