```python
#!/usr/bin/env python3
"""
Shared context-fill computation for Claude Code transcripts. Extracted so
context-baggage.py (the interactive 70%-fill nudge hook) and
self-compact-supervisor.sh (the headless 30%-fill reset trigger) share one
token-window-detection implementation instead of drifting apart.
"""
import json
import os
import sys

BASE_MAX = 200_000
ONE_M = 1_000_000

# Claude Code model IDs that run at 1M context by default. Prefixes, so
# dated variants (claude-opus-4-1-20250805) match too.
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model: str | None, observed_max: int) -> int:
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def compute_usage(transcript_path: str) -> dict:
    result = {
        "total_tokens": 0,
        "max_tokens": BASE_MAX,
        "fraction": 0.0,
        "last_model": None,
        "user_msg_count": 0,
        "has_usage": False,
    }
    if not transcript_path or not os.path.exists(transcript_path):
        return result

    last_usage = None
    last_model = None
    observed_max = 0
    user_msg_count = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue

            obj_type = obj.get("type")
            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = (
                        usage.get("input_tokens", 0)
                        + usage.get("cache_creation_input_tokens", 0)
                        + usage.get("cache_read_input_tokens", 0)
                    )
                    if total > observed_max:
                        observed_max = total

    max_tokens = detect_max_tokens(last_model, observed_max)
    total_tokens = 0
    has_usage = last_usage is not None
    if has_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )

    result.update({
        "total_tokens": total_tokens,
        "max_tokens": max_tokens,
        "fraction": (total_tokens / max_tokens) if (has_usage and max_tokens) else 0.0,
        "last_model": last_model,
        "user_msg_count": user_msg_count,
        "has_usage": has_usage,
    })
    return result


def main():
    if len(sys.argv) != 2:
        print("usage: context_usage.py <transcript_path>", file=sys.stderr)
        sys.exit(2)
    print(json.dumps(compute_usage(sys.argv[1])))


if __name__ == "__main__":
    main()
```

import sys

# Raised from 0.25 -> 0.70 on 2026-08-08 after measuring 95.5% cache-hit
# ratio across 7 days / $3,073 spend (probes/analyze_sessions.py). At that
# hit ratio, holding baggage is cheap and resetting is expensive
# (cache_write ~$5-10 per cold event). Nudge should fire when handoff to
# a fresh session starts to pay off, not on any drift.
THRESHOLD = 0.70
MESSAGE_INTERVAL = 5


def main():
    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    transcript_path = data.get("transcript_path", "")
    if not transcript_path or not os.path.exists(transcript_path):

    if user_msg_count == 0 or user_msg_count % MESSAGE_INTERVAL != 0:
        sys.exit(0)

    max_tokens = usage["max_tokens"]
    if usage["has_usage"]:
        fill_pct = usage["fraction"]
    else:
        fill_pct = file_size / (max_tokens * 4)

    if fill_pct < THRESHOLD:
        sys.exit(0)

    pct_display = int(fill_pct * 100)
    message = (
        f"Context is {pct_display}% full ({user_msg_count} messages in this session; "
        f"window: {max_tokens // 1000}k tokens). "

#!/usr/bin/env bash
# self-compact-supervisor.sh — headless reset loop for autonomous orchestrate
# runs. Alternative to arming the ralph-loop plugin (which keeps ONE process
# alive and growing via Stop-hook block); this script instead launches a
# genuinely FRESH `claude -p` process each iteration (no --resume), refreshing
# tasks/mission.md via reflect.sh's mission mode whenever the just-completed
# iteration's context fill crosses SELF_COMPACT_FILL_THRESHOLD (default 30%
# of a 1M window). Mutually exclusive with an armed ralph-loop.local.md in
# the same worktree — pick one engine per run.
#
# Usage: self-compact-supervisor.sh <worktree-dir>
#
# Required env:
#   SELF_COMPACT_SIGIL   Exact completion-promise text (single line, no
#                         " $ ` \). The loop stops when a claude reply
#                         contains the literal <promise>SIGIL</promise>.
#
# Optional env:
#   SELF_COMPACT_FILL_THRESHOLD         default 0.30
#   SELF_COMPACT_MAX_ITERATIONS         default 25
#   SELF_COMPACT_MAX_WALLCLOCK_SECONDS  default 21600 (6h)
#   SELF_COMPACT_MODEL                  default opus
#   SELF_COMPACT_LOG_DIR                default <worktree>/tasks/self-compact-logs
#   SELF_COMPACT_PROJECTS_DIR           default $HOME/.claude/projects (test override)
#   SELF_COMPACT_REFLECT_SH             default $HOME/.claude/skills/session-reflect/reflect.sh (test override)
set -euo pipefail

WORKTREE="${1:?usage: self-compact-supervisor.sh <worktree-dir>}"
[[ -d "$WORKTREE" ]] || { echo "REFUSE: $WORKTREE is not a directory"; exit 1; }
WORKTREE="$(cd "$WORKTREE" && pwd)"

: "${SELF_COMPACT_FILL_THRESHOLD:=0.30}"
: "${SELF_COMPACT_MAX_ITERATIONS:=25}"
: "${SELF_COMPACT_MAX_WALLCLOCK_SECONDS:=21600}"
: "${SELF_COMPACT_MODEL:=opus}"
: "${SELF_COMPACT_LOG_DIR:=$WORKTREE/tasks/self-compact-logs}"
: "${SELF_COMPACT_PROJECTS_DIR:=$HOME/.claude/projects}"
: "${SELF_COMPACT_REFLECT_SH:=$HOME/.claude/skills/session-reflect/reflect.sh}"
: "${SELF_COMPACT_SIGIL:?REFUSE: SELF_COMPACT_SIGIL must be set to the exact completion-promise text}"

case "$SELF_COMPACT_SIGIL" in *'"'*|*'$'*|*'`'*|*'\'*)
  echo 'REFUSE: SELF_COMPACT_SIGIL must not contain " $ ` or \ (would be unmatchable against the literal reply text)'
  exit 1
  ;;
esac

case "$WORKTREE" in
  "$HOME"|"$HOME/.claude"|"$HOME/Repos")
    echo "REFUSE: run only from a dedicated worktree root, not $WORKTREE"
    exit 1
    ;;
esac
[[ -e "$WORKTREE/.git" ]] || { echo "REFUSE: $WORKTREE is not a repo/worktree root"; exit 1; }
[[ -f "$WORKTREE/tasks/mission.md" ]] || { echo "REFUSE: $WORKTREE/tasks/mission.md missing — run orchestrate Phase 0 first"; exit 1; }

[[ -f "$CONTEXT_USAGE_PY" ]] || { echo "REFUSE: context_usage.py missing at $CONTEXT_USAGE_PY"; exit 1; }

mkdir -p "$SELF_COMPACT_LOG_DIR"
log() { printf '%s %s\n' "$(date -u +%Y-%m-%dT%H:%M:%SZ)" "$*" | tee -a "$SELF_COMPACT_LOG_DIR/supervisor.log"; }

start_epoch=$(date +%s)
iteration=0

log "self-compact supervisor starting worktree=$WORKTREE threshold=$SELF_COMPACT_FILL_THRESHOLD max-iter=$SELF_COMPACT_MAX_ITERATIONS"

while :; do
  iteration=$((iteration + 1))
  if (( iteration > SELF_COMPACT_MAX_ITERATIONS )); then
    log "STOP: max iterations ($SELF_COMPACT_MAX_ITERATIONS) reached"
    exit 0
  fi
  now_epoch=$(date +%s)
  if (( now_epoch - start_epoch > SELF_COMPACT_MAX_WALLCLOCK_SECONDS )); then
    log "STOP: wall-clock cap (${SELF_COMPACT_MAX_WALLCLOCK_SECONDS}s) reached"
    exit 0
  fi

  prompt="Read tasks/mission.md, tasks/session-constraints.md, and tasks/progress.md, then continue from where progress.md left off (empty progress.md = start at the first unchecked acceptance criterion). Orchestrate via subagents; do not implement inline. When, and ONLY when, every acceptance criterion is met and the verifier is freshly green, end your final message with EXACTLY: <promise>${SELF_COMPACT_SIGIL}</promise> and nothing else after it. Never emit that tag unless it is literally true."

  iter_out="$SELF_COMPACT_LOG_DIR/iter-${iteration}.json"
  iter_err="$SELF_COMPACT_LOG_DIR/iter-${iteration}.stderr.log"

  log "iteration $iteration: launching fresh claude -p (no --resume)"
  set +e
  claude_json=$(cd "$WORKTREE" && claude -p "$prompt" --model "$SELF_COMPACT_MODEL" --output-format json 2>"$iter_err")
  rc=$?
  set -e
  printf '%s' "$claude_json" > "$iter_out"

  if (( rc != 0 )); then
    log "STOP: iteration $iteration claude exited $rc, see $iter_err — no auto-retry"
    exit "$rc"
  fi

  session_id=$(jq -r '.session_id // empty' <<<"$claude_json" 2>/dev/null || true)
  result_text=$(jq -r '.result // empty' <<<"$claude_json" 2>/dev/null || true)

  if [[ -z "$session_id" ]]; then
    log "STOP: iteration $iteration produced no session_id in claude output"
    exit 1
  fi

  if grep -qF "<promise>${SELF_COMPACT_SIGIL}</promise>" <<<"$result_text"; then
    log "DONE: iteration $iteration sigil matched — mission complete"
    exit 0
  fi

  transcript_path=$(find "$SELF_COMPACT_PROJECTS_DIR" -maxdepth 2 -type f -name "${session_id}.jsonl" 2>/dev/null | head -1)
  if [[ -z "$transcript_path" ]]; then
    log "WARNING: iteration $iteration transcript not found for session $session_id — refreshing mission.md defensively"
    fraction="1.0"
  else
    usage_json=$(python3 "$CONTEXT_USAGE_PY" "$transcript_path" 2>/dev/null || echo '{"fraction":1.0}')
    fraction=$(jq -r '.fraction // 1.0' <<<"$usage_json")
  fi

  log "iteration $iteration: session=$session_id fill=$fraction threshold=$SELF_COMPACT_FILL_THRESHOLD"

  crossed=$(awk -v f="$fraction" -v t="$SELF_COMPACT_FILL_THRESHOLD" 'BEGIN{print (f>=t)?"1":"0"}')
  if [[ "$crossed" == "1" ]]; then
    log "iteration $iteration: fill >= threshold, refreshing tasks/mission.md via reflect.sh mission (session=$session_id)"
    payload=$(jq -cn --arg sid "$session_id" --arg tp "$transcript_path" --arg cwd "$WORKTREE" \
      '{session_id:$sid, transcript_path:$tp, hook_event_name:"SelfCompact", cwd:$cwd}')
    if printf '%s' "$payload" | timeout 90 bash "$SELF_COMPACT_REFLECT_SH" mission >>"$SELF_COMPACT_LOG_DIR/mission-refresh.log" 2>&1; then
      log "iteration $iteration: mission.md refreshed"
    else
      refresh_rc=$?
      log "WARNING: iteration $iteration mission.md refresh failed rc=$refresh_rc — next iteration resets with the STALE mission.md, see mission-refresh.log"
    fi
  fi
done
