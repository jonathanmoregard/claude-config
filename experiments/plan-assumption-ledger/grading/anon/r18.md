#!/usr/bin/env bash
# reflection-coverage.sh — reports how many of the last N hours' sessions
# have a session-reflect final.json vs. fell back to raw-log-only.
#
# Env: PROJECTS_DIR (default: $HOME/.claude/projects)
#      REFLECT_ROOT  (default: $HOME/.claude/reflections)
#      LOOKBACK_HOURS (default: 24)
#
# Output: one summary line `reflected=N unreflected=M total=T coverage=P%`,
# then one `unreflected: <sid>` line per session missing a final.json.
# Exit 0 on a clean run; this is a diagnostic, not a gate.
set -euo pipefail

if [[ "$(uname -s)" != "Linux" ]]; then
  echo "reflection-coverage: Linux only (uses GNU find/date)" >&2
  exit 2
fi

: "${PROJECTS_DIR:=$HOME/.claude/projects}"
: "${REFLECT_ROOT:=$HOME/.claude/reflections}"
: "${LOOKBACK_HOURS:=24}"

[[ -d "$PROJECTS_DIR" ]] || { echo "reflected=0 unreflected=0 total=0 coverage=0%"; exit 0; }

now_epoch="$(date -u +%s)"
cutoff_epoch=$(( now_epoch - LOOKBACK_HOURS * 3600 ))

reflected=0
unreflected=0
unreflected_sids=()

while IFS= read -r -d '' transcript; do
  case "$transcript" in
    */subagents/*) continue ;;
  esac
  mtime="$(stat -c '%Y' "$transcript" 2>/dev/null || echo 0)"
  (( mtime >= cutoff_epoch )) || continue

  sid="$(basename -- "$transcript" .jsonl)"
  final="$REFLECT_ROOT/sessions/$sid/final.json"
  if [[ -f "$final" ]]; then
    reflected=$(( reflected + 1 ))
  else
    unreflected=$(( unreflected + 1 ))
    unreflected_sids+=( "$sid" )
  fi
done < <(find "$PROJECTS_DIR" -maxdepth 2 -type f -name '*.jsonl' -print0 2>/dev/null)

total=$(( reflected + unreflected ))
coverage=0
if (( total > 0 )); then
  coverage=$(( reflected * 100 / total ))
fi

echo "reflected=$reflected unreflected=$unreflected total=$total coverage=${coverage}%"
for sid in "${unreflected_sids[@]:-}"; do
  [[ -n "$sid" ]] && echo "unreflected: $sid"
done
```

#!/usr/bin/env python3
"""Shared context-window fill-percentage detection.

Importable by context-baggage.py (UserPromptSubmit hook) and invoked as a
CLI by skills/self-compact/driver.sh (`--usage-json`). Two independent
callers computing "how full is the context window" must not carry two
copies of the model-prefix table — that's exactly the kind of drift that
made context-baggage.py's ONE_M_MODEL_PREFIXES list a single point of
staleness before this split.
"""
import argparse
import json
import os
import sys

BASE_MAX = 200_000
ONE_M = 1_000_000

# Claude Code model IDs that run at 1M context by default. Prefixes, so
# dated variants (claude-opus-4-1-20250805) match too.
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)

# Explicit 1M marker in the model ID, e.g. `claude-opus-5[1m]`. Checked
# before the prefix list so it keeps working across model generations
# without another edit here.
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model: str | None, observed_max: int, override: str | None = None) -> int:
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def usage_total_tokens(usage: dict) -> int:
    return (
        usage.get("input_tokens", 0)
        + usage.get("cache_creation_input_tokens", 0)
        + usage.get("cache_read_input_tokens", 0)
    )


def fill_pct_from_usage(usage: dict, model: str | None, override: str | None = None) -> float:
    total = usage_total_tokens(usage)
    max_tokens = detect_max_tokens(model, total, override)
    return total / max_tokens


def _main():
    parser = argparse.ArgumentParser(description="Compute context-window fill percentage.")
    parser.add_argument("--usage-json", required=True, help='JSON object, e.g. {"input_tokens":..,"cache_creation_input_tokens":..,"cache_read_input_tokens":..}')
    parser.add_argument("--model", default=None, help="Model id/alias used for the call")
    args = parser.parse_args()

    override = os.environ.get("CLAUDE_CONTEXT_MAX")

    try:
        usage = json.loads(args.usage_json)
    except json.JSONDecodeError:
        print("0.0")
        return

    pct = fill_pct_from_usage(usage, args.model, override)
    print(f"{pct:.4f}")


if __name__ == "__main__":
    sys.exit(_main())

```bash
#!/usr/bin/env bash
# driver.sh — self-compact headless autonomy loop.
#
# Usage: driver.sh "<mission prompt>" --completion-promise "<sigil>" \
#          [--max-iterations N] [--threshold PCT] [--model NAME]
#
# Runs `claude -p` repeatedly, headless. After each iteration, checks
# context fill% (via context_fill.py) against --threshold (default 0.30).
# Under threshold: next iteration resumes the same session (--resume, warm
# cache). At or over threshold: refreshes tasks/mission.md via
# session-reflect's `reflect.sh mission`, then the NEXT iteration starts a
# FRESH session (no --resume), seeded only from tasks/mission.md — that's
# the "reset". Exits 0 only when the model's final text contains an EXACT
# <promise>SIGIL</promise> match.
#
# Verified against real `claude -p --output-format json` output 2026-08-17:
# envelope carries session_id/result/usage at top level (see Task B2 Step 0).
set -euo pipefail
umask 077

if [[ "$(uname -s)" != "Linux" ]]; then
  echo "self-compact: Linux only (uses GNU date/find)" >&2
  exit 2
fi

SCRIPT_DIR="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)"
REPO_ROOT="$(cd -- "$SCRIPT_DIR/../.." && pwd)"
REFLECT_SH="$REPO_ROOT/skills/session-reflect/reflect.sh"
CONTEXT_FILL="$REPO_ROOT/context_fill.py"

MISSION_PROMPT="${1:-}"
if [[ -z "$MISSION_PROMPT" ]]; then
  echo 'usage: driver.sh "<mission prompt>" --completion-promise "<sigil>" [--max-iterations N] [--threshold PCT] [--model NAME]' >&2
  exit 2
fi
shift

SIGIL=""
MAX_ITERATIONS=25
THRESHOLD="0.30"
MODEL=""

while [[ $# -gt 0 ]]; do
  case "$1" in
    --completion-promise) SIGIL="${2:-}"; shift 2 ;;
    --max-iterations) MAX_ITERATIONS="${2:-}"; shift 2 ;;
    --threshold) THRESHOLD="${2:-}"; shift 2 ;;
    --model) MODEL="${2:-}"; shift 2 ;;
    *) echo "self-compact: unknown arg '$1'" >&2; exit 2 ;;
  esac
done

if [[ -z "$SIGIL" ]]; then
  echo "self-compact: --completion-promise is required" >&2
  exit 2
fi
case "$SIGIL" in
  *'"'*|*'$'*|*'`'*|*'\'*)
    echo 'self-compact: sigil must not contain " $ ` or \' >&2
    exit 2
    ;;
esac
[[ "$MAX_ITERATIONS" =~ ^[0-9]+$ ]] || { echo "self-compact: --max-iterations must be a positive integer" >&2; exit 2; }

case "$PWD" in
  "$HOME"|"$HOME/.claude"|"$HOME/Repos")
    echo "self-compact: refuse to run from $PWD — run from a dedicated worktree root" >&2
    exit 1
    ;;
esac
[[ -e .git ]] || { echo "self-compact: refuse — not a repo/worktree root (.git missing)" >&2; exit 1; }

mkdir -p tasks
: > tasks/progress.md
[[ -f tasks/session-constraints.md ]] || : > tasks/session-constraints.md

CONTINUE_PROMPT='Read tasks/mission.md, tasks/session-constraints.md, and tasks/progress.md, then continue from where progress.md left off (empty progress.md = start at the first unchecked acceptance criterion). Orchestrate via subagents; do not implement inline; paste session-constraints.md verbatim into every subagent briefing.'

sid=""
iteration=1
while (( iteration <= MAX_ITERATIONS )); do
  if [[ $iteration -eq 1 ]]; then
    prompt="$MISSION_PROMPT"
  else
    prompt="$CONTINUE_PROMPT"
  fi

  resume_args=()
  [[ -n "$sid" ]] && resume_args=( --resume "$sid" )
  model_args=()
  [[ -n "$MODEL" ]] && model_args=( --model "$MODEL" )

  envelope="$(claude -p "${resume_args[@]}" "${model_args[@]}" --output-format json <<<"$prompt")"

  result="$(jq -r '.result // empty' <<<"$envelope" 2>/dev/null || true)"
  new_sid="$(jq -r '.session_id // empty' <<<"$envelope" 2>/dev/null || true)"
  usage_json="$(jq -c '.usage // {}' <<<"$envelope" 2>/dev/null || echo '{}')"

  if [[ -z "$result" || -z "$new_sid" ]]; then
    echo "self-compact: iteration $iteration returned no result/session_id, aborting" >&2
    printf -- '- iteration %d: ERROR empty result/session_id from claude -p\n' "$iteration" >> tasks/progress.md
    exit 3
  fi

  found_promise="$(printf '%s' "$result" | grep -oP '(?<=<promise>).*?(?=</promise>)' | head -1 || true)"
  if [[ "$found_promise" == "$SIGIL" ]]; then
    printf -- '- iteration %d: SIGIL matched, done\n' "$iteration" >> tasks/progress.md
    echo "self-compact: sigil matched at iteration $iteration"
    exit 0
  fi

  # Threshold/reset logic lands in Task B3; this slice always resumes.
  printf -- '- iteration %d: no sigil, resuming %s\n' "$iteration" "$new_sid" >> tasks/progress.md
  sid="$new_sid"

  iteration=$(( iteration + 1 ))
done

echo "self-compact: max-iterations ($MAX_ITERATIONS) reached without sigil match" >&2
exit 4
```

# Self-Compact

A headless supervisor for autonomous `claude -p` work. Where `orchestrate`'s
Phase 0 arms the official ralph-loop plugin *inside one long-lived
interactive session* (a human's terminal keeps running, Stop hook blocks
premature exits, native auto-compact eventually fires and
`compact-reinject.sh` restores doctrine), self-compact runs *outside* the
model entirely: a bash loop that invokes `claude -p` per iteration and
decides, from the OUTSIDE, whether the next iteration resumes the same
session or starts fresh.

## Why this exists (and why it's not a hook)

Claude Code has no supported hook or config surface for a script — or the
model itself — to force mid-session compaction. This was verified and
explicitly ruled out during the ralph-loop integration work
(`docs/superpowers/plans/2026-08-02-sigil-gated-autonomous-orchestration.md`,
"Deliberately out of scope": *"Custom OpenHands-style compaction prompt:
no supported Claude Code config surface for the compaction prompt"*). The
only two real "reset" mechanisms Claude Code offers are (1) its own
internal auto-compact, on its own internal threshold, and (2) starting an
actual new process. self-compact uses (2): it's a process supervisor, not
a trick played on a live session.

## When to use this vs. `orchestrate`'s ralph-loop

| | `orchestrate` (interactive ralph-loop) | `self-compact` (headless driver) |
|---|---|---|
| Human watching a terminal | Yes | No |
| Context reset trigger | Native Claude Code auto-compact (high threshold, not user-controllable) | Configurable `--threshold` (default 0.30) |
| Mechanism | Stop hook blocks premature stops in ONE session | External loop launches a NEW `claude -p` process per iteration |
| Doctrine survival across resets | `compact-reinject.sh` (SessionStart matcher=compact) | `tasks/mission.md`, rewritten fresh before every reset |

Use self-compact when you're kicking off work and walking away (overnight
runs, cron-dispatched background missions). Use `orchestrate`'s ralph-loop
arm when you're driving from an interactive terminal — the interactive
flow's Stop-hook gate and native compact are already validated and don't
need this extra machinery while a human is present to `/compact` by hand
or answer `context-baggage.py`'s 70% nudge.

## Usage

```bash
cd /home/jonathan/worktrees/<repo>-<branch>   # dedicated worktree, never $HOME/.claude/Repos
bash ~/.claude/skills/self-compact/driver.sh \
  "Autonomous mission: <task description>" \
  --completion-promise 'All acceptance criteria in tasks/mission.md are implemented, the verifier is green, and a fresh-context reviewer confirmed the diff.' \
  --max-iterations 25 \
  --threshold 0.30
```

Same sigil discipline as `orchestrate`: single line, no `" $ \` \` characters,
TRUE only at genuine completion, never emitted early. Write
`tasks/mission.md` with acceptance criteria before the first invocation —
iteration 1 gets `MISSION_PROMPT` verbatim as its opening turn, so put the
goal there; iterations 2+ always get the fixed "read
tasks/mission.md/progress.md/session-constraints.md and continue" prompt,
same wording `orchestrate`'s ralph arm uses.

## What happens at threshold

At or over `--threshold` fill (measured via `context_fill.py` against the
iteration's own `usage` numbers), the driver synchronously runs
`skills/session-reflect/reflect.sh mission` against the just-finished
session — the SAME mechanism `idle-handoff.sh` and the PreCompact
2nd-compact defense already use, writing `<cwd>/tasks/mission.md` before
anything is lost. The NEXT `claude -p` call then omits `--resume` entirely:
a brand-new session, opened with the fixed continuation prompt, reading
mission.md fresh. `tasks/progress.md` gets one line per iteration
regardless of outcome (resumed / reset / sigil-matched / error) — it's the
audit trail.

## Exit codes

| Code | Meaning |
|---|---|
| 0 | Sigil matched — mission complete |
| 1 | Preflight refused (not a worktree, missing `.git`) |
| 2 | Usage error (missing `--completion-promise`, bad sigil chars, bad `--max-iterations`) |
| 3 | `claude -p` returned no `result`/`session_id` — aborted, does not retry silently |
| 4 | Max-iterations reached without a sigil match — logged to `pending_for_human.md` |

## Pairing with rsi-reflect

Every reset ends a session (`SessionEnd` fires) and starts a new one.
Long autonomous work run through self-compact therefore produces several
short, clean session-reflect `final.json` artifacts instead of one long
drifting session's single reflection — that's more, cheaper, cleaner
signal for `recursive-self-improvement/config/prompt.md`'s daily review
(see `recursive-self-improvement/README.md`). Neither package depends on
the other to function.

## Non-goals

- Not a replacement for `orchestrate`'s interactive ralph-loop — see the
  table above.
- Does not attempt to force Claude Code's own compaction mid-turn; that
  surface doesn't exist (see "Why this exists" above).
- Does not manage git branches, PRs, or worktree creation —
  `superpowers:using-git-worktrees` and `superpowers:finishing-a-development-branch`
  still own those.
