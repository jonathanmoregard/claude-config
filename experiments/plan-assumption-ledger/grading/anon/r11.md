## Assumptions

- **Claude Code cannot programmatically trigger `/compact` or `/clear`.** No hook event or tool-call surface exists for the running assistant to invoke either. Verified by: no `SlashCommand` tool references anywhere in this repo, and every existing "reset" mechanism (`context-baggage.py`, the RSI proposals, `compact-reinject.sh`) either *asks the user* to run `/compact`/`/clear` or *reacts after* Claude Code's own automatic compaction fires. Self-compact is therefore designed to make whichever reset eventually happens (auto-compact or a user-typed `/clear`) lossless, not to force an early reset itself. If this assumption is wrong (a newer Claude Code build exposes such a primitive), the design in Task 5 can be extended without touching Tasks 1–4.
- **`idle-handoff.sh` is currently unscheduled** — confirmed via `grep` across `logs/crontab-backup-*` and `dev-container/systemd/*` (no matches) and no systemd timer references it. Task 2 fixes this gap.
- **`mission` mode's idempotency is currently one-shot-per-session** (Task 1's core finding): `lib.sh:write_manifest` latches `modes.mission.final_complete=true` after the first successful write for a given `session_id`, and `reflect.sh`'s idempotency gate (`reflect.sh:298-305`) is not exempted for `mission` the way it is for `precompact`. Every existing caller of `mission` mode (`reflect.sh`'s 2nd-compact defense, `idle-handoff.sh`) therefore silently stops refreshing `tasks/mission.md` after the first success for that session, unless `REFLECT_FORCE=1` is set (nothing sets it). This is a prerequisite bug fix for self-compact, which fundamentally needs repeated refreshes as a session's context grows.

```python
#!/usr/bin/env python3
"""Shared context-fill detection for UserPromptSubmit hooks.

Used by context-baggage.py (asks the user at 70% fill) and self-compact.py
(autonomously refreshes tasks/mission.md at 30% fill for ralph-armed
sessions). Both need the identical fill_pct computation so their thresholds
are measured against the same number.
"""
import json
import os

BASE_MAX = 200_000
ONE_M = 1_000_000

# Claude Code model IDs that run at 1M context by default. Prefixes, so
# dated variants (claude-opus-4-1-20250805) match too.
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)

# Explicit 1M marker in the model ID, e.g. `claude-opus-5[1m]`. Checked
# before the prefix list so it keeps working across model generations
# without another edit here.
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model, observed_max):
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def compute_fill(transcript_path):
    """Scan a transcript JSONL and return fill metadata, or None if the
    transcript is missing/unreadable.

    Returns a dict with keys: fill_pct, max_tokens, user_msg_count,
    last_usage, file_size.
    """
    if not transcript_path or not os.path.exists(transcript_path):
        return None

    file_size = os.path.getsize(transcript_path)
    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue

            obj_type = obj.get("type")
            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = (
                        usage.get("input_tokens", 0)
                        + usage.get("cache_creation_input_tokens", 0)
                        + usage.get("cache_read_input_tokens", 0)
                    )
                    if total > observed_max:
                        observed_max = total

    max_tokens = detect_max_tokens(last_model, observed_max)

    if last_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )
        fill_pct = total_tokens / max_tokens
    else:
        fill_pct = file_size / (max_tokens * 4)

    return {
        "fill_pct": fill_pct,
        "max_tokens": max_tokens,
        "user_msg_count": user_msg_count,
        "last_usage": last_usage,
        "file_size": file_size,
    }
```

```python
#!/usr/bin/env python3
"""UserPromptSubmit hook: for ralph-loop-armed sessions ONLY, autonomously
refreshes tasks/mission.md once context fill crosses SELF_COMPACT_THRESHOLD
(default 30% of the detected context window) — no asking, unlike
context-baggage.py's 70% ask-nudge for ordinary interactive sessions.

Scope: only fires when .claude/ralph-loop.local.md exists in the session's
cwd (an explicitly armed autonomous mission — see
skills/orchestrate/SKILL.md Phase 0). This keeps normal interactive
sessions unaffected; self-compact never changes behavior for a session the
user hasn't already opted into autonomous operation for.

Fires reflect.sh's `mission` mode fire-and-forget (does not block this hook
/ the user's turn). Requires the session-reflect mission-mode idempotency
fix (mission is a repeatable checkpoint, not a one-shot artifact) — see
skills/session-reflect/lib.sh.
"""
import json
import os
import subprocess
import sys
import time

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from context_fill import compute_fill

THRESHOLD = float(os.environ.get("SELF_COMPACT_THRESHOLD", "0.30"))
MISSION_MAX_AGE_SECONDS = int(os.environ.get("SELF_COMPACT_MISSION_MAX_AGE_SECONDS", str(5 * 60)))
REFLECT_SH = os.environ.get(
    "SELF_COMPACT_REFLECT_SH",
    os.path.expanduser("~/.claude/skills/session-reflect/reflect.sh"),
)
LOG_DIR = os.environ.get(
    "SELF_COMPACT_LOG_DIR",
    os.path.expanduser("~/.claude/reflections/self-compact-logs"),
)


def is_armed(cwd):
    return os.path.isfile(os.path.join(cwd, ".claude", "ralph-loop.local.md"))


def mission_is_fresh(cwd):
    mission_path = os.path.join(cwd, "tasks", "mission.md")
    if not os.path.isfile(mission_path):
        return False
    age = time.time() - os.path.getmtime(mission_path)
    return age < MISSION_MAX_AGE_SECONDS


def fire_mission_refresh(session_id, transcript_path, cwd):
    payload = json.dumps({
        "session_id": session_id,
        "transcript_path": transcript_path,
        "hook_event_name": "SelfCompact",
        "cwd": cwd,
    })
    os.makedirs(LOG_DIR, exist_ok=True)
    log_path = os.path.join(LOG_DIR, f"{session_id}.{int(time.time())}.log")
    with open(log_path, "ab") as logf:
        proc = subprocess.Popen(
            ["bash", REFLECT_SH, "mission"],
            stdin=subprocess.PIPE,
            stdout=logf,
            stderr=subprocess.STDOUT,
            start_new_session=True,
        )
        try:
            proc.stdin.write(payload.encode())
            proc.stdin.flush()
        finally:
            proc.stdin.close()
    # Deliberately no proc.wait() — fire-and-forget, matches idle-handoff.sh /
    # reflect.sh's own detach convention. start_new_session=True (setsid
    # equivalent) means the child survives this hook process exiting.


def main():
    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    cwd = data.get("cwd", "")
    session_id = data.get("session_id", "")
    transcript_path = data.get("transcript_path", "")
    if not cwd or not session_id or not transcript_path:
        sys.exit(0)
    if not os.path.isdir(cwd):
        sys.exit(0)

    if not is_armed(cwd):
        sys.exit(0)  # only autonomous (ralph-armed) sessions get self-compact

    fill = compute_fill(transcript_path)
    if fill is None or fill["fill_pct"] < THRESHOLD:
        sys.exit(0)

    if mission_is_fresh(cwd):
        sys.exit(0)  # already refreshed recently, don't refire every turn

    if not os.path.isfile(REFLECT_SH):
        sys.exit(0)

    fire_mission_refresh(session_id, transcript_path, cwd)

    pct_display = int(fill["fill_pct"] * 100)
    message = (
        f"[self-compact] Context is {pct_display}% full and this is an armed autonomous "
        "mission — tasks/mission.md refresh triggered automatically in the background "
        "(no action needed). It will be current within ~60s. Continue the mission; if "
        "a compact or /clear happens (auto or manual), the next SessionStart will point "
        "you back at the refreshed mission.md/progress.md."
    )
    print(json.dumps({"hookSpecificOutput": {"hookEventName": "UserPromptSubmit", "additionalContext": message}}))


if __name__ == "__main__":
    main()
```
