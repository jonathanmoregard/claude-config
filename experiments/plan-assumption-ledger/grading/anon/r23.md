```python
#!/usr/bin/env python3
"""
Shared context-window fill detection, extracted from context-baggage.py so
scripts/self-compact-loop.sh can reuse the exact same 1M-vs-200K model
detection instead of re-deriving it (and drifting from it) in bash.

Context window detection (in priority order):
    1. CLAUDE_CONTEXT_MAX env var (explicit override, in tokens)
    2. Model ID prefix — Opus/Sonnet 4.x+ run with 1M in this setup
    3. Observed usage — if any assistant turn has exceeded 200k tokens,
       the session is 1M, regardless of model name
    4. Default 200k
"""
import json
import os
import sys

BASE_MAX = 200_000
ONE_M = 1_000_000

# Claude Code model IDs that run at 1M context by default. Prefixes, so
# dated variants (claude-opus-4-1-20250805) match too.
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)

# Explicit 1M marker in the model ID, e.g. `claude-opus-5[1m]`. Checked
# before the prefix list so it keeps working across model generations
# without another edit here.
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model: str | None, observed_max: int) -> int:
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def compute_fill_pct(transcript_path: str) -> tuple[float | None, int, str | None]:
    """Read a transcript JSONL and return (fill_pct, max_tokens, last_model).

    fill_pct is None when no assistant `usage` block has appeared yet
    (too early in the session to judge). Raises FileNotFoundError if
    transcript_path doesn't exist.
    """
    if not os.path.exists(transcript_path):
        raise FileNotFoundError(transcript_path)

    last_usage = None
    last_model = None
    observed_max = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue
            if obj.get("type") != "assistant":
                continue
            msg = obj.get("message", {})
            model = msg.get("model")
            if model and model != "<synthetic>":
                last_model = model
            usage = msg.get("usage")
            if usage:
                last_usage = usage
                total = (
                    usage.get("input_tokens", 0)
                    + usage.get("cache_creation_input_tokens", 0)
                    + usage.get("cache_read_input_tokens", 0)
                )
                if total > observed_max:
                    observed_max = total

    max_tokens = detect_max_tokens(last_model, observed_max)

    if last_usage is None:
        return None, max_tokens, last_model

    total_tokens = (
        last_usage.get("input_tokens", 0)
        + last_usage.get("cache_creation_input_tokens", 0)
        + last_usage.get("cache_read_input_tokens", 0)
    )
    return total_tokens / max_tokens, max_tokens, last_model


def _main():
    if len(sys.argv) != 2:
        print("usage: context_fill.py <transcript_path>", file=sys.stderr)
        sys.exit(1)
    try:
        pct, max_tokens, model = compute_fill_pct(sys.argv[1])
    except FileNotFoundError:
        sys.exit(1)
    pct_field = f"{pct:.6f}" if pct is not None else "-"
    model_field = model if model else "-"
    print(f"{pct_field} {max_tokens} {model_field}")


if __name__ == "__main__":
    _main()
```

```bash
#!/usr/bin/env bash
# self-compact-loop.sh — external supervisor for unattended autonomous
# missions: reset context via tasks/mission.md before quality degrades,
# instead of accumulating context forever in one process (what the
# in-session ralph-loop Stop hook does) or waiting for Claude Code's own
# late auto-compact.
#
# Each iteration:
#   1. Runs `claude -p` — resuming the prior iteration's session
#      (--resume --fork-session, cache-warm) if context fill is below
#      threshold, or starting a brand-new session if the PRIOR iteration
#      crossed threshold (mission.md was already rewritten before this
#      iteration started).
#   2. Checks the iteration's result text for
#      <promise>COMPLETION_PROMISE</promise> — stop, success.
#   3. Computes context fill via context_fill.py against the new
#      session's transcript.
#   4. If fill >= SELF_COMPACT_THRESHOLD: calls `reflect.sh mission`
#      synchronously (writes tasks/mission.md from the session that just
#      finished, cache-warm) and marks the NEXT iteration to start fresh.
#
# No separate state file: tasks/mission.md (+ tasks/progress.md,
# tasks/session-constraints.md if present) IS the state. If this script
# is killed and re-run, the next `claude -p` reads whatever mission.md
# says and continues — the same recovery path a threshold-triggered reset
# already exercises every time.
#
# SAFETY: runs `claude -p --dangerously-skip-permissions` — no TTY exists
# to approve prompts in a headless loop. This mirrors the trust boundary
# skills/orchestrate/SKILL.md's Phase 0 already accepts for the in-session
# ralph-loop; the worktree-only preflight below is the SAME guard
# orchestrate uses to bound blast radius. Do not relax that guard.
#
# Usage:
#   scripts/self-compact-loop.sh --completion-promise "<sigil>" \
#     [--max-iterations N] [--threshold FRACTION]
#
# Requires: tasks/mission.md already exists (written by the orchestrator
# per skills/orchestrate/SKILL.md Phase 0, BEFORE this script is invoked).
#
# Env:
#   SELF_COMPACT_THRESHOLD   Fraction of context fill that triggers a
#                            reset (default: 0.30). --threshold overrides.
#   REFLECT_SH               Path to reflect.sh (default:
#                            $HOME/.claude/skills/session-reflect/reflect.sh)
#   CONTEXT_FILL_PY          Path to context_fill.py (default: this
#                            script's own repo root)
#   REFLECT_ROOT             Artifact root for logs (default:
#                            $HOME/.claude/reflections)
#   SELF_COMPACT_LOG_DIR     (default: $REFLECT_ROOT/self-compact-logs)
#
# Cancel: touch .claude/self-compact.cancel in the worktree root — checked
# at the top of every iteration.
set -euo pipefail
umask 077

SCRIPT_DIR="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)"
REPO_ROOT="$(cd -- "$SCRIPT_DIR/.." && pwd)"

: "${SELF_COMPACT_THRESHOLD:=0.30}"
: "${REFLECT_SH:=$HOME/.claude/skills/session-reflect/reflect.sh}"
: "${CONTEXT_FILL_PY:=$REPO_ROOT/context_fill.py}"
: "${REFLECT_ROOT:=$HOME/.claude/reflections}"
: "${SELF_COMPACT_LOG_DIR:=$REFLECT_ROOT/self-compact-logs}"

COMPLETION_PROMISE=""
MAX_ITERATIONS=25
THRESHOLD="$SELF_COMPACT_THRESHOLD"

while [[ $# -gt 0 ]]; do
  case "$1" in
    --completion-promise) COMPLETION_PROMISE="$2"; shift 2 ;;
    --max-iterations) MAX_ITERATIONS="$2"; shift 2 ;;
    --threshold) THRESHOLD="$2"; shift 2 ;;
    *) echo "self-compact-loop.sh: unknown arg $1" >&2; exit 2 ;;
  esac
done

if [[ -z "$COMPLETION_PROMISE" ]]; then
  echo "REFUSE: --completion-promise is required (mirrors ralph-loop's --completion-promise)" >&2
  exit 1
fi

# --- Preflight: same worktree-only guard as orchestrate's Phase 0 ----------
case "$PWD" in
  "$HOME"|"$HOME/.claude"|"$HOME/Repos")
    echo "REFUSE: arm only from a dedicated worktree root (got: $PWD)" >&2
    exit 1
    ;;
esac
[[ -e .git ]] || { echo "REFUSE: not a repo/worktree root (no .git)" >&2; exit 1; }
[[ -f tasks/mission.md ]] || {
  echo "REFUSE: tasks/mission.md does not exist. Write it first (see skills/orchestrate/SKILL.md Phase 0 step 2) — self-compact-loop.sh continues a mission, it doesn't create one." >&2
  exit 1
}
[[ -x "$REFLECT_SH" ]] || { echo "REFUSE: REFLECT_SH not found/executable: $REFLECT_SH" >&2; exit 1; }
[[ -f "$CONTEXT_FILL_PY" ]] || { echo "REFUSE: CONTEXT_FILL_PY not found: $CONTEXT_FILL_PY" >&2; exit 1; }

mkdir -p "$SELF_COMPACT_LOG_DIR"
WORKTREE_CWD="$PWD"
CANCEL_FLAG=".claude/self-compact.cancel"

CONTINUE_PROMPT="Continue the mission autonomously. Read tasks/mission.md, tasks/progress.md, and tasks/session-constraints.md if present — trust files and git log over conversation memory. Do not ask questions: pick the most reasonable interpretation, document the assumption, and proceed. When every acceptance criterion in tasks/mission.md is verified complete and the verifier is green, output <promise>${COMPLETION_PROMISE}</promise> and nothing else."

prior_session_id=""
iteration=0

while (( iteration < MAX_ITERATIONS )); do
  if [[ -f "$CANCEL_FLAG" ]]; then
    echo "self-compact-loop: cancel flag found ($CANCEL_FLAG) — stopping."
    exit 0
  fi

  iteration=$(( iteration + 1 ))
  iter_log="$SELF_COMPACT_LOG_DIR/$(date -u +%Y%m%dT%H%M%SZ)-iter${iteration}.stderr.log"

  claude_args=(-p --output-format json --dangerously-skip-permissions --model opus)
  if [[ -n "$prior_session_id" ]]; then
    claude_args+=(--resume "$prior_session_id" --fork-session)
  fi

  echo "self-compact-loop: iteration $iteration $( [[ -n "$prior_session_id" ]] && echo "(resume $prior_session_id)" || echo "(fresh session)" )"

  claude_json="$(printf '%s' "$CONTINUE_PROMPT" | claude "${claude_args[@]}" 2>>"$iter_log" || true)"

  session_id="$(printf '%s' "$claude_json" | jq -r '.session_id // empty' 2>/dev/null || true)"
  result_text="$(printf '%s' "$claude_json" | jq -r '.result // empty' 2>/dev/null || true)"

  if [[ -z "$session_id" ]]; then
    echo "self-compact-loop: iteration $iteration produced no session_id (see $iter_log) — stopping." >&2
    exit 1
  fi

  # --- Sigil check (same extraction as ralph-loop's stop-hook.sh) --------
  promise_text="$(printf '%s' "$result_text" | perl -0777 -pe 's/.*?<promise>(.*?)<\/promise>.*/$1/s; s/^\s+|\s+$//g; s/\s+/ /g' 2>/dev/null || echo "")"
  if [[ -n "$promise_text" ]] && [[ "$promise_text" = "$COMPLETION_PROMISE" ]]; then
    echo "self-compact-loop: sigil matched at iteration $iteration — mission complete."
    exit 0
  fi

  # --- Locate the new session's transcript --------------------------------
  transcript_path="$(find "$HOME/.claude/projects" -maxdepth 2 -type f -name "${session_id}.jsonl" 2>/dev/null | head -1)"
  if [[ -z "$transcript_path" ]]; then
    echo "self-compact-loop: could not locate transcript for session $session_id — continuing without a fill check this iteration." >&2
    prior_session_id="$session_id"
    continue
  fi

  fill_line="$(python3 "$CONTEXT_FILL_PY" "$transcript_path" 2>/dev/null || true)"
  fill_pct="$(printf '%s' "$fill_line" | awk '{print $1}')"

  if [[ -z "$fill_pct" || "$fill_pct" == "-" ]]; then
    prior_session_id="$session_id"
    continue
  fi

  over_threshold="$(awk -v f="$fill_pct" -v t="$THRESHOLD" 'BEGIN{print (f >= t) ? "1" : "0"}')"

  if [[ "$over_threshold" == "1" ]]; then
    echo "self-compact-loop: fill=${fill_pct} >= threshold=${THRESHOLD} — writing tasks/mission.md and resetting."
    mission_payload="$(jq -cn \
      --arg sid "$session_id" --arg tp "$transcript_path" --arg cwd "$WORKTREE_CWD" \
      '{session_id:$sid, transcript_path:$tp, hook_event_name:"SelfCompact", cwd:$cwd}')"
    printf '%s' "$mission_payload" | env -u MODE bash "$REFLECT_SH" mission \
      >> "$iter_log" 2>&1 || echo "self-compact-loop: reflect.sh mission failed (see $iter_log) — proceeding with a reset anyway; next iteration reads whatever mission.md currently holds." >&2
    prior_session_id=""   # next iteration starts fresh, no --resume
  else
    echo "self-compact-loop: fill=${fill_pct} < threshold=${THRESHOLD} — continuing via resume."
    prior_session_id="$session_id"
  fi
done

echo "self-compact-loop: max iterations ($MAX_ITERATIONS) reached without the sigil."
exit 0
```
