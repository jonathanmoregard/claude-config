# before the prefix list so it keeps working across model generations
# without another edit here.
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model, observed_max, override=None):
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def scan_transcript(transcript_path):
    """Returns dict: user_msg_count, last_usage, last_model, observed_max, file_size."""
    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0
    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue
            obj_type = obj.get("type")
            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = (
                        usage.get("input_tokens", 0)
                        + usage.get("cache_creation_input_tokens", 0)
                        + usage.get("cache_read_input_tokens", 0)
                    )
                    if total > observed_max:
                        observed_max = total
    file_size = os.path.getsize(transcript_path)
    return {
        "user_msg_count": user_msg_count,
        "last_usage": last_usage,
        "last_model": last_model,
        "observed_max": observed_max,
        "file_size": file_size,
    }


def compute_fill_pct(last_usage, max_tokens, file_size):
    if last_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )
        return total_tokens / max_tokens
    return file_size / (max_tokens * 4)


def fill_for_transcript(transcript_path, override_max=None):
    """One-shot: scan + detect + compute."""
    scan = scan_transcript(transcript_path)
    max_tokens = detect_max_tokens(scan["last_model"], scan["observed_max"], override_max)
    fill_pct = compute_fill_pct(scan["last_usage"], max_tokens, scan["file_size"])
    total_tokens = None
    if scan["last_usage"]:
        total_tokens = (
            scan["last_usage"].get("input_tokens", 0)
            + scan["last_usage"].get("cache_creation_input_tokens", 0)
            + scan["last_usage"].get("cache_read_input_tokens", 0)
        )
    return {
        "fill_pct": fill_pct,
        "max_tokens": max_tokens,
        "user_msg_count": scan["user_msg_count"],
        "last_model": scan["last_model"],
        "total_tokens": total_tokens,
    }


def main():
    if len(sys.argv) < 2:
        print("usage: context_fill.py <transcript_path>", file=sys.stderr)
        sys.exit(2)
    transcript_path = sys.argv[1]
    if not os.path.exists(transcript_path):
        print(json.dumps({"error": "transcript_not_found"}))
        sys.exit(1)
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    result = fill_for_transcript(transcript_path, override)
    print(json.dumps(result))


if __name__ == "__main__":
    main()
```

#!/usr/bin/env bash
# Stop hook: autonomous self-compact trigger for ralph-armed sessions.
#
# When context fill crosses SELF_COMPACT_THRESHOLD (default 30% of the
# detected context window) in a session with an ARMED ralph loop
# (.claude/ralph-loop.local.md present in cwd), this hook:
#   1. writes tasks/mission.md synchronously (reflect.sh mission, same
#      nested-call pattern reflect.sh's own 2nd-compact defence uses), then
#   2. returns {"decision":"block","reason":...} instructing Claude to call
#      the SlashCommand tool with "/compact ..." as its entire next action.
#
# Scope: interactive (non-armed) sessions are untouched — they already get
# context-baggage.py's 70% ask-the-user nudge, and a human is present to
# decide. Self-compact only acts where nobody is watching and the user has
# already pre-approved unattended operation by arming the loop.
#
# Idempotency: a per-project marker (.claude/self-compact.pending) tracks
# attempts so a Stop event that fires every ralph iteration doesn't spam
# the /compact instruction forever if the model doesn't act on it. After
# SELF_COMPACT_MAX_ATTEMPTS (default 3) consecutive Stops still above
# threshold, this hook gives up silently and lets ralph's own Stop hook
# run the loop normally — logged once to SELF_COMPACT_ROOT/warnings.log.
# skills/orchestrate/hooks/compact-reinject.sh clears the marker once an
# actual compaction/clear/restart happens, resetting the retry cycle.
set -euo pipefail
umask 077

: "${SELF_COMPACT_THRESHOLD:=0.30}"
: "${SELF_COMPACT_MAX_ATTEMPTS:=3}"
: "${SELF_COMPACT_ROOT:=$HOME/.claude/self-compact}"
: "${SELF_COMPACT_DISABLE_FLAG:=$HOME/.claude/self-compact.disable}"
: "${SELF_COMPACT_REFLECT_SH:=$HOME/.claude/skills/session-reflect/reflect.sh}"
SCRIPT_DIR="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)"
CONTEXT_FILL_PY="$SCRIPT_DIR/../lib/context_fill.py"

[[ -f "$SELF_COMPACT_DISABLE_FLAG" ]] && exit 0

PAYLOAD="$(cat 2>/dev/null || true)"
[[ -n "$PAYLOAD" ]] || exit 0

SESSION_ID="$(printf '%s' "$PAYLOAD" | jq -r '.session_id // empty' 2>/dev/null || true)"
TRANSCRIPT_PATH="$(printf '%s' "$PAYLOAD" | jq -r '.transcript_path // empty' 2>/dev/null || true)"
CWD="$(printf '%s' "$PAYLOAD" | jq -r '.cwd // empty' 2>/dev/null || true)"

[[ -n "$SESSION_ID" && -n "$TRANSCRIPT_PATH" && -n "$CWD" && -d "$CWD" ]] || exit 0
[[ -f "$TRANSCRIPT_PATH" ]] || exit 0

STATE="$CWD/.claude/ralph-loop.local.md"
[[ -f "$STATE" ]] || exit 0   # not an armed autonomous session — self-compact is opt-in via arming

MARKER="$CWD/.claude/self-compact.pending"

FILL_JSON="$(python3 "$CONTEXT_FILL_PY" "$TRANSCRIPT_PATH" 2>/dev/null || echo '{}')"
FILL_PCT="$(printf '%s' "$FILL_JSON" | jq -r '.fill_pct // empty' 2>/dev/null || true)"
[[ -n "$FILL_PCT" ]] || exit 0   # can't read transcript — fail open, no-op

BELOW_THRESHOLD=$(awk -v f="$FILL_PCT" -v t="$SELF_COMPACT_THRESHOLD" 'BEGIN{print (f < t) ? 1 : 0}')
if [[ "$BELOW_THRESHOLD" == "1" ]]; then
  [[ -f "$MARKER" ]] && rm -f -- "$MARKER"
  exit 0
fi

mkdir -p "$(dirname -- "$MARKER")" 2>/dev/null || exit 0

ATTEMPTS=0
GAVE_UP=false
if [[ -f "$MARKER" ]]; then
  ATTEMPTS="$(jq -r '.attempts // 0' "$MARKER" 2>/dev/null || echo 0)"
  GAVE_UP="$(jq -r '.gave_up // false' "$MARKER" 2>/dev/null || echo false)"
  [[ "$ATTEMPTS" =~ ^[0-9]+$ ]] || ATTEMPTS=0
fi

[[ "$GAVE_UP" == "true" ]] && exit 0   # already warned once this cycle; wait for compact-reinject to clear it

if (( ATTEMPTS >= SELF_COMPACT_MAX_ATTEMPTS )); then
  mkdir -p "$SELF_COMPACT_ROOT" 2>/dev/null || true
  printf '%s %s attempts=%s fill_pct=%s cwd=%s — giving up, letting ralph continue\n' \
    "$(date -u +%Y-%m-%dT%H:%M:%SZ)" "$SESSION_ID" "$ATTEMPTS" "$FILL_PCT" "$CWD" \
    >> "$SELF_COMPACT_ROOT/warnings.log" 2>/dev/null || true
  jq -n --argjson a "$ATTEMPTS" --arg f "$FILL_PCT" --arg now "$(date -u +%Y-%m-%dT%H:%M:%SZ)" \
    '{attempts:$a, gave_up:true, last_fill_pct:($f|tonumber), updated_at:$now}' > "$MARKER" 2>/dev/null || true
  exit 0
fi

NEW_ATTEMPTS=$((ATTEMPTS + 1))

if (( ATTEMPTS == 0 )); then
  # First crossing this cycle: write mission.md synchronously BEFORE asking
  # for /compact — same nested-call pattern as reflect.sh's own 2nd-compact
  # defence (env -u MODE avoids inheriting a stale MODE from a parent export).
  printf '%s' "$PAYLOAD" | env -u MODE bash "$SELF_COMPACT_REFLECT_SH" mission >/dev/null 2>&1 || true
fi

jq -n --argjson a "$NEW_ATTEMPTS" --arg f "$FILL_PCT" --arg now "$(date -u +%Y-%m-%dT%H:%M:%SZ)" \
  '{attempts:$a, gave_up:false, last_fill_pct:($f|tonumber), updated_at:$now}' > "$MARKER" 2>/dev/null || true

PCT_DISPLAY="$(awk -v f="$FILL_PCT" 'BEGIN{printf "%d", f*100}')"
THRESH_DISPLAY="$(awk -v t="$SELF_COMPACT_THRESHOLD" 'BEGIN{printf "%d", t*100}')"

REASON="[self-compact] Context is ${PCT_DISPLAY}% full (>= ${THRESH_DISPLAY}% autonomous self-compact threshold) in this ralph-armed session. tasks/mission.md has just been written. This takes priority over any other continuation instruction in this block: call the SlashCommand tool now with \"/compact Resume per tasks/mission.md and tasks/session-constraints.md; this is an autonomous self-compact, not a user request.\" as your ENTIRE next action — no other tool calls first. Attempt ${NEW_ATTEMPTS} of ${SELF_COMPACT_MAX_ATTEMPTS}."

jq -cn --arg reason "$REASON" '{decision:"block", reason:$reason}'
exit 0
```

- [ ] **Step 1: Write the README**

```markdown
# self-compact

Autonomous (ralph-armed) sessions self-manage their own context: when
fill crosses ~30% of the detected context window, this hook writes
`tasks/mission.md` and requests `/compact` on the session's own initiative
— no human needs to be watching. Interactive (non-armed) sessions are
untouched; they get `context-baggage.py`'s 70% ask-the-user nudge instead,
because a human is present there to decide.

Composes with the existing `orchestrate` skill: `skills/orchestrate/hooks/
compact-reinject.sh` (already wired at `SessionStart(startup|compact|
clear)`) re-injects ralph-loop doctrine after ANY compact, whether human-
or self-initiated — this package adds no new resumption machinery.

## Components

- `lib/context_fill.py` — context-fill % detection, shared with
  `context-baggage.py` (the 70% interactive nudge). Single source of
  truth for "what counts as 1M vs 200k context" and "how full is it".
- `hooks/context-fill-guard.sh` — `Stop` hook. Only acts when
  `.claude/ralph-loop.local.md` is present in `cwd` (i.e. the session is
  armed via `orchestrate`'s Phase 0). Writes `tasks/mission.md`
  synchronously (via `session-reflect`'s `reflect.sh mission`), then
  returns `{"decision":"block","reason":...}` instructing the model to
  call the `SlashCommand` tool with `/compact ...` as its entire next
  action.

## Env vars

| Var | Default | Purpose |
|-----|---------|---------|
| `SELF_COMPACT_THRESHOLD` | `0.30` | Fraction of the detected context window that triggers self-compact. |
| `SELF_COMPACT_MAX_ATTEMPTS` | `3` | Consecutive Stop events still above threshold before giving up and letting ralph's own Stop hook continue the loop unassisted. |
| `SELF_COMPACT_ROOT` | `$HOME/.claude/self-compact` | Where give-up warnings are logged (`warnings.log`). |
| `SELF_COMPACT_DISABLE_FLAG` | `$HOME/.claude/self-compact.disable` | Kill switch — touch this file to pause self-compact globally without editing settings.json. |
| `SELF_COMPACT_REFLECT_SH` | `$HOME/.claude/skills/session-reflect/reflect.sh` | Override for testing. |

## State

`.claude/self-compact.pending` (project-scoped, next to `ralph-loop.local.md`):
`{"attempts": int, "gave_up": bool, "last_fill_pct": float, "updated_at": ISO8601}`.
Cleared by `skills/orchestrate/hooks/compact-reinject.sh` on every
`SessionStart(startup|compact|clear)` — a real compaction event resets the
retry cycle regardless of who initiated it.

## Failure modes

- **Transcript unreadable / malformed payload.** Fails open — exit 0, no
  block, no marker write.
- **`reflect.sh mission` fails or times out.** Logged to its own worker
  log (`~/.claude/reflections/logs/`); self-compact still requests
  `/compact` even if the mission write failed (`|| true`) — a degraded
  handoff beats none, and the model still has `progress.md`/
  `session-constraints.md` from the ralph doctrine.
- **Model ignores the `/compact` instruction.** After
  `SELF_COMPACT_MAX_ATTEMPTS` consecutive Stops still above threshold,
  this hook gives up silently (one line to `warnings.log`) and steps
  aside — ralph's own Stop hook keeps the loop running normally. Not a
  hard failure: the session just runs hotter than intended until it
  either naturally compacts or hits max-iterations.
- **Two Stop hooks (self-compact + ralph-loop plugin) both return
  `block` on the same turn.** Claude Code surfaces both reasons; the
  self-compact reason explicitly states it takes priority. Validated in
  Task 4 of the implementation plan
  (`docs/superpowers/plans/2026-08-17-self-compact-and-rsi-reflect.md`).
