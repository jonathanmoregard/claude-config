## Assumptions

- **UserPromptSubmit hook payload includes `session_id` and `cwd`.** `context-baggage.py` today only reads `transcript_path`; other hooks in this repo (`aggregator-recall-hook.py`, `reflect.sh`'s own payload contract) rely on `session_id`/`cwd` being present on hook payloads generally, and `reflect.sh` itself requires both for `mission` mode. #!/usr/bin/env python3
"""Shared context-window-fill detection.

Extracted from context-baggage.py so the logic is unit-testable and reusable
by the self-compact branch inside it. Pure functions only — no subprocess,
no writes; scan_transcript does read-only file I/O.
"""
from __future__ import annotations

import json
import re

BASE_MAX = 200_000
ONE_M = 1_000_000

# Claude Code model IDs that run at 1M context by default. Prefixes, so
# dated variants (claude-opus-4-1-20250805) match too.
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)

# Explicit 1M marker in the model ID, e.g. `claude-opus-5[1m]`. Checked
# before the prefix list so it keeps working across model generations
# without another edit here.
ONE_M_MARKER = "[1m]"

_SID_RE = re.compile(r"^[A-Za-z0-9._-]{1,128}$")


def detect_max_tokens(model: str | None, observed_max: int, override: str | None = None) -> int:
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def usage_total(usage: dict) -> int:
    return (
        usage.get("input_tokens", 0)
        + usage.get("cache_creation_input_tokens", 0)
        + usage.get("cache_read_input_tokens", 0)
    )


def fill_pct(total_tokens: int, max_tokens: int) -> float:
    if max_tokens <= 0:
        return 0.0
    return total_tokens / max_tokens


def scan_transcript(path: str) -> dict:
    """Read-only scan of a Claude Code transcript JSONL. Malformed lines
    are skipped, not fatal — a partially-written transcript (session still
    live) must not crash the caller.
    """
    import os

    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0

    with open(path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue

            obj_type = obj.get("type")
            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = usage_total(usage)
                    if total > observed_max:
                        observed_max = total

    return {
        "user_msg_count": user_msg_count,
        "last_usage": last_usage,
        "last_model": last_model,
        "observed_max": observed_max,
        "file_size": os.path.getsize(path),
    }


def is_valid_session_id(sid: str) -> bool:
    if not sid or sid.startswith("."):
        return False
    return bool(_SID_RE.match(sid))
```

#!/usr/bin/env python3
"""
UserPromptSubmit hook: warns when context is over THRESHOLD with potentially
stale/off-topic content. Fires every MESSAGE_INTERVAL user messages once
threshold is met.

Context window detection is delegated to context_fill.py (priority order
documented there): explicit override, model-id prefix/marker, observed
usage, default 200k.
"""
import json
import os
import sys
from pathlib import Path

sys.path.insert(0, str(Path.home() / ".claude"))
import context_fill as cf  # noqa: E402

# Raised from 0.25 -> 0.70 on 2026-08-08 after measuring 95.5% cache-hit
# ratio across 7 days / $3,073 spend (probes/analyze_sessions.py). At that
# hit ratio, holding baggage is cheap and resetting is expensive
# (cache_write ~$5-10 per cold event). Nudge should fire when handoff to
# a fresh session starts to pay off, not on any drift.
THRESHOLD = 0.70
MESSAGE_INTERVAL = 5


def build_message(pct_display: int, max_tokens: int, user_msg_count: int) -> str:
    return (
        f"Context is {pct_display}% full ({user_msg_count} messages in this session; "
        f"window: {max_tokens // 1000}k tokens). "
        "At this fill, a fresh-session handoff via tasks/mission.md is usually cheaper "
        "than an in-place /compact: --fork-session on a warm cache measured ~77% hit "
        "(session-reflect empirical), while /compact invalidates the prefix and forces "
        "cache_write on subsequent turns. "
        "Suggest to the user: (a) write tasks/mission.md (done / next / blocked / open "
        "questions), (b) start a fresh session in the same cwd and load it, OR (c) "
        "continue here and accept the drift. Ask which."
    )


def main():
    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    transcript_path = data.get("transcript_path", "")
    if not transcript_path or not os.path.exists(transcript_path):
        sys.exit(0)

    scan = cf.scan_transcript(transcript_path)
    user_msg_count = scan["user_msg_count"]
    last_usage = scan["last_usage"]
    last_model = scan["last_model"]
    observed_max = scan["observed_max"]
    file_size = scan["file_size"]

    if user_msg_count == 0 or user_msg_count % MESSAGE_INTERVAL != 0:
        sys.exit(0)

    max_tokens = cf.detect_max_tokens(last_model, observed_max, override=os.environ.get("CLAUDE_CONTEXT_MAX"))

    if last_usage:
        total_tokens = cf.usage_total(last_usage)
        fill = cf.fill_pct(total_tokens, max_tokens)
    else:
        fill = file_size / (max_tokens * 4)

    if fill < THRESHOLD:
        sys.exit(0)

    pct_display = int(fill * 100)
    message = build_message(pct_display, max_tokens, user_msg_count)
    print(json.dumps({"hookSpecificOutput": {"hookEventName": "UserPromptSubmit", "additionalContext": message}}))


if __name__ == "__main__":
    main()

REFLECT_SH = os.environ.get(
    "SELF_COMPACT_REFLECT_SH", str(Path.home() / ".claude" / "skills" / "session-reflect" / "reflect.sh")
)
MARKER_DIR = Path(os.environ.get("SELF_COMPACT_MARKER_DIR", str(Path.home() / ".claude" / "reflections" / "self-compact")))
DISABLE_FLAG = Path(os.environ.get("SELF_COMPACT_DISABLE_FLAG", str(Path.home() / ".claude" / "self-compact.disable")))
COOLDOWN_SECONDS = int(os.environ.get("SELF_COMPACT_COOLDOWN_SECONDS", "1200"))
MISSION_TIMEOUT_SECONDS = int(os.environ.get("SELF_COMPACT_MISSION_TIMEOUT_SECONDS", "75"))


def _marker_path(session_id: str) -> Path:
    safe = "".join(c for c in session_id if c.isalnum() or c in "._-")[:128]
    return MARKER_DIR / f"{safe}.marker"


def _cooldown_active(session_id: str) -> bool:
    p = _marker_path(session_id)
    if not p.exists():
        return False
    try:
        age = time.time() - p.stat().st_mtime
    except OSError:
        return False
    return age < COOLDOWN_SECONDS


def _write_marker(session_id: str) -> None:
    try:
        MARKER_DIR.mkdir(parents=True, exist_ok=True)
        _marker_path(session_id).write_text(str(time.time()))
    except OSError:
        pass


def try_self_compact(session_id: str, transcript_path: str, cwd: str) -> bool:
    """Synchronously invoke `reflect.sh mission`. Success is verified by
    checking tasks/mission.md's mtime actually advanced — reflect.sh's own
    malformed-payload path exits 0 without writing anything, so exit code
    alone is not proof of a write.
    """
    mission_path = Path(cwd) / "tasks" / "mission.md"
    before_mtime = mission_path.stat().st_mtime if mission_path.exists() else -1.0

    payload = json.dumps(
        {
            "session_id": session_id,
            "transcript_path": transcript_path,
            "hook_event_name": "SelfCompact",
            "cwd": cwd,
        }
    )
    try:
        subprocess.run(
            ["bash", REFLECT_SH, "mission"],
            input=payload,
            capture_output=True,
            text=True,
            timeout=MISSION_TIMEOUT_SECONDS,
        )
    except (subprocess.TimeoutExpired, OSError):
        return False

    after_mtime = mission_path.stat().st_mtime if mission_path.exists() else -1.0
    return after_mtime > before_mtime


def build_self_compact_message(pct_display: int, max_tokens: int) -> str:
    return (
        f"Self-compact: context is {pct_display}% full ({max_tokens // 1000}k window). "
        "tasks/mission.md has just been refreshed (done/next/blocked/open questions) so "
        "this session can hand off cleanly. Do not ask the user whether to continue or "
        "compact — wrap up the current step in 1-2 lines, then keep working from "
        "mission.md's Next item. If this context compacts (auto or manual) or the "
        "session restarts, tasks/mission.md is authoritative — resume immediately "
        "without asking permission, per CLAUDE.md's autonomy default. Only stop for "
        "something genuinely irreversible, credentialed, or already marked blocked."
    )


def main():
    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    transcript_path = data.get("transcript_path", "")
    if not transcript_path or not os.path.exists(transcript_path):
        sys.exit(0)

    scan = cf.scan_transcript(transcript_path)
    user_msg_count = scan["user_msg_count"]
    last_usage = scan["last_usage"]
    last_model = scan["last_model"]
    observed_max = scan["observed_max"]
    file_size = scan["file_size"]

    if user_msg_count == 0 or user_msg_count % MESSAGE_INTERVAL != 0:
        sys.exit(0)

    max_tokens = cf.detect_max_tokens(last_model, observed_max, override=os.environ.get("CLAUDE_CONTEXT_MAX"))

    if last_usage:
        total_tokens = cf.usage_total(last_usage)
        fill = cf.fill_pct(total_tokens, max_tokens)
    else:
        fill = file_size / (max_tokens * 4)

    if fill < THRESHOLD:
        sys.exit(0)

    pct_display = int(fill * 100)
    session_id = data.get("session_id", "") or ""
    cwd = data.get("cwd", "") or ""

    if not DISABLE_FLAG.exists() and session_id:
        if _cooldown_active(session_id):
            sys.exit(0)  # already handed off recently; stay quiet
        if cf.is_valid_session_id(session_id) and cwd and os.path.isdir(cwd):
            ok = try_self_compact(session_id, transcript_path, cwd)
            _write_marker(session_id)
            if ok:
                message = build_self_compact_message(pct_display, max_tokens)
                print(json.dumps({"hookSpecificOutput": {"hookEventName": "UserPromptSubmit", "additionalContext": message}}))
                return

    # Legacy path: disabled, no usable session_id/cwd, or the write failed.
    message = build_message(pct_display, max_tokens, user_msg_count)
    print(json.dumps({"hookSpecificOutput": {"hookEventName": "UserPromptSubmit", "additionalContext": message}}))
```
