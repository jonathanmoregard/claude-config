```python
# context_fill_lib.py
"""Shared context-fill detection for UserPromptSubmit hooks.

Used by self-compact.py. Extracted from the retired context-baggage.py so
the fill-percentage math and model-window heuristics can't drift between
call sites — a THRESHOLD tuning here (last changed 0.25 -> 0.70 on
2026-08-08, see git history) previously lived duplicated in one file only.
"""
import json
import os

BASE_MAX = 200_000
ONE_M = 1_000_000

# Claude Code model IDs that run at 1M context by default. Prefixes, so
# dated variants (claude-opus-4-1-20250805) match too.
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)
# Explicit 1M marker in the model ID, e.g. `claude-opus-5[1m]`.
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model, observed_max):
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def compute_fill(transcript_path):
    """Parse a transcript JSONL and return fill stats, or None if unreadable.

    Returns a dict with keys: fill_pct, user_msg_count, max_tokens,
    last_model, total_tokens.
    """
    if not transcript_path or not os.path.exists(transcript_path):
        return None

    file_size = os.path.getsize(transcript_path)
    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue

            obj_type = obj.get("type")
            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = (
                        usage.get("input_tokens", 0)
                        + usage.get("cache_creation_input_tokens", 0)
                        + usage.get("cache_read_input_tokens", 0)
                    )
                    if total > observed_max:
                        observed_max = total

    max_tokens = detect_max_tokens(last_model, observed_max)

    if last_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )
        fill_pct = total_tokens / max_tokens
    else:
        total_tokens = 0
        fill_pct = file_size / (max_tokens * 4)

    return {
        "fill_pct": fill_pct,
        "user_msg_count": user_msg_count,
        "max_tokens": max_tokens,
        "last_model": last_model,
        "total_tokens": total_tokens,
    }
```

```python
#!/usr/bin/env python3
# self-compact.py
"""UserPromptSubmit hook: at high context fill, proactively (and silently)
arm a mission.md handoff instead of asking the user which of three options
they want.

Supersedes context-baggage.py's advisory nudge (retired — see settings.json
and git history). Where context-baggage.py stopped at "ask the user",
self-compact writes a marker and injects instructions telling Claude to
write tasks/mission.md itself, this turn, without asking — then continue
the user's actual request.
skills/orchestrate/hooks/compact-reinject.sh reads the marker on the next
SessionStart(compact) and upgrades its reminder to "proceed with Next now"
for a fresh marker.

Kill switch: touch ~/.claude/self-compact.disable to pause without editing
settings.json.
"""
import datetime
import json
import os
import sys
import tempfile

from context_fill_lib import compute_fill

THRESHOLD = float(os.environ.get("SELF_COMPACT_THRESHOLD", "0.70"))
MESSAGE_INTERVAL = int(os.environ.get("SELF_COMPACT_MESSAGE_INTERVAL", "5"))
MISSION_FRESH_SECONDS = int(os.environ.get("SELF_COMPACT_MISSION_FRESH_SECONDS", "600"))
DISABLE_FLAG = os.path.expanduser("~/.claude/self-compact.disable")


def atomic_write(path, text):
    d = os.path.dirname(path)
    os.makedirs(d, exist_ok=True)
    fd, tmp = tempfile.mkstemp(dir=d, prefix=".self-compact-tmp-")
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as fh:
            fh.write(text)
        os.replace(tmp, path)
    except BaseException:
        try:
            os.unlink(tmp)
        except OSError:
            pass
        raise


def build_context(fill_pct, max_tokens):
    pct_display = int(fill_pct * 100)
    return (
        f"Context is {pct_display}% full (window: {max_tokens // 1000}k tokens). "
        "Before anything else this turn, write/refresh tasks/mission.md via the "
        "Write tool — this is routine self-maintenance, do not ask the user for "
        "permission. Use this structure:\n\n"
        "# Mission — <one-line project + current focus>\n\n"
        "## Done\n- Concrete artifacts shipped this session (file paths, shas, "
        "test names). No plans or intents.\n\n"
        "## Next\n- The single next action, executable, then 2-3 more in "
        "dependency order.\n\n"
        "## Blocked\n- Anything waiting on a person/credential/review, and what "
        "unblocks it. (none) if empty.\n\n"
        "## Open questions\n- Unresolved design/scope questions with a "
        "best-guess answer. (none) if empty.\n\n"
        "## State on disk\n- Uncommitted files, scratch dirs later work depends "
        "on. (none) if clean.\n\n"
        "After writing it, continue the user's request below normally. "
        "Compaction (automatic or manual) may happen soon — tasks/mission.md "
        "and tasks/.self-compact-armed are the resumption anchor, not this "
        "conversation."
    )


def main():
    if os.path.exists(DISABLE_FLAG):
        return 0
    try:
        data = json.load(sys.stdin)
    except Exception:
        return 0

    transcript_path = data.get("transcript_path", "")
    cwd = data.get("cwd", "")
    session_id = data.get("session_id", "")
    if not cwd or not os.path.isdir(cwd):
        return 0

    stats = compute_fill(transcript_path)
    if stats is None:
        return 0

    if stats["user_msg_count"] == 0 or stats["user_msg_count"] % MESSAGE_INTERVAL != 0:
        return 0
    if stats["fill_pct"] < THRESHOLD:
        return 0

    mission_path = os.path.join(cwd, "tasks", "mission.md")
    if os.path.isfile(mission_path):
        age = datetime.datetime.now().timestamp() - os.path.getmtime(mission_path)
        if age < MISSION_FRESH_SECONDS:
            return 0  # already armed recently this cycle

    marker_path = os.path.join(cwd, "tasks", ".self-compact-armed")
    marker = json.dumps({
        "armed_at": datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ"),
        "fill_pct": round(stats["fill_pct"], 4),
        "session_id": session_id,
    })
    try:
        atomic_write(marker_path, marker + "\n")
    except OSError:
        return 0  # can't write the marker; skip silently rather than half-arm

    message = build_context(stats["fill_pct"], stats["max_tokens"])
    json.dump({
        "hookSpecificOutput": {
            "hookEventName": "UserPromptSubmit",
            "additionalContext": message,
        }
    }, sys.stdout)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
#!/usr/bin/env python3
# scripts/reflection-digest.py
"""Aggregate session-reflect artifacts into a daily digest for RSI review.

Trusted, stdlib-only script (mirrors scripts/proposal-intake-gate.py's
separate-scorer discipline): grouping and dedup here are deterministic
string matching, not a model call. Reads only from
~/.claude/reflections/sessions/; the review agent
(recursive-self-improvement/config/prompt.md) still owns proposal-writing
judgment — this script only makes its Step 2 input cheap.

Usage:
  python3 reflection-digest.py [--since-hours 24] [--reflect-root PATH] [--out PATH]
"""
import argparse
import datetime
import json
import os
import re
import sys
import tempfile

DEFAULT_REFLECT_ROOT = "~/.claude/reflections"
DEFAULT_SINCE_HOURS = 24
FIELDS = ("obstacles_to_autonomy", "patterns", "discoveries", "turning_points", "digressions")
FIELD_TITLES = {
    "obstacles_to_autonomy": "Obstacles to autonomy",
    "patterns": "Patterns",
    "discoveries": "Discoveries",
    "turning_points": "Turning points",
    "digressions": "Digressions",
}


def normalize(text):
    """Collapse whitespace, drop punctuation, lowercase — for near-dup grouping."""
    text = text.lower()
    text = re.sub(r"[^\w\s]", " ", text)
    return re.sub(r"\s+", " ", text).strip()


def _mtime_or_none(path):
    try:
        return os.path.getmtime(path)
    except OSError:
        return None


def _latest_checkpoint(sdir, since_epoch):
    cp_dir = os.path.join(sdir, "checkpoints")
    if not os.path.isdir(cp_dir):
        return None
    candidates = []
    for f in os.listdir(cp_dir):
        if not f.endswith(".json"):
            continue
        path = os.path.join(cp_dir, f)
        mtime = _mtime_or_none(path)
        if mtime is not None and mtime >= since_epoch:
            candidates.append((mtime, path))
    if not candidates:
        return None
    return max(candidates)[1]


def find_artifacts(reflect_root, since_epoch):
    """Yield (session_id, path, is_subagent, parent_sid) for every final.json
    (or, absent that, the newest checkpoint) modified since `since_epoch`."""
    sessions_dir = os.path.join(reflect_root, "sessions")
    if not os.path.isdir(sessions_dir):
        return
    for sid in sorted(os.listdir(sessions_dir)):
        sdir = os.path.join(sessions_dir, sid)
        if not os.path.isdir(sdir) or sid.startswith("_"):
            continue
        final_path = os.path.join(sdir, "final.json")
        final_mtime = _mtime_or_none(final_path)
        if final_mtime is not None and final_mtime >= since_epoch:
            yield sid, final_path, False, None
        else:
            latest = _latest_checkpoint(sdir, since_epoch)
            if latest:
                yield sid, latest, False, None
        sub_dir = os.path.join(sdir, "subagents")
        if os.path.isdir(sub_dir):
            for sub_sid in sorted(os.listdir(sub_dir)):
                sub_final = os.path.join(sub_dir, sub_sid, "final.json")
                sub_mtime = _mtime_or_none(sub_final)
                if sub_mtime is not None and sub_mtime >= since_epoch:
                    yield sub_sid, sub_final, True, sid


def load_artifact(path):
    try:
        with open(path, encoding="utf-8") as fh:
            data = json.load(fh)
    except (OSError, json.JSONDecodeError):
        return None
    if not isinstance(data, dict):
        return None
    return {k: data.get(k, []) for k in FIELDS if isinstance(data.get(k), list)}


def collect(reflect_root, since_epoch):
    buckets = {f: {} for f in FIELDS}
    session_index = {}
    for sid, path, is_sub, parent in find_artifacts(reflect_root, since_epoch):
        artifact = load_artifact(path)
        if artifact is None:
            continue
        session_index[sid] = {"path": path, "is_subagent": is_sub, "parent": parent}
        for field in FIELDS:
            for entry in artifact.get(field, []):
                if not isinstance(entry, str) or not entry.strip():
                    continue
                key = normalize(entry)
                if not key:
                    continue
                bucket = buckets[field].setdefault(key, {"raw": entry.strip(), "sessions": set()})
                bucket["sessions"].add(sid)
    return buckets, session_index


def render(buckets, session_index, since_hours, generated_at):
    lines = [f"# Reflection digest — {generated_at}", "",
             f"Window: last {since_hours}h. Sessions covered: {len(session_index)}.", ""]

    for field in FIELDS:
        bucket = buckets[field]
        lines.append(f"## {FIELD_TITLES[field]}")
        if not bucket:
            lines.append("(none)")
        else:
            ordered = sorted(bucket.items(), key=lambda kv: (-len(kv[1]["sessions"]), kv[1]["raw"]))
            for _key, b in ordered:
                cite = ", ".join(sorted(b["sessions"]))
                lines.append(f"- ({len(b['sessions'])}x) {b['raw']} — sessions: {cite}")
        lines.append("")

    lines.append("## Session artifact index")
    if not session_index:
        lines.append("(none)")
    else:
        for sid in sorted(session_index):
            info = session_index[sid]
            tag = f" (subagent of {info['parent']})" if info["is_subagent"] else ""
            lines.append(f"- {sid}{tag}: {info['path']}")
    lines.append("")
    return "\n".join(lines)


def atomic_write(path, text):
    d = os.path.dirname(path)
    os.makedirs(d, exist_ok=True)
    fd, tmp = tempfile.mkstemp(dir=d, prefix=".digest-tmp-")
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as fh:
            fh.write(text)
        os.replace(tmp, path)
    except BaseException:
        try:
            os.unlink(tmp)
        except OSError:
            pass
        raise


def main(argv=None):
    parser = argparse.ArgumentParser()
    parser.add_argument("--since-hours", type=float, default=DEFAULT_SINCE_HOURS)
    parser.add_argument("--reflect-root", default=os.environ.get("REFLECT_ROOT", DEFAULT_REFLECT_ROOT))
    parser.add_argument("--out", default=None)
    args = parser.parse_args(argv)

    reflect_root = os.path.expanduser(args.reflect_root)
    now = datetime.datetime.now(datetime.timezone.utc)
    since_epoch = now.timestamp() - args.since_hours * 3600
    generated_at = now.strftime("%Y-%m-%d")

    out_path = os.path.expanduser(args.out) if args.out else os.path.join(
        reflect_root, "digests", f"{generated_at}.md")

    buckets, session_index = collect(reflect_root, since_epoch)
    text = render(buckets, session_index, args.since_hours, generated_at)
    atomic_write(out_path, text)
    print(f"reflection-digest: wrote {out_path} ({len(session_index)} session(s))")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
