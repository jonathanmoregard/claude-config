#!/usr/bin/env python3
"""Shared context-fill-percentage calculation.

Extracted from context-baggage.py (2026-08-17) so the 70%-interactive-nudge
and the 30%-autonomous-self-compact trigger read the same window-detection
logic instead of two copies that can silently drift out of sync.
"""
from __future__ import annotations

import json
import os
import sys

BASE_MAX = 200_000
ONE_M = 1_000_000

ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)

# Explicit 1M marker in the model ID, e.g. `claude-opus-5[1m]`. Checked
# before the prefix list so it keeps working across model generations
# without another edit here.
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model: str | None, observed_max: int) -> int:
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def compute_fill(transcript_path: str) -> dict:
    """Returns {fill_pct, max_tokens, model, total_tokens, user_msg_count}
    or {error: "..."} if the transcript can't be read."""
    if not transcript_path or not os.path.exists(transcript_path):
        return {"error": "transcript_not_found", "transcript_path": transcript_path}

    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0

    try:
        with open(transcript_path, "r", errors="replace") as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                try:
                    obj = json.loads(line)
                except json.JSONDecodeError:
                    continue

                obj_type = obj.get("type")
                if obj_type == "user":
                    user_msg_count += 1
                elif obj_type == "assistant":
                    msg = obj.get("message", {})
                    model = msg.get("model")
                    if model and model != "<synthetic>":
                        last_model = model
                    usage = msg.get("usage")
                    if usage:
                        last_usage = usage
                        total = (
                            usage.get("input_tokens", 0)
                            + usage.get("cache_creation_input_tokens", 0)
                            + usage.get("cache_read_input_tokens", 0)
                        )
                        if total > observed_max:
                            observed_max = total
    except OSError as e:
        return {"error": f"read_failed: {e}", "transcript_path": transcript_path}

    max_tokens = detect_max_tokens(last_model, observed_max)

    if last_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )
        fill_pct = total_tokens / max_tokens
    else:
        total_tokens = None
        file_size = os.path.getsize(transcript_path)
        fill_pct = file_size / (max_tokens * 4)

    return {
        "fill_pct": fill_pct,
        "max_tokens": max_tokens,
        "model": last_model,
        "total_tokens": total_tokens,
        "user_msg_count": user_msg_count,
    }


def main():
    if len(sys.argv) != 2:
        print(json.dumps({"error": "usage: context_fill.py <transcript_path>"}))
        sys.exit(1)
    result = compute_fill(sys.argv[1])
    print(json.dumps(result))
    sys.exit(1 if "error" in result else 0)


if __name__ == "__main__":
    main()
EOF

#!/usr/bin/env bash
# self-compact-monitor.sh — systemd-timer-driven watcher for ralph-loop-armed
# sessions. When a session's context fill crosses its configured threshold,
# writes a fresh tasks/mission.md, gracefully stops the process, and
# relaunches it in the same worktree so compact-reinject.sh's existing
# SessionStart(startup) doctrine re-injection picks the mission back up.
#
# Usage: self-compact-monitor.sh              (run the monitor sweep)
#        self-compact-monitor.sh --source-only (load functions only; used by tests)
set -euo pipefail
umask 077

if [[ "$(uname -s)" != "Linux" ]]; then
  echo "self-compact-monitor: Linux only (uses GNU date/stat)" >&2
  exit 2
fi

: "${SELF_COMPACT_ROOT:=$HOME/.claude/self-compact}"
: "${SELF_COMPACT_LOG_DIR:=$SELF_COMPACT_ROOT/logs}"
: "${SELF_COMPACT_DISABLE_FLAG:=$HOME/.claude/self-compact.disable}"
: "${REFLECT_SH:=$HOME/.claude/skills/session-reflect/reflect.sh}"
: "${CONTEXT_FILL_PY:=$HOME/.claude/scripts/context_fill.py}"
: "${SELF_COMPACT_KILL_WAIT_SECONDS:=20}"
PROJECTS_DIR="${PROJECTS_DIR:-$HOME/.claude/projects}"

log() {
  mkdir -p "$SELF_COMPACT_LOG_DIR" 2>/dev/null || true
  printf '%s %s\n' "$(date -u +%Y-%m-%dT%H:%M:%SZ)" "$*" >> "$SELF_COMPACT_LOG_DIR/monitor.log" 2>/dev/null || true
}

# --- State-file parsing ------------------------------------------------------
# parse_self_compact_state <path>
# Emits eval-safe key=value lines for the YAML-ish frontmatter (between the
# two `---` lines). Values are NOT validated here (that's the caller's job,
# via fill_meets_threshold's own numeric check) — this function's only
# responsibility is extraction, mirroring session-reflect/lib.sh's
# paths_for_session pattern (parse now, validate at point of use).
parse_self_compact_state() {
  local path="$1"
  local enabled threshold cooldown
  enabled=$(awk -F': *' '/^enabled:/{print $2; exit}' "$path")
  threshold=$(awk -F': *' '/^threshold:/{print $2; exit}' "$path")
  cooldown=$(awk -F': *' '/^cooldown_seconds:/{print $2; exit}' "$path")
  printf 'sc_enabled=%q\n' "${enabled:-}"
  printf 'sc_threshold=%q\n' "${threshold:-}"
  printf 'sc_cooldown_seconds=%q\n' "${cooldown:-}"
}

# sc_launch_cmd <path> — prints the body (everything after the second `---`
# line) verbatim. The launch command is stored as free text, not YAML, so it
# never needs quote-escaping for shell metacharacters — same rationale as
# ralph-loop.local.md storing its mission prompt as body text rather than a
# frontmatter field.
sc_launch_cmd() {
  local path="$1"
  awk 'BEGIN{n=0} /^---$/{n++; next} n>=2{print}' "$path"
}

# fill_meets_threshold <fill_pct> <threshold> — numeric compare via awk
# (bash has no float arithmetic). Refuses (exit 2, not just false) on a
# non-numeric threshold so a corrupted state file can't silently be treated
# as "never trigger" or "always trigger" — the caller must see the failure.
fill_meets_threshold() {
  local fill="$1" threshold="$2"
  if ! [[ "$threshold" =~ ^[0-9]+(\.[0-9]+)?$ ]]; then
    return 2
  fi
  awk -v f="$fill" -v t="$threshold" 'BEGIN{exit !(f >= t)}'
}

# cooldown_active <marker_path> <cooldown_seconds> — true if the marker file
# exists and is younger than cooldown_seconds.
cooldown_active() {
  local marker="$1" cooldown="$2"
  [[ -f "$marker" ]] || return 1
  local mtime now age
  mtime=$(stat -c '%Y' "$marker" 2>/dev/null || echo 0)
  now=$(date +%s)
  age=$(( now - mtime ))
  (( age < cooldown ))
}

# ============================================================================
# MAIN — only runs when invoked directly, not when sourced for tests.
# ============================================================================
if [[ "${1:-}" == "--source-only" ]]; then
  return 0 2>/dev/null || exit 0
fi

main() {
  log "sweep start"
  # Full sweep pipeline implemented in Task 6.
  :
}

if [[ "${BASH_SOURCE[0]}" == "${0}" ]]; then
  main "$@"
fi
EOF
chmod +x skills/self-compact/hooks/self-compact-monitor.sh
```

```bash
main() {
  log "sweep start"

  if [[ -f "$SELF_COMPACT_DISABLE_FLAG" ]]; then
    log "disabled (flag file exists)"
    return 0
  fi
  if [[ ! -x "$REFLECT_SH" ]]; then
    log "reflect.sh not found at $REFLECT_SH"
    return 0
  fi
  if [[ ! -d "$PROJECTS_DIR" ]]; then
    log "no $PROJECTS_DIR"
    return 0
  fi

  exec 8>>"$SELF_COMPACT_ROOT/monitor.lock" 2>/dev/null || { log "cannot open lock"; return 0; }
  mkdir -p "$SELF_COMPACT_ROOT" 2>/dev/null || true
  if ! flock -n 8; then
    log "another monitor sweep in progress, skipping"
    return 0
  fi

  # LIVE map: sid -> pid, for `claude --resume <sid>` processes.
  declare -A LIVE_PID
  while read -r pid sid; do
    [[ -n "$sid" ]] && LIVE_PID["$sid"]="$pid"
  done < <(pgrep -af 'claude --resume' 2>/dev/null | \
    awk '{pid=$1; for(i=1;i<=NF;i++){if($i ~ /^[a-f0-9-]{36}$/){print pid, $i; break}}}')

  for project_dir in "$PROJECTS_DIR"/*/; do
    [[ -d "$project_dir" ]] || continue
    case "$project_dir" in
      *homunculus*|*observer*|*step-back*) continue ;;
    esac

    transcript=$(find "$project_dir" -maxdepth 1 -type f -name '*.jsonl' \
      -printf '%T@ %p\n' 2>/dev/null | sort -rn | head -1 | awk '{print $2}')
    [[ -n "$transcript" && -f "$transcript" ]] || continue

    sid=$(basename "$transcript" .jsonl)
    pid="${LIVE_PID[$sid]:-}"
    [[ -n "$pid" ]] || continue   # not live -> nothing actively burning context

    cwd=$(tac "$transcript" 2>/dev/null | jq -r 'select(type=="object") | .cwd // empty' 2>/dev/null | head -1)
    [[ -n "$cwd" && -d "$cwd" ]] || continue

    state="$cwd/.claude/self-compact.local"
    [[ -f "$state" ]] || continue
    [[ -f "$cwd/.claude/ralph-loop.local.md" ]] || { log "sid=$sid skip: self-compact.local present but ralph-loop not armed in $cwd"; continue; }

    eval "$(parse_self_compact_state "$state")"
    [[ "$sc_enabled" == "true" ]] || continue

    cooldown_marker="$SELF_COMPACT_ROOT/cooldowns/$(echo -n "$cwd" | sha256sum | cut -d' ' -f1)"
    mkdir -p "$(dirname -- "$cooldown_marker")" 2>/dev/null || true
    if cooldown_active "$cooldown_marker" "${sc_cooldown_seconds:-600}"; then
      continue
    fi

    fill_json=$(python3 "$CONTEXT_FILL_PY" "$transcript" 2>/dev/null) || { log "sid=$sid context_fill.py failed"; continue; }
    fill_pct=$(jq -r '.fill_pct // empty' <<<"$fill_json" 2>/dev/null)
    [[ -n "$fill_pct" ]] || { log "sid=$sid no fill_pct in $fill_json"; continue; }

    if ! fill_meets_threshold "$fill_pct" "${sc_threshold:-}"; then
      rc=$?
      if (( rc == 2 )); then
        log "sid=$sid REFUSING: non-numeric threshold '$sc_threshold' in $state"
      fi
      continue
    fi

    acted_marker="$SELF_COMPACT_ROOT/acted/$sid"
    mkdir -p "$(dirname -- "$acted_marker")" 2>/dev/null || true
    if [[ -f "$acted_marker" ]]; then
      continue   # already handling this sid
    fi
    : > "$acted_marker"   # write BEFORE acting: a crash mid-action must not hot-loop

    log "TRIGGER sid=$sid cwd=$cwd fill=${fill_pct} pid=$pid"

    # 1. Synchronous mission write while cache is warm. Never proceed to kill
    #    the session if this fails — do not destroy state we couldn't hand off.
    mission_payload=$(jq -cn --arg sid "$sid" --arg tp "$transcript" --arg cwd "$cwd" \
      '{session_id:$sid, transcript_path:$tp, hook_event_name:"SelfCompact", cwd:$cwd}')
    if ! printf '%s' "$mission_payload" | env -u MODE timeout 90 bash "$REFLECT_SH" mission \
         >>"$SELF_COMPACT_LOG_DIR/${sid}.mission.log" 2>&1; then
      log "sid=$sid ABORT: mission write failed, not killing session (see ${sid}.mission.log)"
      continue
    fi

    # 2. Graceful stop. SIGTERM, not SIGHUP: the target process was very
    #    likely started under nohup/setsid for exactly the unattended-
    #    longevity reason self-compact assumes, and nohup makes a process
    #    IGNORE SIGHUP by default — SIGTERM is the signal nohup does not
    #    suppress. (session-reflect's own docs assume SIGHUP works because
    #    they're describing a foreground kitty-pane close, a different
    #    scenario from a detached autonomous run.)
    kill -TERM "$pid" 2>/dev/null || true
    waited=0
    while kill -0 "$pid" 2>/dev/null; do
      sleep 1
      waited=$(( waited + 1 ))
      if (( waited >= SELF_COMPACT_KILL_WAIT_SECONDS )); then
        log "sid=$sid ABORT: pid=$pid still alive after ${SELF_COMPACT_KILL_WAIT_SECONDS}s, not relaunching"
        continue 2
      fi
    done
    log "sid=$sid pid=$pid exited after ${waited}s"

    # 3. Relaunch. Detach so the monitor's own exit doesn't kill the child.
    launch_cmd=$(sc_launch_cmd "$state")
    if [[ -z "$launch_cmd" ]]; then
      log "sid=$sid ABORT: empty launch_cmd in $state, not relaunching (mission.md is on disk for manual resume)"
      continue
    fi
    ( cd "$cwd" && setsid env SELF_COMPACT_RELAUNCH=1 nohup bash -c "$launch_cmd" \
        </dev/null >>"$SELF_COMPACT_LOG_DIR/${sid}.relaunch.log" 2>&1 & disown )
    touch "$cooldown_marker"
    log "sid=$sid relaunched in $cwd"
  done

  log "sweep end"
}

cat > skills/self-compact/README.md <<'EOF'
# self-compact

Opt-in, systemd-timer-driven companion to `ralph-loop`/`orchestrate`'s
autonomous runs. Watches context fill on ralph-loop-armed sessions and, at a
configurable threshold (default 30% of a 1M-token window — well below
Claude Code's own auto-compact point and below the 70% interactive nudge in
`context-baggage.py`), autonomously:

1. writes a fresh `tasks/mission.md` (via `session-reflect`'s `reflect.sh
   mission` mode, synchronously, while the API prompt cache is still warm),
2. gracefully stops the session process (`SIGTERM`, bounded wait),
3. relaunches it in the same worktree via the operator-supplied `launch_cmd`.

The relaunch is deliberately just a normal process start in that directory —
`skills/orchestrate/hooks/compact-reinject.sh`'s existing
`SessionStart(startup)` hook already detects the still-armed
`.claude/ralph-loop.local.md` and re-injects the autonomy doctrine plus a
pointer to `tasks/mission.md`. self-compact does not duplicate any of that;
it only decides *when* to trigger a fresh start.

## Why SIGTERM, not SIGHUP

`session-reflect`'s docs note `SessionEnd` fires reliably on `SIGHUP`
because they're describing a kitty pane closing. An unattended autonomous
run is very likely already running under `nohup`/`setsid` for its own
longevity — and `nohup` specifically makes a process **ignore** `SIGHUP`.
self-compact sends `SIGTERM` instead, which `nohup` does not suppress.

## Requirements

- The target worktree must be ralph-loop-armed (`.claude/ralph-loop.local.md`
  present) — self-compact refuses to activate otherwise, even if
  `.claude/self-compact.local` exists. There is no doctrine to resume an
  unarmed session with, and killing an interactive session nobody
  re-arms is a straight regression.
- `.claude/self-compact.local` must be written from a dedicated worktree,
  never `$HOME`, `$HOME/.claude`, or `$HOME/Repos` — same rule the
  `orchestrate` skill's ralph-loop arm block already enforces, for the same
  reason (blast radius).

## `.claude/self-compact.local` format

```
---
enabled: true
threshold: 0.30
cooldown_seconds: 600
armed_at: "2026-08-17T12:00:00Z"
---
claude -p "Autonomous mission: read tasks/mission.md, tasks/session-constraints.md, and tasks/progress.md, then continue." --dangerously-skip-permissions
```

The frontmatter is parsed field-by-field with `awk` (same style as
`ralph-loop.local.md`); the **body** (everything after the second `---`) is
the literal shell command used to relaunch — stored as free text, not a YAML
field, so it never needs quote-escaping.

**The launch command is the operator's responsibility.** It runs completely
unattended, in a brand-new process, with nobody present to answer a
permission prompt. If it hangs on one, self-compact's own kill-wait window
does not apply to it (that window only covers stopping the *old* process) —
a launch_cmd that isn't actually non-interactive will just sit there until
someone notices. Test the exact command manually before arming self-compact
with it.

## Kill-switches

- Global pause, no un-arming needed: `touch ~/.claude/self-compact.disable`
  (remove the file to resume — same convention as `idle-handoff.disable`).
- Per-worktree cancel: `rm .claude/self-compact.local` (self-compact simply
  stops seeing the marker on the next sweep; does not affect the ralph loop
  itself, which needs its own `rm .claude/ralph-loop.local.md` /
  `/cancel-ralph` to stop).

## State on disk

```
~/.claude/self-compact/
  monitor.lock              # sweep concurrency guard
  logs/monitor.log          # one line per sweep decision
  logs/<sid>.mission.log    # reflect.sh mission stdout/stderr for that trigger
  logs/<sid>.relaunch.log   # relaunched process's stdout/stderr
  acted/<sid>               # idempotency marker — one trigger per session id
  cooldowns/<sha256(cwd)>   # cooldown marker, mtime = last relaunch time for that worktree
```

`acted/<sid>` is keyed by session id, which changes on every relaunch, so
it naturally never blocks a later trigger for the *new* session — it only
prevents re-triggering the *same* session repeatedly within one sweep loop
or across ticks before it's actually stopped. `cooldowns/` is keyed by
worktree path instead, because that survives across session ids and is
what actually prevents a relaunch-storm in one worktree.
