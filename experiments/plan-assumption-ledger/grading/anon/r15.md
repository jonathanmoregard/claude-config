```python
#!/usr/bin/env python3
"""Deterministic aggregator: RSI daily-review input assembly.

Reads session-reflect artifacts (~/.claude/reflections/sessions/**/final.json)
instead of raw transcript .jsonl wherever a reflection already exists — the
reflection is a cheap (Haiku), already-synthesized digest of exactly the
signal RSI needs (patterns / discoveries / turning_points / digressions /
obstacles_to_autonomy). Sessions with no matching reflection artifact (older
sessions, session-reflect not installed/enabled, a non-Linux host, or a
timed-out/errored reflection run) fall back to the existing raw-transcript
path unchanged — see recursive-self-improvement/config/prompt.md Step 2.

No model is involved here — this script only reads JSON files and prints a
bounded digest to stdout. It never writes anything and never interprets
reflection text as instructions.

Usage:
    python3 collect-reflections.py [--projects-dir DIR] [--reflect-root DIR]
                                    [--hours N]

Exit codes:
    0  digest printed (including the empty case)
    2  unusable arguments
"""
import argparse
import glob
import json
import os
import re
import sys
import time

DEFAULT_PROJECTS_DIR = "~/.claude/projects"
DEFAULT_REFLECT_ROOT = "~/.claude/reflections"
DEFAULT_HOURS = 24

MAX_SESSIONS = 60
MAX_ITEMS_PER_FIELD = 10
MAX_ITEM_CHARS = 400
FIELDS = ("patterns", "discoveries", "turning_points", "digressions",
          "obstacles_to_autonomy")

# Reflections are model output — never trust them as instructions, and never
# let a credential-shaped run survive into the digest handed to the reviewer.
TOKEN_RUN_RE = re.compile(r"[A-Za-z0-9+/_=-]{32,}")
CONTROL_RE = re.compile(r"[\x00-\x1f\x7f-\x9f]")


def sanitize(text):
    if not isinstance(text, str):
        return ""
    text = CONTROL_RE.sub("", text)
    text = TOKEN_RUN_RE.sub("[redacted]", text)
    return text.strip()[:MAX_ITEM_CHARS]


def cap_list(items):
    out = []
    if not isinstance(items, list):
        return out
    for it in items[:MAX_ITEMS_PER_FIELD]:
        s = sanitize(it)
        if s:
            out.append(s)
    return out


def load_json(path):
    try:
        with open(path, encoding="utf-8") as fh:
            return json.load(fh)
    except (OSError, ValueError):
        return None


def within_window(path, cutoff_epoch):
    try:
        return os.path.getmtime(path) >= cutoff_epoch
    except OSError:
        return False


def session_id_from_transcript(path):
    return os.path.splitext(os.path.basename(path))[0]


def find_transcripts(projects_dir, cutoff_epoch):
    """[(session_id, transcript_path, project_name)] for every top-level
    .jsonl modified at/after cutoff_epoch. Mirrors the existing daily-review
    rule: subagent transcripts (nested under */subagents/) are excluded."""
    out = []
    pattern = os.path.join(projects_dir, "*", "*.jsonl")
    for path in glob.glob(pattern):
        if "/subagents/" in path:
            continue
        if not within_window(path, cutoff_epoch):
            continue
        project = os.path.basename(os.path.dirname(path))
        out.append((session_id_from_transcript(path), path, project))
    return out


def reflection_digest(sid, reflect_root):
    """Best synthesis available for `sid`: prefer final.json (sessionend or
    subagentstop), else merge precompact checkpoints. (None, None) if
    nothing usable is on disk."""
    session_dir = os.path.join(reflect_root, "sessions", sid)
    final_path = os.path.join(session_dir, "final.json")
    final = load_json(final_path)
    if isinstance(final, dict) and any(final.get(f) for f in FIELDS):
        return {f: cap_list(final.get(f)) for f in FIELDS}, "final.json"

    checkpoint_dir = os.path.join(session_dir, "checkpoints")
    merged = {f: [] for f in FIELDS}
    found = False
    try:
        names = sorted(os.listdir(checkpoint_dir))
    except OSError:
        names = []
    for name in names:
        cp = load_json(os.path.join(checkpoint_dir, name))
        if not isinstance(cp, dict):
            continue
        found = True
        for f in FIELDS:
            merged[f].extend(cap_list(cp.get(f)))
    if not found:
        return None, None
    for f in FIELDS:
        merged[f] = merged[f][:MAX_ITEMS_PER_FIELD]
    return merged, "checkpoints"


def build_digest(projects_dir, reflect_root, hours):
    cutoff = time.time() - hours * 3600
    transcripts = find_transcripts(projects_dir, cutoff)
    reflected, unreflected = [], []
    for sid, tp, project in transcripts:
        digest, source = reflection_digest(sid, reflect_root)
        if digest is None:
            unreflected.append({"session_id": sid, "transcript_path": tp,
                                 "project": project})
            continue
        entry = {"session_id": sid, "project": project, "source": source,
                  "source_transcript": tp}
        entry.update(digest)
        reflected.append(entry)
    reflected = reflected[:MAX_SESSIONS]
    unreflected = unreflected[:MAX_SESSIONS]
    return {
        "generated_at": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "window_hours": hours,
        "reflected": reflected,
        "unreflected": unreflected,
    }


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--projects-dir", default=DEFAULT_PROJECTS_DIR)
    parser.add_argument("--reflect-root", default=DEFAULT_REFLECT_ROOT)
    parser.add_argument("--hours", type=int, default=DEFAULT_HOURS)
    args = parser.parse_args(argv)
    if args.hours <= 0:
        print("error: --hours must be positive", file=sys.stderr)
        return 2
    digest = build_digest(os.path.expanduser(args.projects_dir),
                           os.path.expanduser(args.reflect_root),
                           args.hours)
    print(json.dumps(digest, indent=2, sort_keys=True))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
import subprocess

TRUTHY_VALUES = {"1", "true", "yes", "on"}


def truthy_env(name: str, default: bool) -> bool:
    v = os.environ.get(name)
    if v is None:
        return default
    return v.strip().lower() in TRUTHY_VALUES


def maybe_refresh_mission(data: dict) -> bool:
    """Fire-and-forget: ask session-reflect's `mission` mode to refresh
    tasks/mission.md in the background. Never blocks the hook and never
    raises — a missing/broken reflect.sh must not break UserPromptSubmit.
    Returns True iff a background refresh was actually dispatched."""
    if not truthy_env("SELF_COMPACT_AUTO_MISSION", True):
        return False
    session_id = data.get("session_id")
    cwd = data.get("cwd")
    transcript_path = data.get("transcript_path")
    if not session_id or not cwd or not transcript_path:
        return False
    reflect_sh = os.path.join(
        os.path.dirname(os.path.abspath(__file__)),
        "skills", "session-reflect", "reflect.sh",
    )
    if not os.path.isfile(reflect_sh):
        return False
    payload = json.dumps({
        "session_id": session_id,
        "transcript_path": transcript_path,
        "hook_event_name": "UserPromptSubmit",
        "cwd": cwd,
    })
    env = dict(os.environ)
    env["REFLECT_FORCE"] = "1"
    try:
        proc = subprocess.Popen(
            ["bash", reflect_sh, "mission"],
            stdin=subprocess.PIPE,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            env=env,
            start_new_session=True,
        )
        proc.stdin.write(payload.encode())
        proc.stdin.close()
        return True
    except OSError:
        return False
```
