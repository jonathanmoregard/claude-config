```python
#!/usr/bin/env python3
"""Shared context-window fill detection for Claude Code transcripts.

Single source of truth for "how full is this session's context window."
Used by context-baggage.py (UserPromptSubmit hook, advisory nudge at a high
threshold) and skills/orchestrate/self-compact-loop.sh (external autonomous
driver, acts at a low threshold). Detection priority order:
    1. CLAUDE_CONTEXT_MAX env var (explicit override, in tokens)
    2. Model ID prefix — Opus/Sonnet 4.x+ run with 1M in this setup
    3. Observed usage — if any assistant turn exceeded 200k tokens, the
       session is 1M regardless of model name
    4. Default 200k
"""
import json
import os
import sys

BASE_MAX = 200_000
ONE_M = 1_000_000

# Claude Code model IDs that run at 1M context by default. Prefixes, so
# dated variants (claude-opus-4-1-20250805) match too.
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)

# Explicit 1M marker in the model ID, e.g. `claude-opus-5[1m]`. Checked
# before the prefix list so it keeps working across model generations
# without another edit here.
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model: str | None, observed_max: int) -> int:
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def compute_fill(transcript_path: str) -> dict:
    """Scan a transcript JSONL and return fill stats.

    Never raises — a missing/unreadable transcript yields a zeroed result
    so callers (hooks, loops) can treat it as "not full" rather than crash.
    """
    if not transcript_path or not os.path.exists(transcript_path):
        return {"fill_pct": 0.0, "total_tokens": 0, "max_tokens": BASE_MAX, "model": None, "file_size": 0}

    file_size = os.path.getsize(transcript_path)
    last_usage = None
    last_model = None
    observed_max = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue
            if obj.get("type") != "assistant":
                continue
            msg = obj.get("message", {})
            model = msg.get("model")
            if model and model != "<synthetic>":
                last_model = model
            usage = msg.get("usage")
            if usage:
                last_usage = usage
                total = (
                    usage.get("input_tokens", 0)
                    + usage.get("cache_creation_input_tokens", 0)
                    + usage.get("cache_read_input_tokens", 0)
                )
                if total > observed_max:
                    observed_max = total

    max_tokens = detect_max_tokens(last_model, observed_max)

    if last_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )
        fill_pct = total_tokens / max_tokens
    else:
        total_tokens = 0
        fill_pct = file_size / (max_tokens * 4)

    return {
        "fill_pct": fill_pct,
        "total_tokens": total_tokens,
        "max_tokens": max_tokens,
        "model": last_model,
        "file_size": file_size,
    }


def main():
    import argparse
    parser = argparse.ArgumentParser(description="Report context-window fill for a Claude Code transcript.")
    parser.add_argument("--transcript", required=True)
    args = parser.parse_args()
    print(json.dumps(compute_fill(args.transcript)))


if __name__ == "__main__":
    main()
```

#!/usr/bin/env python3
"""
UserPromptSubmit hook: warns when context is over THRESHOLD with potentially
stale/off-topic content. Fires every MESSAGE_INTERVAL user messages once
threshold is met.

Context-window fill detection lives in context_fill.py (shared with
skills/orchestrate/self-compact-loop.sh) — see that file's docstring for the
detection priority order. This file reads user_msg_count separately from
compute_fill's own scan (two passes over the transcript instead of one) —
a deliberate simplicity-over-micro-perf tradeoff; transcripts are small
relative to this hook's budget and this only runs every 5 messages.
"""
import json
import os
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from context_fill import compute_fill  # noqa: E402

# Raised from 0.25 -> 0.70 on 2026-08-08 after measuring 95.5% cache-hit
# ratio across 7 days / $3,073 spend (probes/analyze_sessions.py). At that
# hit ratio, holding baggage is cheap and resetting is expensive
# (cache_write ~$5-10 per cold event). Nudge should fire when handoff to
# a fresh session starts to pay off, not on any drift.
THRESHOLD = 0.70
MESSAGE_INTERVAL = 5


def main():
    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    transcript_path = data.get("transcript_path", "")
    if not transcript_path or not os.path.exists(transcript_path):
        sys.exit(0)

    user_msg_count = 0
    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue
            if obj.get("type") == "user":
                user_msg_count += 1

    if user_msg_count == 0 or user_msg_count % MESSAGE_INTERVAL != 0:
        sys.exit(0)

    fill = compute_fill(transcript_path)
    fill_pct = fill["fill_pct"]
    max_tokens = fill["max_tokens"]

    if fill_pct < THRESHOLD:
        sys.exit(0)

    pct_display = int(fill_pct * 100)
    message = (
        f"Context is {pct_display}% full ({user_msg_count} messages in this session; "
        f"window: {max_tokens // 1000}k tokens). "
        "At this fill, a fresh-session handoff via tasks/mission.md is usually cheaper "
        "than an in-place /compact: --fork-session on a warm cache measured ~77% hit "
        "(session-reflect empirical), while /compact invalidates the prefix and forces "
        "cache_write on subsequent turns. "
        "Suggest to the user: (a) write tasks/mission.md (done / next / blocked / open "
        "questions), (b) start a fresh session in the same cwd and load it, OR (c) "
        "continue here and accept the drift. Ask which."
    )
    print(json.dumps({"hookSpecificOutput": {"hookEventName": "UserPromptSubmit", "additionalContext": message}}))


if __name__ == "__main__":
    main()

#!/usr/bin/env bash
# self-compact-loop.sh — headless autonomous driver with context-aware reset.
#
# The official ralph-loop plugin re-feeds the SAME session/context every
# iteration via a Stop-hook block; that session's context only shrinks via
# Claude Code's own lossy /compact, which fires late (~90%+ full) and
# invalidates the prompt cache. This script instead runs each iteration as
# its own `claude -p` process and, when context fill crosses THRESHOLD
# (default 0.30 of the detected window), writes tasks/mission.md via
# session-reflect's mission mode and starts the NEXT iteration as a BRAND
# NEW session (no --resume) seeded from mission.md — a hard, deterministic
# reset instead of a lossy in-place compact. Every iteration still ends as
# its own OS process, so the existing SessionEnd/PreCompact hooks
# (session-reflect) fire normally and keep producing reflection artifacts.
#
# Usage:
#   self-compact-loop.sh --mission "<task description>" --sigil "<TEXT>" \
#     [--max-iterations N] [--threshold 0.30] [--model MODEL]
#
# Requires: run from a dedicated worktree root. This script arms unattended
# --dangerously-skip-permissions execution; blast radius must be a
# throwaway worktree, never $HOME/~/.claude/~/Repos (same guard as
# skills/orchestrate/SKILL.md Phase 0, copied verbatim below).
set -euo pipefail
umask 077

if [[ "$(uname -s)" != "Linux" ]]; then
  echo "self-compact-loop: Linux only (uses GNU date conventions shared with session-reflect)" >&2
  exit 2
fi

REFLECT_SH="${SELF_COMPACT_REFLECT_SH:-$HOME/.claude/skills/session-reflect/reflect.sh}"
CONTEXT_FILL_PY="${SELF_COMPACT_CONTEXT_FILL_PY:-$HOME/.claude/context_fill.py}"

MISSION_PROMPT=""
SIGIL=""
MAX_ITERATIONS=25
THRESHOLD=0.30
MODEL="${SELF_COMPACT_MODEL:-sonnet}"

while [[ $# -gt 0 ]]; do
  case "$1" in
    --mission) MISSION_PROMPT="$2"; shift 2 ;;
    --sigil) SIGIL="$2"; shift 2 ;;
    --max-iterations) MAX_ITERATIONS="$2"; shift 2 ;;
    --threshold) THRESHOLD="$2"; shift 2 ;;
    --model) MODEL="$2"; shift 2 ;;
    *) echo "self-compact-loop: unknown arg '$1'" >&2; exit 2 ;;
  esac
done

[[ -n "$MISSION_PROMPT" ]] || { echo "REFUSE: --mission is required"; exit 1; }
[[ -n "$SIGIL" ]] || { echo "REFUSE: --sigil is required"; exit 1; }
case "$SIGIL" in *'"'*|*'$'*|*'`'*|*'\'*)

#!/usr/bin/env bash
# Verifies the find command in recursive-self-improvement/config/prompt.md
# Step 2 enumerates exactly the right reflection artifacts: recent
# final.json + checkpoint files, INCLUDING files whose sibling manifest
# says error/timeout (the agent filters those via manifest inspection, not
# via this find command — this test only proves the LISTING is correct;
# Task 5 Step 1's Step 3 instruction is what tells the agent to skip them).
set -u
set -o pipefail

fail() { echo "FAIL: $*" >&2; exit 1; }
info() { echo "  -> $*"; }

ROOT="$(mktemp -d -t rsi-reflect-listing-XXXXXX)"
trap 'rm -rf "$ROOT"' EXIT

mkdir -p "$ROOT/sessions/sid-recent-complete"
echo '{"patterns":["x"]}' > "$ROOT/sessions/sid-recent-complete/final.json"

mkdir -p "$ROOT/sessions/sid-recent-error"
echo '{"patterns":[]}' > "$ROOT/sessions/sid-recent-error/final.json"
echo '{"modes":{"sessionend":{"status":"error"}}}' > "$ROOT/sessions/sid-recent-error/manifest.json"

mkdir -p "$ROOT/sessions/sid-recent-precompact-only/checkpoints"
echo '{"patterns":["y"]}' > "$ROOT/sessions/sid-recent-precompact-only/checkpoints/1234-XXXX-precompact.json"

mkdir -p "$ROOT/sessions/sid-old-complete"
echo '{"patterns":["stale"]}' > "$ROOT/sessions/sid-old-complete/final.json"
touch -d '3 days ago' "$ROOT/sessions/sid-old-complete/final.json"

mkdir -p "$ROOT/sessions/_malformed"
echo '{"reason":"invalid_session_id"}' > "$ROOT/sessions/_malformed/manifest-abc.json"

# The exact command from prompt.md Step 2, parameterized on $ROOT.
FOUND=$(find "$ROOT/sessions" -mindepth 1 -maxdepth 3 \
  \( -name 'final.json' -o -path '*/checkpoints/*.json' \) -mtime -1 | sort)

echo "$FOUND" | grep -q 'sid-recent-complete/final.json' \
  || fail "expected recent complete final.json in listing"
echo "$FOUND" | grep -q 'sid-recent-error/final.json' \
  || fail "expected recent error final.json in listing (agent filters via manifest, not find)"
echo "$FOUND" | grep -q 'sid-recent-precompact-only/checkpoints/1234-XXXX-precompact.json' \
  || fail "expected recent checkpoint-only session in listing"
echo "$FOUND" | grep -q 'sid-old-complete/final.json' \
  && fail "3-day-old final.json should NOT be in a -mtime -1 listing"
echo "$FOUND" | grep -q '_malformed' \
  && fail "_malformed dir has no final.json/checkpoints so should never appear, but did"

info "listing correctly includes recent final.json + checkpoints, excludes stale and _malformed"
echo "OK: reflection-listing test passed"
exit 0
