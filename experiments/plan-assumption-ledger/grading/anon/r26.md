```python
#!/usr/bin/env python3
"""Shared context-window fill calculation for context-baggage.py and
self-compact-mission.py. Single source of truth so the two hooks can never
disagree on THRESHOLD or the model->window-size mapping — the historical bug
this guards against: a stale prefix list once reported 89% full during a
session that was actually at 18% of a 1M window.
"""
import json
import os

BASE_MAX = 200_000
ONE_M = 1_000_000
# Raised from 0.25 -> 0.70 on 2026-08-08 after measuring 95.5% cache-hit
# ratio across 7 days / $3,073 spend (probes/analyze_sessions.py). At that
# hit ratio, holding baggage is cheap and resetting is expensive.
THRESHOLD = 0.70

# Claude Code model IDs that run at 1M context by default. Prefixes, so
# dated variants (claude-opus-4-1-20250805) match too.
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)

# Explicit 1M marker in the model ID, e.g. `claude-opus-5[1m]`. Checked
# before the prefix list so it keeps working across model generations
# without another edit here.
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model, observed_max, override=None):
    """override is the CLAUDE_CONTEXT_MAX env value, pre-read by the caller —
    keeps this module free of os.environ reads so it stays trivially unit
    testable."""
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


def scan_transcript(transcript_path):
    """Single pass over a transcript JSONL. Tolerant of malformed lines
    (skips them, matching the pre-refactor context-baggage.py behavior)."""
    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue

            obj_type = obj.get("type")
            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = (
                        usage.get("input_tokens", 0)
                        + usage.get("cache_creation_input_tokens", 0)
                        + usage.get("cache_read_input_tokens", 0)
                    )
                    if total > observed_max:
                        observed_max = total

    return {
        "user_msg_count": user_msg_count,
        "last_usage": last_usage,
        "last_model": last_model,
        "observed_max": observed_max,
        "file_size": os.path.getsize(transcript_path),
    }


def fill_pct(scan, max_tokens):
    """Fraction of the context window in use. Falls back to a byte-size
    heuristic (~4 chars/token) when no usage data was found in the scan —
    mirrors the pre-refactor context-baggage.py fallback exactly."""
    last_usage = scan["last_usage"]
    if last_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )
        return total_tokens / max_tokens
    return scan["file_size"] / (max_tokens * 4)
```

import os
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))
import context_fill  # noqa: E402

MESSAGE_INTERVAL = 5


def main():
    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    transcript_path = data.get("transcript_path", "")
    if not transcript_path or not os.path.exists(transcript_path):

- [ ] **Step 1: Write `self-compact-mission.py`**

```python
#!/usr/bin/env python3
"""
Stop hook: once context crosses context_fill.THRESHOLD (70% used, ~30%
remaining of a 1M window), proactively refresh tasks/mission.md via
reflect.sh's `mission` mode — BEFORE Claude Code's own auto-compact fires,
not just reactively on the 2nd compact (see reflect.sh's compact_count
defence and
recursive-self-improvement/proposals/2026-08-02-repeat-compact-in-session.md).

Fires on every Stop (unlike context-baggage.py's UserPromptSubmit hook,
which never runs during an unattended ralph-loop turn). Whenever the next
reset actually happens — Claude Code's auto-compact, a manual /compact, or
/clear — skills/orchestrate/hooks/compact-reinject.sh (already wired on
SessionStart) re-arms doctrine and tells Claude to re-read tasks/mission.md,
closing the loop end to end.

Silent by design (scripts/idle-handoff.sh precedent, "Notifications
intentionally omitted... Silent operation"): this hook never emits
chat-visible output, only writes a file. Never blocks the Stop event: any
internal failure exits 0 with no side effect.
"""
import json
import os
import subprocess
import sys
import tempfile
import time
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))
import context_fill  # noqa: E402

REFLECT_SH = os.environ.get(
    "SELF_COMPACT_REFLECT_SH",
    str(Path.home() / ".claude" / "skills" / "session-reflect" / "reflect.sh"),
)
MISSION_MAX_AGE_SECONDS = int(os.environ.get("SELF_COMPACT_MISSION_MAX_AGE_SECONDS", "300"))
DISABLE_FLAG = Path(
    os.environ.get("SELF_COMPACT_DISABLE_FLAG", str(Path.home() / ".claude" / "self-compact.disable"))
)
LOG_DIR = Path(os.environ.get("SELF_COMPACT_LOG_DIR", str(Path.home() / ".claude" / "reflections" / "logs")))

# Automated fire-and-forget transcripts nobody resumes — same exclusion as
# scripts/idle-handoff.sh.
_SKIP_CWD_MARKERS = ("homunculus", "observer", "step-back")


def resolve_cwd(data, transcript_path):
    cwd = data.get("cwd")
    if cwd and os.path.isdir(cwd):
        return cwd
    # Stop payloads don't reliably carry a top-level cwd (step-back-detector.py
    # falls back to scanning transcript entries for the same reason) — walk
    # the transcript backwards for the most recent cwd-bearing entry.
    try:
        with open(transcript_path, "r", errors="replace") as f:
            lines = f.readlines()
    except OSError:
        return None
    for line in reversed(lines):
        line = line.strip()
        if not line:
            continue
        try:
            obj = json.loads(line)
        except json.JSONDecodeError:
            continue
        c = obj.get("cwd")
        if c and os.path.isdir(c):
            return c
    return None


def mission_is_fresh(mission_path):
    try:
        age = time.time() - os.path.getmtime(mission_path)
    except OSError:
        return False
    return age < MISSION_MAX_AGE_SECONDS


def refresh_mission(session_id, transcript_path, cwd):
    """Fire-and-forget: detach reflect.sh mission so this Stop hook returns
    immediately (mirrors reflect.sh's own sessionend detach pattern). The
    stdin payload file is intentionally left for the OS to reap rather than
    raced-deleted from the parent."""
    payload = json.dumps(
        {
            "session_id": session_id,
            "transcript_path": transcript_path,
            "hook_event_name": "SelfCompact",
            "cwd": cwd,
        }
    )
    tmp = tempfile.NamedTemporaryFile(mode="w", suffix=".json", delete=False, prefix="self-compact-payload-")
    tmp.write(payload)
    tmp.close()

    env = dict(os.environ)
    env.pop("MODE", None)  # mirrors idle-handoff.sh: unset MODE so ARG1 wins inside the child

    try:
        LOG_DIR.mkdir(parents=True, exist_ok=True)
        log_path = LOG_DIR / f"{session_id}-selfcompact-{time.time_ns()}.log"
        with open(tmp.name, "rb") as stdin_f, open(log_path, "wb") as log_f:
            subprocess.Popen(
                ["bash", REFLECT_SH, "mission"],
                stdin=stdin_f,
                stdout=log_f,
                stderr=log_f,
                env=env,
                start_new_session=True,
            )
    except OSError:
        pass


def main():
    if DISABLE_FLAG.exists():
        sys.exit(0)
    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    session_id = data.get("session_id", "")
    transcript_path = data.get("transcript_path", "")
    if not session_id or not transcript_path or not os.path.exists(transcript_path):
        sys.exit(0)

    cwd = resolve_cwd(data, transcript_path)
    if not cwd or any(marker in cwd for marker in _SKIP_CWD_MARKERS):
        sys.exit(0)

    scan = context_fill.scan_transcript(transcript_path)
    max_tokens = context_fill.detect_max_tokens(
        scan["last_model"], scan["observed_max"], os.environ.get("CLAUDE_CONTEXT_MAX")
    )
    if context_fill.fill_pct(scan, max_tokens) < context_fill.THRESHOLD:
        sys.exit(0)

    mission_path = os.path.join(cwd, "tasks", "mission.md")
    if not mission_is_fresh(mission_path):
        refresh_mission(session_id, transcript_path, cwd)

    sys.exit(0)


if __name__ == "__main__":
    main()
```
