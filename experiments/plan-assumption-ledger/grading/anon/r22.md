## Assumptions

1. **Claude Code invokes hook commands with CWD = the session's project
   directory**, not `$HOME`. Both `ralph-loop`'s vendored `stop-hook.sh`
   and this repo's own `skills/orchestrate/hooks/compact-reinject.sh` rely

   silently no-ops everywhere (`RALPH_STATE.exists()` always false) — a
   safe degradation, not a corruption risk, but the feature would never
   fire. Task 7's integration test exercises this against the real
   `reflect.sh`; if it needs to be re-verified against a live Claude Code
   session later, check that `.claude/self-compact.local.json` appears in
   a real ralph-armed project after crossing 70% fill.
2. **Self-compact does not force an early native compact/clear.** The
   alternative design (have a Stop hook return `decision: block, reason:
   "/clear"` to force a reset) was rejected: it would compete with
   ralph-loop's own Stop hook for the block decision, and this repo has
   already documented empirically that only one Stop hook should be the
   blocker. Instead, self-compact treats Claude Code's own native
   auto-compaction as the "reset" — proven and already wired via the
   existing `PreCompact` / `SessionStart(compact)` hooks — and its whole
   contribution is making sure `tasks/mission.md` is fresh *before* that
   native reset happens, at 30%-remaining rather than reactively on a 2nd
   compact. If a future need for a genuinely forced early reset emerges,
   it should be scoped as its own follow-up plan, not bundled here.
3. **`reflect.sh`'s `mission` mode prompt (`prompts/mission.md`) is reused
   unchanged.** It already produces a well-specified, resumable handoff
   doc; this plan treats it as a black box and only fixes the dispatcher's
   idempotency handling around it.

```python
#!/usr/bin/env python3
"""
Shared context-window fill-percentage detection, used by context-baggage.py
(interactive advisory nudge) and self-compact.py (autonomous proactive
mission.md trigger). Single source of truth for the 1M-context model list,
the fill threshold, and the transcript-scanning arithmetic — divergence
between the two callers would mean one warns/acts at a different point
than the other for the exact same session.
"""
from __future__ import annotations

import json
import os
from dataclasses import dataclass

BASE_MAX = 200_000
ONE_M = 1_000_000

# Raised from 0.25 -> 0.70 on 2026-08-08 after measuring 95.5% cache-hit
# ratio across 7 days / $3,073 spend (probes/analyze_sessions.py). At that
# hit ratio, holding baggage is cheap and resetting is expensive
# (cache_write ~$5-10 per cold event). Nudge/act when handoff to a fresh
# session starts to pay off, not on any drift. Overridable so both callers
# stay in lockstep if ever tuned.
THRESHOLD = float(os.environ.get("CLAUDE_CONTEXT_COMPACT_THRESHOLD", "0.70"))

# Claude Code model IDs that run at 1M context by default. Prefixes, so
# dated variants (claude-opus-4-1-20250805) match too.
ONE_M_MODEL_PREFIXES = (
    "claude-opus-4-",
    "claude-sonnet-4-",
    "claude-opus-5",
    "claude-sonnet-5",
)

# Explicit 1M marker in the model ID, e.g. `claude-opus-5[1m]`. Checked
# before the prefix list so it keeps working across model generations
# without another edit here — the prefix list going stale is exactly how
# this hook came to report 89% full during a session that was at 18% of a
# 1M window (every reading was fullness-of-200k).
ONE_M_MARKER = "[1m]"


def detect_max_tokens(model: str | None, observed_max: int) -> int:
    override = os.environ.get("CLAUDE_CONTEXT_MAX")
    if override:
        try:
            return int(override)
        except ValueError:
            pass
    if model:
        if ONE_M_MARKER in model.lower():
            return ONE_M
        if any(model.startswith(p) for p in ONE_M_MODEL_PREFIXES):
            return ONE_M
    if observed_max > BASE_MAX:
        return ONE_M
    return BASE_MAX


@dataclass
class FillResult:
    fill_pct: float
    max_tokens: int
    last_model: str | None
    user_msg_count: int
    total_tokens: int  # 0 when no usage seen (fill_pct derived from file size)


def compute_fill(transcript_path: str) -> FillResult | None:
    """Scan a transcript JSONL and compute context fill percentage.

    Returns None if the transcript does not exist. Arithmetic matches
    context-baggage.py's pre-extraction behavior exactly — see
    tests/test_context_baggage.py for the golden-behavior regression check.
    """
    if not transcript_path or not os.path.exists(transcript_path):
        return None

    file_size = os.path.getsize(transcript_path)

    user_msg_count = 0
    last_usage = None
    last_model = None
    observed_max = 0

    with open(transcript_path, "r", errors="replace") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
            except json.JSONDecodeError:
                continue

            obj_type = obj.get("type")

            if obj_type == "user":
                user_msg_count += 1
            elif obj_type == "assistant":
                msg = obj.get("message", {})
                model = msg.get("model")
                if model and model != "<synthetic>":
                    last_model = model
                usage = msg.get("usage")
                if usage:
                    last_usage = usage
                    total = (
                        usage.get("input_tokens", 0)
                        + usage.get("cache_creation_input_tokens", 0)
                        + usage.get("cache_read_input_tokens", 0)
                    )
                    if total > observed_max:
                        observed_max = total

    max_tokens = detect_max_tokens(last_model, observed_max)

    if last_usage:
        total_tokens = (
            last_usage.get("input_tokens", 0)
            + last_usage.get("cache_creation_input_tokens", 0)
            + last_usage.get("cache_read_input_tokens", 0)
        )
        fill_pct = total_tokens / max_tokens
    else:
        total_tokens = 0
        fill_pct = file_size / (max_tokens * 4)

    return FillResult(
        fill_pct=fill_pct,
        max_tokens=max_tokens,
        last_model=last_model,
        user_msg_count=user_msg_count,
        total_tokens=total_tokens,
    )
```

```python
#!/usr/bin/env python3
"""
UserPromptSubmit hook: warns when context is over THRESHOLD with potentially
stale/off-topic content. Fires every MESSAGE_INTERVAL user messages once
threshold is met.

Context window detection and fill-percentage arithmetic live in
context_fill.py (shared with self-compact.py's autonomous trigger) so the
two callers can never disagree about when a session has crossed the
handoff-is-cheaper-than-compact line.
"""
import json
import sys

import context_fill

MESSAGE_INTERVAL = 5


def main():
    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    transcript_path = data.get("transcript_path", "")
    result = context_fill.compute_fill(transcript_path)
    if result is None:
        sys.exit(0)

    if result.user_msg_count == 0 or result.user_msg_count % MESSAGE_INTERVAL != 0:
        sys.exit(0)

    if result.fill_pct < context_fill.THRESHOLD:
        sys.exit(0)

    pct_display = int(result.fill_pct * 100)
    message = (
        f"Context is {pct_display}% full ({result.user_msg_count} messages in this session; "
        f"window: {result.max_tokens // 1000}k tokens). "
        "At this fill, a fresh-session handoff via tasks/mission.md is usually cheaper "
        "than an in-place /compact: --fork-session on a warm cache measured ~77% hit "
        "(session-reflect empirical), while /compact invalidates the prefix and forces "
        "cache_write on subsequent turns. "
        "Suggest to the user: (a) write tasks/mission.md (done / next / blocked / open "
        "questions), (b) start a fresh session in the same cwd and load it, OR (c) "
        "continue here and accept the drift. Ask which."
    )
    print(json.dumps({"hookSpecificOutput": {"hookEventName": "UserPromptSubmit", "additionalContext": message}}))


if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""
Stop hook: proactive self-compact trigger for autonomous (ralph-loop-armed)
sessions.

Interactive sessions already get an advisory nudge from context-baggage.py
(UserPromptSubmit) asking the user which of write-mission/fresh-session/
continue they want. Autonomous sessions have no one to ask — this hook
instead takes the safe default automatically: once context fill crosses
context_fill.THRESHOLD, keep tasks/mission.md continuously fresh (reusing
skills/session-reflect/reflect.sh's existing `mission` mode) so that
whenever Claude Code's own native compaction actually fires — first time
or Nth — a current, concrete handoff is already on disk instead of relying
on a lossy in-place summary.

This hook never blocks the Stop event (no `decision` output) and never
raises: only ralph-loop's own Stop hook is allowed to be the blocker
(see skills/orchestrate/SKILL.md "Multiple Stop hooks coexist fine").
"""
from __future__ import annotations

import json
import os
import subprocess
import sys
import time
from pathlib import Path

import context_fill

RALPH_STATE = Path(".claude/ralph-loop.local.md")
THROTTLE_STATE = Path(".claude/self-compact.local.json")

DEFAULT_REFLECT_SH = str(Path.home() / ".claude" / "skills" / "session-reflect" / "reflect.sh")


def _now_iso() -> str:
    return time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())


def _read_throttle() -> dict:
    try:
        return json.loads(THROTTLE_STATE.read_text())
    except (OSError, ValueError):
        return {}


def _write_throttle(fill_pct: float, status: str) -> None:
    prev = _read_throttle()
    record = {
        "last_triggered_at": _now_iso(),
        "last_fill_pct": round(fill_pct, 4),
        "last_status": status,
        "trigger_count": int(prev.get("trigger_count", 0)) + 1,
    }
    THROTTLE_STATE.parent.mkdir(parents=True, exist_ok=True)
    tmp = THROTTLE_STATE.with_suffix(f".tmp-{os.getpid()}")
    tmp.write_text(json.dumps(record))
    tmp.replace(THROTTLE_STATE)


def _throttled(refresh_seconds: int) -> bool:
    prev = _read_throttle()
    last_at = prev.get("last_triggered_at")
    if not last_at:
        return False
    try:
        last_epoch = time.mktime(time.strptime(last_at, "%Y-%m-%dT%H:%M:%SZ"))
    except ValueError:
        return False
    return (time.time() - last_epoch) < refresh_seconds


def main() -> None:
    try:
        data = json.load(sys.stdin)
    except Exception:
        sys.exit(0)

    session_id = data.get("session_id", "")
    transcript_path = data.get("transcript_path", "")
    if not session_id or not transcript_path:
        sys.exit(0)

    if not RALPH_STATE.exists():
        sys.exit(0)  # interactive session — advisory nudge (context-baggage.py) handles this case

    result = context_fill.compute_fill(transcript_path)
    if result is None or result.fill_pct < context_fill.THRESHOLD:
        sys.exit(0)

    refresh_seconds = int(os.environ.get("SELF_COMPACT_REFRESH_SECONDS", "600"))
    if _throttled(refresh_seconds):
        sys.exit(0)

    reflect_sh = os.environ.get("SELF_COMPACT_REFLECT_SH", DEFAULT_REFLECT_SH)
    subprocess_timeout = int(os.environ.get("SELF_COMPACT_TIMEOUT_SECONDS", "65"))
    payload = json.dumps(
        {
            "session_id": session_id,
            "transcript_path": transcript_path,
            "hook_event_name": "Stop",
            "cwd": os.getcwd(),
        }
    )

    status = "ok"
    try:
        proc = subprocess.run(
            ["bash", reflect_sh, "mission"],
            input=payload,
            capture_output=True,
            text=True,
            timeout=subprocess_timeout,
        )
        if proc.returncode != 0:
            status = "error"
    except subprocess.TimeoutExpired:
        status = "timeout"
    except OSError:
        status = "error"

    _write_throttle(result.fill_pct, status)
    sys.exit(0)


if __name__ == "__main__":
    main()
```
